defined_text = {
	name = GetPeMagRep
	text = {
		trigger = { check_variable = { which = pe_mag_rep value >= 2 } }
		localization_key = pe_mag_rep_high
	}
	text = {
		trigger = { check_variable = { which = pe_mag_rep value <= -2 } }
		localization_key = pe_mag_rep_low
	}
	default = pe_mag_rep_mixed
}
