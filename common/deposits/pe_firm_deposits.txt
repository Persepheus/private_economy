# Private firms on a planet. Each occupies one district slot (like a blocker) and adds
# private jobs the player cannot prioritise. Placed only by script (potential = no).
# One district in 4.5 gives about 100 workforce per job type; a firm gives 120 to
# model private efficiency.

# Single-sector firm: heavy industry.
d_pe_firm_steelworks = {
	icon = d_city
	is_for_colonizable = yes
	category = pe_deposit_cat_firms
	can_be_cleared = { always = no }
	potential = { always = no }
	drop_weight = { weight = 0 }

	planet_modifier = {
		planet_max_districts_add = -1
		job_pe_private_metallurgist_add = 120
	}
}
