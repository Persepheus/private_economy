# Private Economy triggers.

# Colony scope. Normal planets only for now; habitats, ring worlds, ecumenopolises and
# other special planets get their own solution later. The yearly cooldown (pe_cooldown)
# only limits actions; accounting runs every quarter.
pe_planet_eligible = {
	exists = owner
	is_colony = yes
	# An occupied planet's firms stand still (pe_occupation_step).
	is_occupied_flag = no
	is_under_colonization = no
	colony_age >= 6
	owner = { is_country_type = default is_gestalt = no }
	planet_is_habitat_equivalent = no
	planet_is_ecu_equivalent = no
	planet_is_ring_world_equivalent = no
	# Nomad arkships (their carrier is a ship) wait for the special-planet solution.
	NOT = { carrier_is_type = ship }
}

# Per-variant triggers (pe_can_found_/buy_/retool_<firm>): pe_firm_triggers_generated.txt.

# Country scope: private firms may buy state districts (Free Market and Laissez-faire).
pe_buyout_allowed = {
	OR = {
		has_policy_flag = pe_order_free_market
		has_policy_flag = pe_order_laissez_faire
	}
}

# Colony scope: idle workforce for new jobs. In 4.x jobless pops work the "civilian"
# job. Idle pops = civilians minus the jobs still open on the planet (pe_idle, set each
# quarter in pe_planet_step); at least 50 count.
pe_has_idle_pops = {
	check_variable = { which = pe_idle value >= 50 }
}

# Colony scope: private specialist jobs stand empty here (at least one job's worth).
pe_spec_short = {
	OR = {
		free_jobs_of_type = { job = pe_private_metallurgist value >= 100 }
		free_jobs_of_type = { job = pe_private_artisan value >= 100 }
		free_jobs_of_type = { job = pe_private_trader value >= 100 }
		free_jobs_of_type = { job = pe_private_physicist value >= 100 }
		free_jobs_of_type = { job = pe_private_biologist value >= 100 }
		free_jobs_of_type = { job = pe_private_engineer value >= 100 }
		free_jobs_of_type = { job = pe_private_bureaucrat value >= 100 }
	}
}

# Colony scope: private worker jobs stand empty here (at least one job's worth).
pe_rural_short = {
	OR = {
		free_jobs_of_type = { job = pe_private_miner value >= 100 }
		free_jobs_of_type = { job = pe_private_technician value >= 100 }
		free_jobs_of_type = { job = pe_private_farmer value >= 100 }
	}
}
