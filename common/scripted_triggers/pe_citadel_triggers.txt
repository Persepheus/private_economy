# ours with a starbase, no corporate citadel or site yet, and below the vanilla citadel limit.
pe_cit_system_ok = {
	exists = starbase
	exists = owner
	owner = { is_same_value = root }
	NOT = { has_star_flag = pe_cit_site }
	NOT = { any_fleet_in_system = { has_fleet_flag = pe_ccit } }
	count_ship_in_system = {
		limit = { is_space_citadel_ship_size = yes }
		count < root.modifier:deep_space_citadel_limit_add
	}
}
