# Corporate deep space citadels (pe_citadel_step). System scope, root = our empire: a system of
# ours with a starbase where corporation $K$ has no citadel and no site yet (one per corporation
# and system; other corporations may have theirs there too).
pe_cit_system_ok = {
	exists = starbase
	exists = owner
	owner = { is_same_value = root }
	NOT = { has_star_flag = pe_cit_site_$K$ }
	NOT = { any_fleet_in_system = { has_fleet_flag = pe_ccit is_variable_set = pe_ccit_k check_variable = { which = pe_ccit_k value = $K$ } } }
}
