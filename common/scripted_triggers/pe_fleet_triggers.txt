# Corporate fleets: space creatures in this system that are hostile to us (system scope,
# root = our country; hunting rights).
pe_monster_system = {
	any_fleet_in_system = { owner = { OR = { is_amoeba_country_type = yes is_crystal_country_type = yes is_drone_country_type = yes is_voidworms_hostile_country_type = yes } is_hostile = root } }
}

# Corporate fleets: pirates or marauder raiders in this system (system scope).
pe_pirate_system = {
	any_fleet_in_system = { owner = { OR = { is_country_type = pirate is_country_type = marauder_raiders } } }
}

# Corporate Dyson swarm: a single star without a megastructure, with an ordinary orbital
# deposit whose station the corporation can have (none yet, or ours) (system scope).
pe_swarm_system = {
	is_binary_star = no
	is_trinary_star = no
	NOR = {
		is_star_class = sc_black_hole
		is_star_class = sc_neutron_star
		is_star_class = sc_pulsar
	}
	any_system_planet = { pe_swarm_star = yes }
}

# Planet scope: a star for a corporate swarm (see pe_swarm_system).
pe_swarm_star = {
	is_star = yes
	NOT = { has_planet_flag = has_megastructure }
	pe_orbital_allowed = yes
	OR = {
		has_orbital_station = no
		pe_state_station = yes
	}
}

# Planet scope: a molten planet in our space for a corporate arc furnace (vanilla placement:
# no anomaly, no megastructure, no furnace or Dyson sphere in the system), not colonised,
# its station none or ours.
pe_furnace_site = {
	is_planet_class = pc_molten
	is_surveyed = { who = root status = yes }
	has_anomaly = no
	NOT = { exists = owner }
	NOR = { has_planet_flag = megastructure has_planet_flag = has_megastructure }
	system_has_arc_furnace = no
	system_has_dyson_sphere = no
	solar_system = { NOT = { has_star_flag = arc_furnace_construction } }
	OR = {
		has_orbital_station = no
		pe_state_station = yes
	}
}
