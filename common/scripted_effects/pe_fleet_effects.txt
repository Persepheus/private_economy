# Corporate fleets (pe_fleet_events.txt): fleets of a corporation, owned by our Private
# Holdings (with one of their leaders) and leased to us (set_controller). Upkeep for us: the
# 10% the engine cannot take off (MIN_ECONOMIC_MODIFIER_MULT), as a leasing fee; no naval
# capacity. Each kind keeps its own variables (prefix), so offers of one quarter do not mix:
#   pe_wl_  war lease (pe.282 / 283): at war, five years, then buy or return
#   pe_cl_  clearing (pe.284 / 285): space creatures next to our border, loot rights
#   pe_cv_  convoy protection (pe.287 / 288): pirates within our borders
#   pe_ch_  exploration charter (pe.286): a science ship for five years, mining rights
#   pe_sv_  salvage (pe.289); pe_sw_ corporate Dyson swarm (pe.292)
# Debug (pe.277, flag pe_fl_debug): offers only, running contracts are not advanced.
# Call depths: pe_corp_fleets, pe_corp_charter, pe_corp_convoy, pe_corp_salvage,
# pe_corp_swarm 3 (from pe_corp_step) -> the start and exec effects 4.

# Country scope, quarterly: war lease and clearing.
pe_corp_fleets = {
	# --- War lease: running ---
	if = {
		limit = { check_variable = { which = pe_lease_war_q value > 0 } NOT = { has_country_flag = pe_fl_debug } }
		change_variable = { which = pe_lease_war_q value = -1 }
		if = {
			limit = { NOT = { any_controlled_fleet = { has_fleet_flag = pe_lease_war } } }
			set_variable = { which = pe_lease_war_q value = 0 }
		}
		else_if = {
			limit = { check_variable = { which = pe_lease_war_q value <= 0 } }
			set_variable = { which = pe_fl_price value = pe_wl_ships }
			multiply_variable = { which = pe_fl_price value = pe_wl_cprice }
			if = {
				limit = { is_ai = yes }
				if = {
					limit = { has_resource = { type = alloys amount >= 500 } }
					pe_corp_fleet_buy = yes
				}
				else = { every_controlled_fleet = { limit = { has_fleet_flag = pe_lease_war } if = { limit = { exists = leader } kill_leader = { show_notification = no fire = yes } } delete_fleet = this } }
			}
			else = { country_event = { id = pe.283 } }
		}
	}
	# --- War lease: an offer (at war, an industry corporation with 600 cash, otherwise any
	# with 1200, 30% a quarter, then five years' pause) ---
	else_if = {
		limit = {
			NOT = { check_variable = { which = pe_lease_war_q value > 0 } }
			OR = { is_at_war = yes has_country_flag = pe_fl_debug }
			exists = event_target:pe_hold
			NOT = { has_country_flag = pe_lease_war_cd }
		}
		set_variable = { which = pe_wl_k value = 0 }
		set_variable = { which = pe_wl_best value = 599 }
		if = { limit = { check_variable = { which = pe_c1_on value = 1 } check_variable = { which = pe_c1_grp value = 1 } check_variable = { which = pe_c1_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 1 } set_variable = { which = pe_wl_best value = pe_c1_cash } }
		if = { limit = { check_variable = { which = pe_c2_on value = 1 } check_variable = { which = pe_c2_grp value = 1 } check_variable = { which = pe_c2_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 2 } set_variable = { which = pe_wl_best value = pe_c2_cash } }
		if = { limit = { check_variable = { which = pe_c3_on value = 1 } check_variable = { which = pe_c3_grp value = 1 } check_variable = { which = pe_c3_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 3 } set_variable = { which = pe_wl_best value = pe_c3_cash } }
		if = { limit = { check_variable = { which = pe_c4_on value = 1 } check_variable = { which = pe_c4_grp value = 1 } check_variable = { which = pe_c4_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 4 } set_variable = { which = pe_wl_best value = pe_c4_cash } }
		if = { limit = { check_variable = { which = pe_c5_on value = 1 } check_variable = { which = pe_c5_grp value = 1 } check_variable = { which = pe_c5_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 5 } set_variable = { which = pe_wl_best value = pe_c5_cash } }
		if = { limit = { check_variable = { which = pe_c6_on value = 1 } check_variable = { which = pe_c6_grp value = 1 } check_variable = { which = pe_c6_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 6 } set_variable = { which = pe_wl_best value = pe_c6_cash } }
		if = { limit = { check_variable = { which = pe_c7_on value = 1 } check_variable = { which = pe_c7_grp value = 1 } check_variable = { which = pe_c7_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 7 } set_variable = { which = pe_wl_best value = pe_c7_cash } }
		if = { limit = { check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_grp value = 1 } check_variable = { which = pe_c8_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 8 } set_variable = { which = pe_wl_best value = pe_c8_cash } }
		if = {
			limit = { check_variable = { which = pe_wl_k value = 0 } }
			set_variable = { which = pe_wl_k value = 0 }
			set_variable = { which = pe_wl_best value = 1199 }
			if = { limit = { check_variable = { which = pe_c1_on value = 1 }  check_variable = { which = pe_c1_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 1 } set_variable = { which = pe_wl_best value = pe_c1_cash } }
			if = { limit = { check_variable = { which = pe_c2_on value = 1 }  check_variable = { which = pe_c2_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 2 } set_variable = { which = pe_wl_best value = pe_c2_cash } }
			if = { limit = { check_variable = { which = pe_c3_on value = 1 }  check_variable = { which = pe_c3_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 3 } set_variable = { which = pe_wl_best value = pe_c3_cash } }
			if = { limit = { check_variable = { which = pe_c4_on value = 1 }  check_variable = { which = pe_c4_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 4 } set_variable = { which = pe_wl_best value = pe_c4_cash } }
			if = { limit = { check_variable = { which = pe_c5_on value = 1 }  check_variable = { which = pe_c5_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 5 } set_variable = { which = pe_wl_best value = pe_c5_cash } }
			if = { limit = { check_variable = { which = pe_c6_on value = 1 }  check_variable = { which = pe_c6_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 6 } set_variable = { which = pe_wl_best value = pe_c6_cash } }
			if = { limit = { check_variable = { which = pe_c7_on value = 1 }  check_variable = { which = pe_c7_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 7 } set_variable = { which = pe_wl_best value = pe_c7_cash } }
			if = { limit = { check_variable = { which = pe_c8_on value = 1 }  check_variable = { which = pe_c8_cash value > pe_wl_best } } set_variable = { which = pe_wl_k value = 8 } set_variable = { which = pe_wl_best value = pe_c8_cash } }
		}
		if = {
			limit = { check_variable = { which = pe_wl_k value > 0 } }
			random_list = {
				30 = {
					set_timed_country_flag = { flag = pe_lease_war_cd days = 1800 }
					set_variable = { which = pe_wl_budget value = pe_wl_best }
					multiply_variable = { which = pe_wl_budget value = 0.5 }
					export_trigger_value_to_variable = { trigger = fleet_power variable = pe_wl_need }
					multiply_variable = { which = pe_wl_need value = 0.25 }
					if = { limit = { check_variable = { which = pe_wl_need value < 500 } } set_variable = { which = pe_wl_need value = 500 } }
					capital_scope = { save_event_target_as = pe_wl_loc }
					if = {
						limit = { is_ai = yes }
						random = { chance = 80 pe_corp_fleet_war_start = yes }
					}
					else = { country_event = { id = pe.282 } }
				}
				70 = { modifier = { factor = 0 check_variable = { which = pe_fl_force value = 1 } } }
			}
		}
	}
	# --- Clearing: running ---
	if = {
		limit = { check_variable = { which = pe_lease_clear_q value > 0 } NOT = { has_country_flag = pe_fl_debug } }
		change_variable = { which = pe_lease_clear_q value = -1 }
		set_variable = { which = pe_cl_done value = 0 }
		if = {
			limit = { NOT = { any_controlled_fleet = { has_fleet_flag = pe_lease_clear } } }
			set_variable = { which = pe_cl_done value = 2 }
		}
		else_if = {
			limit = { any_system = { has_star_flag = pe_clear@root NOT = { pe_monster_system = yes } } }
			set_variable = { which = pe_cl_done value = 1 }
		}
		else_if = {
			limit = { check_variable = { which = pe_lease_clear_q value <= 0 } }
			set_variable = { which = pe_cl_done value = 2 }
		}
		if = {
			limit = { check_variable = { which = pe_cl_done value > 0 } }
			random_system = {
				limit = { has_star_flag = pe_clear@root }
				save_event_target_as = pe_cl_sys
				remove_star_flag = pe_clear@root
			}
			every_controlled_fleet = { limit = { has_fleet_flag = pe_lease_clear } if = { limit = { exists = leader } kill_leader = { show_notification = no fire = yes } } delete_fleet = this }
			set_variable = { which = pe_lease_clear_q value = 0 }
			if = {
				limit = { check_variable = { which = pe_cl_done value = 1 } }
				# The loot: a third of the creatures' power for the corporation in cash (plus its
				# ships back), a twentieth for us as society research (at least 25).
				set_variable = { which = pe_cl_loot value = pe_cl_power }
				multiply_variable = { which = pe_cl_loot value = 0.35 }
				change_variable = { which = pe_cl_loot value = pe_cl_spent }
				round_variable = pe_cl_loot
				set_variable = { which = pe_cl_res value = pe_cl_power }
				multiply_variable = { which = pe_cl_res value = 0.05 }
				if = { limit = { check_variable = { which = pe_cl_res value < 25 } } set_variable = { which = pe_cl_res value = 25 } }
				round_variable = pe_cl_res
				if = { limit = { check_variable = { which = pe_cl_run value = 1 } } change_variable = { which = pe_c1_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 2 } } change_variable = { which = pe_c2_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 3 } } change_variable = { which = pe_c3_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 4 } } change_variable = { which = pe_c4_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 5 } } change_variable = { which = pe_c5_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 6 } } change_variable = { which = pe_c6_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 7 } } change_variable = { which = pe_c7_cash value = pe_cl_loot } }
				else_if = { limit = { check_variable = { which = pe_cl_run value = 8 } } change_variable = { which = pe_c8_cash value = pe_cl_loot } }
				add_resource = { society_research = 1 mult = pe_cl_res }
			}
			if = { limit = { is_ai = no } country_event = { id = pe.285 } }
		}
	}
	# --- Clearing: an offer (creatures next to our border, a corporation with 600 cash, 25%,
	# then four years' pause) ---
	else_if = {
		limit = {
			NOT = { check_variable = { which = pe_lease_clear_q value > 0 } }
			exists = event_target:pe_hold
			NOT = { has_country_flag = pe_lease_clear_cd }
			any_system_within_border = { any_neighbor_system = { NOT = { exists = owner } pe_monster_system = yes } }
		}
		set_variable = { which = pe_cl_k value = 0 }
		set_variable = { which = pe_cl_best value = 599 }
		if = { limit = { check_variable = { which = pe_c1_on value = 1 }  check_variable = { which = pe_c1_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 1 } set_variable = { which = pe_cl_best value = pe_c1_cash } }
		if = { limit = { check_variable = { which = pe_c2_on value = 1 }  check_variable = { which = pe_c2_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 2 } set_variable = { which = pe_cl_best value = pe_c2_cash } }
		if = { limit = { check_variable = { which = pe_c3_on value = 1 }  check_variable = { which = pe_c3_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 3 } set_variable = { which = pe_cl_best value = pe_c3_cash } }
		if = { limit = { check_variable = { which = pe_c4_on value = 1 }  check_variable = { which = pe_c4_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 4 } set_variable = { which = pe_cl_best value = pe_c4_cash } }
		if = { limit = { check_variable = { which = pe_c5_on value = 1 }  check_variable = { which = pe_c5_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 5 } set_variable = { which = pe_cl_best value = pe_c5_cash } }
		if = { limit = { check_variable = { which = pe_c6_on value = 1 }  check_variable = { which = pe_c6_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 6 } set_variable = { which = pe_cl_best value = pe_c6_cash } }
		if = { limit = { check_variable = { which = pe_c7_on value = 1 }  check_variable = { which = pe_c7_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 7 } set_variable = { which = pe_cl_best value = pe_c7_cash } }
		if = { limit = { check_variable = { which = pe_c8_on value = 1 }  check_variable = { which = pe_c8_cash value > pe_cl_best } } set_variable = { which = pe_cl_k value = 8 } set_variable = { which = pe_cl_best value = pe_c8_cash } }
		if = {
			limit = { check_variable = { which = pe_cl_k value > 0 } }
			random_list = {
				25 = {
					set_timed_country_flag = { flag = pe_lease_clear_cd days = 1440 }
					random_system_within_border = {
						limit = { any_neighbor_system = { NOT = { exists = owner } pe_monster_system = yes } }
						save_event_target_as = pe_cl_loc
						random_neighbor_system = {
							limit = { NOT = { exists = owner } pe_monster_system = yes }
							save_event_target_as = pe_cl_sys
						}
					}
					set_variable = { which = pe_cl_power value = 0 }
					event_target:pe_cl_sys = {
						every_fleet_in_system = {
							limit = { owner = { OR = { is_amoeba_country_type = yes is_crystal_country_type = yes is_drone_country_type = yes is_voidworms_hostile_country_type = yes } } }
							export_trigger_value_to_variable = { trigger = fleet_power variable = pe_tmp }
							root = { change_variable = { which = pe_cl_power value = prev.pe_tmp } }
						}
					}
					set_variable = { which = pe_cl_need value = pe_cl_power }
					multiply_variable = { which = pe_cl_need value = 1.5 }
					set_variable = { which = pe_cl_budget value = pe_cl_best }
					multiply_variable = { which = pe_cl_budget value = 0.8 }
					if = {
						limit = { is_ai = yes }
						random = { chance = 80 pe_corp_fleet_clear_start = yes }
					}
					else = { country_event = { id = pe.284 } }
				}
				75 = { modifier = { factor = 0 check_variable = { which = pe_fl_force value = 1 } } }
			}
		}
	}
}

# Country scope: start the war lease (five years).
pe_corp_fleet_war_start = {
	set_variable = { which = pe_fl_class value = 1 }
	if = { limit = { has_technology = tech_destroyers } set_variable = { which = pe_fl_class value = 2 } }
	if = { limit = { has_technology = tech_cruisers } set_variable = { which = pe_fl_class value = 3 } }
	if = { limit = { has_technology = tech_battleships } set_variable = { which = pe_fl_class value = 4 } }
	if = { limit = { check_variable = { which = pe_fl_class value = 1 } } set_variable = { which = pe_fl_ccost value = 150 } set_variable = { which = pe_fl_cprice value = 40 } }
	if = { limit = { check_variable = { which = pe_fl_class value = 2 } } set_variable = { which = pe_fl_ccost value = 300 } set_variable = { which = pe_fl_cprice value = 80 } }
	if = { limit = { check_variable = { which = pe_fl_class value = 3 } } set_variable = { which = pe_fl_ccost value = 600 } set_variable = { which = pe_fl_cprice value = 160 } }
	if = { limit = { check_variable = { which = pe_fl_class value = 4 } } set_variable = { which = pe_fl_ccost value = 1200 } set_variable = { which = pe_fl_cprice value = 320 } }
	set_variable = { which = pe_wl_spent value = 0 }
	set_variable = { which = pe_wl_ships value = 0 }
	create_fleet = {
		name = pe_corp_fleet_name
		settings = { uses_naval_capacity = no }
		effect = {
			set_owner = root
			set_variable = { which = pe_fp value = 0 }
			while = {
				count = 12
				limit = {
					check_variable = { which = pe_fp value < root.pe_wl_need }
					root = { check_variable = { which = pe_wl_spent value < pe_wl_budget } }
				}
				if = { limit = { root = { check_variable = { which = pe_fl_class value = 1 } } } create_ship = { name = random random_existing_design = corvette } }
				else_if = { limit = { root = { check_variable = { which = pe_fl_class value = 2 } } } create_ship = { name = random random_existing_design = destroyer } }
				else_if = { limit = { root = { check_variable = { which = pe_fl_class value = 3 } } } create_ship = { name = random random_existing_design = cruiser } }
				else_if = { limit = { root = { check_variable = { which = pe_fl_class value = 4 } } } create_ship = { name = random random_existing_design = battleship } }
				root = {
					change_variable = { which = pe_wl_spent value = pe_fl_ccost }
					change_variable = { which = pe_wl_ships value = 1 }
				}
				export_trigger_value_to_variable = { trigger = fleet_power variable = pe_fp }
			}
			set_location = event_target:pe_wl_loc
			set_fleet_flag = pe_leased
			set_fleet_flag = pe_lease_war
		}
	}
	# The leader is the holdings' own; assigned while they own and control the fleet, then leased.
	event_target:pe_hold = {
		create_leader = { class = commander species = root name = random skill = 2 }
		last_created_leader = { save_event_target_as = pe_fl_leader }
	}
	last_created_fleet = {
		set_owner = event_target:pe_hold
		assign_leader = event_target:pe_fl_leader
		set_controller = root
		add_modifier = { modifier = pe_leased_fleet days = -1 }
		if = { limit = { NOT = { exists = leader } } log = "private_economy: corporate fleet without a leader (assign_leader failed)" }
	}
	if = { limit = { check_variable = { which = pe_wl_k value = 1 } } subtract_variable = { which = pe_c1_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 2 } } subtract_variable = { which = pe_c2_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 3 } } subtract_variable = { which = pe_c3_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 4 } } subtract_variable = { which = pe_c4_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 5 } } subtract_variable = { which = pe_c5_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 6 } } subtract_variable = { which = pe_c6_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 7 } } subtract_variable = { which = pe_c7_cash value = pe_wl_spent } }
	else_if = { limit = { check_variable = { which = pe_wl_k value = 8 } } subtract_variable = { which = pe_c8_cash value = pe_wl_spent } }
	set_variable = { which = pe_lease_war_q value = 20 }
	set_variable = { which = pe_wl_cprice value = pe_fl_cprice }
}

# --- Convoy protection (pe.287 / pe.288) ------------------------------------------------------
# Pirates or marauder raiders in a system within our borders: a corporation with 600 cash lends
# a patrol of 1.5 times their strength (80% of its cash at most), three years at most. Once the
# system is free the patrol returns; the corporation keeps the loot (a third of their power).
# 35% a quarter, then three years' pause.
pe_corp_convoy = {
	if = {
		limit = { check_variable = { which = pe_convoy_q value > 0 } NOT = { has_country_flag = pe_fl_debug } }
		change_variable = { which = pe_convoy_q value = -1 }

			limit = { NOT = { any_controlled_fleet = { has_fleet_flag = pe_lease_convoy } } }
			set_variable = { which = pe_cv_done value = 2 }
		}
		else_if = {
			limit = { any_system = { has_star_flag = pe_convoy@root NOT = { pe_pirate_system = yes } } }
			set_variable = { which = pe_cv_done value = 1 }
		}
		else_if = {
			limit = { check_variable = { which = pe_convoy_q value <= 0 } }
			set_variable = { which = pe_cv_done value = 2 }
		}
		if = {
			limit = { check_variable = { which = pe_cv_done value > 0 } }
			random_system = {
				limit = { has_star_flag = pe_convoy@root }
				save_event_target_as = pe_cv_sys
				remove_star_flag = pe_convoy@root
			}
			every_controlled_fleet = { limit = { has_fleet_flag = pe_lease_convoy } if = { limit = { exists = leader } kill_leader = { show_notification = no fire = yes } } delete_fleet = this }

				set_variable = { which = pe_cv_loot value = pe_cv_power }
				multiply_variable = { which = pe_cv_loot value = 0.35 }
				change_variable = { which = pe_cv_loot value = pe_cv_spent }
				round_variable = pe_cv_loot
				if = { limit = { check_variable = { which = pe_cv_run value = 1 } } change_variable = { which = pe_c1_cash value = pe_cv_loot } }

				else_if = { limit = { check_variable = { which = pe_cv_run value = 3 } } change_variable = { which = pe_c3_cash value = pe_cv_loot } }
				else_if = { limit = { check_variable = { which = pe_cv_run value = 4 } } change_variable = { which = pe_c4_cash value = pe_cv_loot } }
				else_if = { limit = { check_variable = { which = pe_cv_run value = 5 } } change_variable = { which = pe_c5_cash value = pe_cv_loot } }
				else_if = { limit = { check_variable = { which = pe_cv_run value = 6 } } change_variable = { which = pe_c6_cash value = pe_cv_loot } }
				else_if = { limit = { check_variable = { which = pe_cv_run value = 7 } } change_variable = { which = pe_c7_cash value = pe_cv_loot } }
				else_if = { limit = { check_variable = { which = pe_cv_run value = 8 } } change_variable = { which = pe_c8_cash value = pe_cv_loot } }
			}
			if = { limit = { is_ai = no } country_event = { id = pe.288 } }
		}
	}
	else_if = {
		limit = {
			NOT = { check_variable = { which = pe_convoy_q value > 0 } }
			exists = event_target:pe_hold
			NOT = { has_country_flag = pe_convoy_cd }
			any_system_within_border = { pe_pirate_system = yes }
		}
		set_variable = { which = pe_cv_k value = 0 }
		set_variable = { which = pe_cv_best value = 599 }
		if = { limit = { check_variable = { which = pe_c1_on value = 1 }  check_variable = { which = pe_c1_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 1 } set_variable = { which = pe_cv_best value = pe_c1_cash } }
		if = { limit = { check_variable = { which = pe_c2_on value = 1 }  check_variable = { which = pe_c2_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 2 } set_variable = { which = pe_cv_best value = pe_c2_cash } }
		if = { limit = { check_variable = { which = pe_c3_on value = 1 }  check_variable = { which = pe_c3_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 3 } set_variable = { which = pe_cv_best value = pe_c3_cash } }
		if = { limit = { check_variable = { which = pe_c4_on value = 1 }  check_variable = { which = pe_c4_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 4 } set_variable = { which = pe_cv_best value = pe_c4_cash } }
		if = { limit = { check_variable = { which = pe_c5_on value = 1 }  check_variable = { which = pe_c5_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 5 } set_variable = { which = pe_cv_best value = pe_c5_cash } }

		if = { limit = { check_variable = { which = pe_c7_on value = 1 }  check_variable = { which = pe_c7_cash value > pe_cv_best } } set_variable = { which = pe_cv_k value = 7 } set_variable = { which = pe_cv_best value = pe_c7_cash } }
