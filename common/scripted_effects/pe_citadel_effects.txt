# Corporate deep space citadels (pe.315, common/megastructures/pe_megastructures.txt).
# Any corporation that can afford it builds a deep space citadel in its important systems:
# headquarters, corporate habitat, Dyson swarm or arc furnace, a system with two of its stations
# (in this order). It belongs to our
# Private Holdings (the state cannot upgrade or dismantle it); the holdings join our wars, so it
# fights for us. Only with our technology Deep Space Citadel (the holdings get our technologies).
# At most one per corporation and system (two headquarters in a system: two citadels).
# Build: a site of the holdings, built up with the vanilla build time (3 years); then the vanilla
# citadel appears (deep_space_citadel_0). Levels II and III are starbase levels of it (vanilla
# next_level, upgraded like a starbase): set_starbase_size after the vanilla build time. Level II / III when the corporation can afford it
# (vanilla build time), defence platforms while it can afford them (4 per level). Costs from
# its cash (alloys x4, unity x3: 11500 / 17500 / 35000), never more than a third of it; the
# upkeep (level I 10 energy 5 alloys, II 15 / 10, III 30 / 20, platforms 2 energy each) goes to
# the holdings, the corporation pays its value. A citadel whose corporation is gone or whose system
# is no longer ours goes to the richest other corporation, otherwise it is decommissioned.
# Citadel fleet (holdings): flag pe_ccit, pe_ccit_k slot, pe_ccit_nm name, pe_ccit_l level,
# pe_ccit_ul months left of an upgrade, pe_ccit_p platforms, pe_ccit_new (just built).
# Call depths: pe_citadel_step 3 (pe_corp_step) -> pe_corp_msg 4; pe_cit_exec 1 (pe.315 /
# pe.319, with the start message); pe_cit_war 1 (pe.100).

# Country scope, quarterly.
pe_citadel_step = {
	if = {
		limit = { exists = event_target:pe_hold }
		# The heir of an orphaned citadel: the richest corporation.
		set_variable = { which = pe_cit_heir value = 0 }
		set_variable = { which = pe_cit_hbest value = -1000000 }
		if = { limit = { check_variable = { which = pe_c1_on value = 1 } check_variable = { which = pe_c1_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 1 } set_variable = { which = pe_cit_hbest value = pe_c1_cash } set_variable = { which = pe_cit_hnm value = pe_c1_name } }
		if = { limit = { check_variable = { which = pe_c2_on value = 1 } check_variable = { which = pe_c2_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 2 } set_variable = { which = pe_cit_hbest value = pe_c2_cash } set_variable = { which = pe_cit_hnm value = pe_c2_name } }
		if = { limit = { check_variable = { which = pe_c3_on value = 1 } check_variable = { which = pe_c3_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 3 } set_variable = { which = pe_cit_hbest value = pe_c3_cash } set_variable = { which = pe_cit_hnm value = pe_c3_name } }
		if = { limit = { check_variable = { which = pe_c4_on value = 1 } check_variable = { which = pe_c4_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 4 } set_variable = { which = pe_cit_hbest value = pe_c4_cash } set_variable = { which = pe_cit_hnm value = pe_c4_name } }
		if = { limit = { check_variable = { which = pe_c5_on value = 1 } check_variable = { which = pe_c5_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 5 } set_variable = { which = pe_cit_hbest value = pe_c5_cash } set_variable = { which = pe_cit_hnm value = pe_c5_name } }
		if = { limit = { check_variable = { which = pe_c6_on value = 1 } check_variable = { which = pe_c6_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 6 } set_variable = { which = pe_cit_hbest value = pe_c6_cash } set_variable = { which = pe_cit_hnm value = pe_c6_name } }
		if = { limit = { check_variable = { which = pe_c7_on value = 1 } check_variable = { which = pe_c7_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 7 } set_variable = { which = pe_cit_hbest value = pe_c7_cash } set_variable = { which = pe_cit_hnm value = pe_c7_name } }
		if = { limit = { check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_cash value > pe_cit_hbest } } set_variable = { which = pe_cit_heir value = 8 } set_variable = { which = pe_cit_hbest value = pe_c8_cash } set_variable = { which = pe_cit_hnm value = pe_c8_name } }
		set_variable = { which = pe_c1_citn value = 0 }
		set_variable = { which = pe_c2_citn value = 0 }
		set_variable = { which = pe_c3_citn value = 0 }
		set_variable = { which = pe_c4_citn value = 0 }
		set_variable = { which = pe_c5_citn value = 0 }
		set_variable = { which = pe_c6_citn value = 0 }
		set_variable = { which = pe_c7_citn value = 0 }
		set_variable = { which = pe_c8_citn value = 0 }
		set_variable = { which = pe_cit_ge value = 0 }
		set_variable = { which = pe_cit_ga value = 0 }
		event_target:pe_hold = {
			every_owned_fleet = {
				limit = { has_fleet_flag = pe_ccit }
				if = { limit = { NOT = { is_variable_set = pe_ccit_l } } set_variable = { which = pe_ccit_l value = 1 } set_variable = { which = pe_ccit_ul value = 0 } set_variable = { which = pe_ccit_p value = 0 } }
				# Its corporation still there?
				set_variable = { which = pe_ccit_ok value = 0 }
				if = { limit = { check_variable = { which = pe_ccit_k value = 1 } root = { check_variable = { which = pe_c1_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c1_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 2 } root = { check_variable = { which = pe_c2_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c2_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 3 } root = { check_variable = { which = pe_c3_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c3_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 4 } root = { check_variable = { which = pe_c4_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c4_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 5 } root = { check_variable = { which = pe_c5_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c5_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 6 } root = { check_variable = { which = pe_c6_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c6_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 7 } root = { check_variable = { which = pe_c7_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c7_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = { limit = { check_variable = { which = pe_ccit_k value = 8 } root = { check_variable = { which = pe_c8_on value = 1 } } check_variable = { which = pe_ccit_nm value = root.pe_c8_name } } set_variable = { which = pe_ccit_ok value = 1 } }
				if = {
					limit = { check_variable = { which = pe_ccit_ok value = 1 } solar_system = { exists = owner owner = { is_same_value = root } } }
				}
				# The system is no longer ours, or the corporation is gone: the heir, or nobody.
				else_if = {
					limit = { root = { check_variable = { which = pe_cit_heir value > 0 } } solar_system = { exists = owner owner = { is_same_value = root } } }
					set_variable = { which = pe_ccit_k value = root.pe_cit_heir }
					set_variable = { which = pe_ccit_nm value = root.pe_cit_hnm }
					set_variable = { which = pe_ccit_ok value = 1 }
					if = {
						limit = { root = { is_ai = no } }
						solar_system = { star = { save_event_target_as = pe_msg_planet } }
						root = { set_variable = { which = pe_msg_name value = pe_cit_hnm } if = { limit = { NOT = { has_country_flag = pe_mute_military } } pe_corp_msg = { KEY = pe_msg_cit_heir CATEGORY = military } } }
					}
				}
				else = {
					if = {
						limit = { root = { is_ai = no } }
						solar_system = { star = { save_event_target_as = pe_msg_planet } }
						root = { if = { limit = { NOT = { has_country_flag = pe_mute_military } } create_message = { type = pe_private_economy_military localization = pe_msg_cit_lost days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } } } }
					}
					set_fleet_flag = pe_ccit_kill
				}
				if = {
					limit = { check_variable = { which = pe_ccit_ok value = 1 } }
					# Just built: message.
					if = {
						limit = { is_variable_set = pe_ccit_new check_variable = { which = pe_ccit_new value = 1 } }
						set_variable = { which = pe_ccit_new value = 0 }
						if = {
							limit = { root = { is_ai = no } }
							solar_system = { star = { save_event_target_as = pe_msg_planet } }
							root = { set_variable = { which = pe_msg_name value = prev.pe_ccit_nm } if = { limit = { NOT = { has_country_flag = pe_mute_military } } pe_corp_msg = { KEY = pe_msg_cit_done CATEGORY = military } } }
						}
					}
					# Upkeep for the next three months: to the holdings; the corporation pays its value.
					if = { limit = { check_variable = { which = pe_ccit_l value = 1 } } set_variable = { which = pe_ccit_e value = 10 } set_variable = { which = pe_ccit_a value = 5 } }
					else_if = { limit = { check_variable = { which = pe_ccit_l value = 2 } } set_variable = { which = pe_ccit_e value = 15 } set_variable = { which = pe_ccit_a value = 10 } }
					else_if = { limit = { check_variable = { which = pe_ccit_l value = 3 } } set_variable = { which = pe_ccit_e value = 30 } set_variable = { which = pe_ccit_a value = 20 } }
					set_variable = { which = pe_tmp value = pe_ccit_p }
					multiply_variable = { which = pe_tmp value = 2 }
					change_variable = { which = pe_ccit_e value = pe_tmp }
					multiply_variable = { which = pe_ccit_e value = 3 }
					multiply_variable = { which = pe_ccit_a value = 3 }
					root = { change_variable = { which = pe_cit_ge value = prev.pe_ccit_e } change_variable = { which = pe_cit_ga value = prev.pe_ccit_a } }
					set_variable = { which = pe_ccit_v value = pe_ccit_a }
					multiply_variable = { which = pe_ccit_v value = 4 }
					change_variable = { which = pe_ccit_v value = pe_ccit_e }
					if = { limit = { check_variable = { which = pe_ccit_k value = 1 } } root = { subtract_variable = { which = pe_c1_cash value = prev.pe_ccit_v } change_variable = { which = pe_c1_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c1_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 2 } } root = { subtract_variable = { which = pe_c2_cash value = prev.pe_ccit_v } change_variable = { which = pe_c2_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c2_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 3 } } root = { subtract_variable = { which = pe_c3_cash value = prev.pe_ccit_v } change_variable = { which = pe_c3_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c3_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 4 } } root = { subtract_variable = { which = pe_c4_cash value = prev.pe_ccit_v } change_variable = { which = pe_c4_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c4_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 5 } } root = { subtract_variable = { which = pe_c5_cash value = prev.pe_ccit_v } change_variable = { which = pe_c5_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c5_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 6 } } root = { subtract_variable = { which = pe_c6_cash value = prev.pe_ccit_v } change_variable = { which = pe_c6_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c6_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 7 } } root = { subtract_variable = { which = pe_c7_cash value = prev.pe_ccit_v } change_variable = { which = pe_c7_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c7_cash } } }
					else_if = { limit = { check_variable = { which = pe_ccit_k value = 8 } } root = { subtract_variable = { which = pe_c8_cash value = prev.pe_ccit_v } change_variable = { which = pe_c8_citn value = 1 } set_variable = { which = pe_cit_cash value = pe_c8_cash } } }
					# The design of its level (set_starbase_size keeps the old one): only the citadel's own
					# ship, not the platforms (pe_ccit_dl = the level its design is for).
					if = { limit = { NOT = { is_variable_set = pe_ccit_dl } } set_variable = { which = pe_ccit_dl value = 1 } }
					if = {
						limit = { NOT = { check_variable = { which = pe_ccit_dl value = pe_ccit_l } } }
						if = {
							limit = { check_variable = { which = pe_ccit_l value = 2 } }
							every_owned_ship = { limit = { is_ship_size = starbase_deep_space_citadel_2 } set_ship_design = { random_existing_design = starbase_deep_space_citadel_2 } }
						}
						else_if = {
							limit = { check_variable = { which = pe_ccit_l value = 3 } }
							every_owned_ship = { limit = { is_ship_size = starbase_deep_space_citadel_3 } set_ship_design = { random_existing_design = starbase_deep_space_citadel_3 } }
						}
						set_variable = { which = pe_ccit_dl value = pe_ccit_l }
						log = "private_economy: corporate citadel design set to level \\[This.pe_ccit_l]"
					}
					# An upgrade under way.
					if = {
						limit = { check_variable = { which = pe_ccit_ul value > 0 } }
						change_variable = { which = pe_ccit_ul value = -3 }
						if = {
							limit = { check_variable = { which = pe_ccit_ul value <= 0 } }
							set_variable = { which = pe_ccit_ul value = 0 }
							change_variable = { which = pe_ccit_l value = 1 }
							if = { limit = { check_variable = { which = pe_ccit_l value = 2 } } starbase = { set_starbase_size = starbase_deep_space_citadel_2 } }
							else_if = { limit = { check_variable = { which = pe_ccit_l value = 3 } } starbase = { set_starbase_size = starbase_deep_space_citadel_3 } }
							set_variable = { which = pe_ccit_dl value = 0 }
							log = "private_economy: corporate citadel upgrade finished: slot \\[This.pe_ccit_k], level \\[This.pe_ccit_l]"
							if = {
								limit = { root = { is_ai = no } }
								solar_system = { star = { save_event_target_as = pe_msg_planet } }
								root = { set_variable = { which = pe_msg_name value = prev.pe_ccit_nm } if = { limit = { NOT = { has_country_flag = pe_mute_military } } pe_corp_msg = { KEY = pe_msg_cit_up CATEGORY = military } } }
							}
						}
					}
					# The next level as soon as it can afford it (a third of its cash).
					else_if = {
						limit = { check_variable = { which = pe_ccit_l value < 3 } }
						if = { limit = { check_variable = { which = pe_ccit_l value = 1 } } set_variable = { which = pe_ccit_c value = 17500 } set_variable = { which = pe_ccit_m value = 36 } }
						else = { set_variable = { which = pe_ccit_c value = 35000 } set_variable = { which = pe_ccit_m value = 48 } }
						set_variable = { which = pe_tmp value = pe_ccit_c }
						multiply_variable = { which = pe_tmp value = 3 }
						if = {
							limit = { root = { check_variable = { which = pe_cit_cash value >= prev.pe_tmp } } }
							set_variable = { which = pe_ccit_ul value = pe_ccit_m }
							if = { limit = { check_variable = { which = pe_ccit_k value = 1 } } root = { subtract_variable = { which = pe_c1_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 2 } } root = { subtract_variable = { which = pe_c2_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 3 } } root = { subtract_variable = { which = pe_c3_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 4 } } root = { subtract_variable = { which = pe_c4_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 5 } } root = { subtract_variable = { which = pe_c5_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 6 } } root = { subtract_variable = { which = pe_c6_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 7 } } root = { subtract_variable = { which = pe_c7_cash value = prev.pe_ccit_c } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 8 } } root = { subtract_variable = { which = pe_c8_cash value = prev.pe_ccit_c } } }
							log = "private_economy: corporate citadel upgrade started: slot \\[This.pe_ccit_k], to level \\[This.pe_ccit_l] + 1, \\[This.pe_ccit_m] months"
						}
					}
					# Defence platforms while it can afford them (500 each, 4 per level; a third of its cash).
					set_variable = { which = pe_tmp value = pe_ccit_l }
					multiply_variable = { which = pe_tmp value = 4 }
					if = {
						limit = { check_variable = { which = pe_ccit_p value < pe_tmp } root = { check_variable = { which = pe_cit_cash value >= 1500 } } }
						export_trigger_value_to_variable = { trigger = num_ships variable = pe_fl_n0 }
						create_ship = { name = random random_existing_design = military_station_small graphical_culture = owner }
						export_trigger_value_to_variable = { trigger = num_ships variable = pe_fl_n1 }
						if = {
							limit = { check_variable = { which = pe_fl_n1 value > pe_fl_n0 } }
							change_variable = { which = pe_ccit_p value = 1 }
							if = { limit = { check_variable = { which = pe_ccit_k value = 1 } } root = { subtract_variable = { which = pe_c1_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 2 } } root = { subtract_variable = { which = pe_c2_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 3 } } root = { subtract_variable = { which = pe_c3_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 4 } } root = { subtract_variable = { which = pe_c4_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 5 } } root = { subtract_variable = { which = pe_c5_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 6 } } root = { subtract_variable = { which = pe_c6_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 7 } } root = { subtract_variable = { which = pe_c7_cash value = 500 } } }
							else_if = { limit = { check_variable = { which = pe_ccit_k value = 8 } } root = { subtract_variable = { which = pe_c8_cash value = 500 } } }
						}
					}
				}
			}
			every_owned_fleet = {
				limit = { has_fleet_flag = pe_ccit_kill }
				destroy_fleet = this
			}
			add_resource = { energy = 1 mult = root.pe_cit_ge }
			add_resource = { alloys = 1 mult = root.pe_cit_ga }
		}
		# Building claims: a site that is gone ends them.
		if = { limit = { check_variable = { which = pe_c1_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_1 } } } set_variable = { which = pe_c1_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c2_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_2 } } } set_variable = { which = pe_c2_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c3_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_3 } } } set_variable = { which = pe_c3_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c4_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_4 } } } set_variable = { which = pe_c4_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c5_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_5 } } } set_variable = { which = pe_c5_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c6_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_6 } } } set_variable = { which = pe_c6_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c7_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_7 } } } set_variable = { which = pe_c7_citb value = 0 } }
		if = { limit = { check_variable = { which = pe_c8_citb value = 1 } NOT = { any_system_within_border = { has_star_flag = pe_cit_site_8 } } } set_variable = { which = pe_c8_citb value = 0 } }
		# A new citadel: with our technology, one of the corporations that can afford it (three
		# times the cost, no site under way, not in its own pause of two years after an offer), at
		# random; 20% a quarter, at most one a quarter.
		if = {
			limit = { has_technology = tech_deep_space_citadel }
			set_variable = { which = pe_cit_k value = 0 }
			set_variable = { which = pe_cit_n value = 0 }
			if = { limit = { check_variable = { which = pe_c1_on value = 1 } NOT = { check_variable = { which = pe_c1_citb value = 1 } } check_variable = { which = pe_c1_cash value >= 34500 } OR = { NOT = { has_country_flag = pe_cit_cd_1 } check_variable = { which = pe_fl_force value = 1 } } } change_variable = { which = pe_cit_n value = 1 } }

			graphical_culture = root
			orbit_distance = trigger:inner_radius
			orbit_angle = 200
		}
	}
	# The holdings need the technology for the vanilla citadel (we have it: pe_citadel_step).
	event_target:pe_hold = { give_technology = { tech = tech_deep_space_citadel message = no } }
	event_target:pe_hold = {
		every_owned_megastructure = {
			limit = { is_megastructure_type = pe_corp_citadel_site is_upgrading = no }
			set_variable = { which = pe_cit_k value = root.pe_cit_k }
			set_variable = { which = pe_cit_nm value = root.pe_cit_nm }
			upgrade_megastructure_to = pe_corp_citadel_build
		}
	}
	log = "private_economy: corporate citadel started by slot \\[Root.pe_cit_k] in \\[event_target:pe_cit_sys.GetName]"
}

# Country scope, monthly (pe.100): while we are at war and our holdings have a citadel, they
# join our wars (on our side), so that it fights for us.
pe_cit_war = {
	if = {
		limit = { is_at_war = yes }
		random_country = {
			limit = {
				is_country_type = pe_holdings
				has_relation_flag = { who = root flag = pe_holdings_of }
				is_at_war = no
				any_owned_fleet = { has_fleet_flag = pe_ccit }
			}
			join_war = root
		}
	}
}
