# Private economy logic. Runs quarterly per country (pe.100); construction sites
# advance monthly.
# Call depth stays at 4 or less: the engine initialises only 5 nested levels.
#   pe_monthly_sites (1) -> pe_site_complete (2)
#   pe_quarterly (1) -> pe_init (2)
#                    -> pe_census (2)
#                    -> pe_market (2) -> pe_shadow (3)
#                    -> pe_job_values (2) -> pe_job_count (3), pe_job_res (3)
#                    -> pe_compute_values (2)
#                    -> pe_planet_step (2) -> pe_planet_counts (3)
#                                          -> pe_planet_prepare (3), pe_chab_prep (3), pe_chab_pay (3)
#                                          -> pe_planet_fill (3)      -> pe_count_planet_job (4)
#                                          -> pe_firm_accounting (3)  -> pe_firm_identify (4)
#                                                                     -> pe_deposit_fetch (4)
#                                                                     -> pe_firm_book (4)
#                                          -> pe_cluster_counts (3), pe_cluster_apply (3)
#                                          -> pe_corp_lending (3)
#                                          -> pe_firm_change (3)
#                                          -> pe_choose_variant (3)   -> pe_consider_variant (4)
#                                          -> pe_found_variant (3)    -> pe_found_firm (4)
#                                          -> pe_buy_or_offer (3)     -> pe_buyout_execute (4)
#                                          -> pe_nationalise_one (3)
#                                          -> pe_restructure (3)      -> pe_retool_consider (4)
#                                                                     -> pe_retool_guard (4)
#                    -> pe_command_economy (2) -> pe_firm_identify (3), pe_deposit_fetch (3)
#                    -> pe_supply_step (2)
#                    -> pe_auto_cover_step (2)
#                    -> pe_corp_step (2) -> pe_corp_census, _crisis, _dissolve, _found, _acquire,
#                                           _rivalry, _branches, pe_holdings_ensure,
#                                           pe_corp_stations, pe_corp_station_output,
#                                           pe_holdings_surplus, pe_corp_fleets, pe_corp_charter,
#                                           pe_corp_convoy, pe_corp_salvage, pe_corp_swarm,
#                                           pe_corp_furnace, pe_corp_auto_msg,
#                                           pe_chab_quarterly, pe_corp_social, pe_dc_step,
#                                           pe_citadel_step (3, pe_citadel_effects.txt)
#                                        -> pe_corp_free, pe_corp_msg, pe_corp_found_new,
#                                           pe_corp_station_pick, pe_ledger_bits,
#                                           pe_chab_pick, pe_chab_exec (4)
#                    -> pe_relocate (2) -> pe_relocate_from (3) -> pe_relocate_exec (4)
#                                                             -> pe_relocate_check_dest (4)
#                    -> pe_patron (2) -> pe_patron_fire (3)   (pe_patron_effects.txt)
#                    -> pe_deal (2) -> pe_deal_fire (3)   (pe_patron_effects.txt)
#                    -> pe_ledgers (2) -> pe_ledger_planet (3) -> pe_ledger_bits (4)
#                                      -> pe_ledger_bits (3)
# Variant lists and dispatchers: pe_firm_effects_generated.txt (tools/generate-firms.mjs).
#
# The loop: shadow prices (market price plus need) -> value added per private job ->
# the firms keep a profit share of it (the economic order lowers private output by the
# same share) -> firm capital and investor pool -> firms grow in stages, the pool builds
# new firms and buys districts -> more supply (construction sites count in advance)
# lowers the price -> less profit -> restructuring or bankruptcy.
#
# Capital is counted in trade. Investor pool (country pe_capital): starts at 500, gets
# private savings (a share of the trade income, by economic order) and half of each
# debt-free firm's profit (all of it from corporations). It pays 40% down on foundings
# (the state's district cost: city 500, rural 300) and buyouts; the firm owes the rest
# and repays it before it pays out or grows. It also pays restructuring when the firm
# cannot. Firm capital (deposit pe_fcap): the other half; pays expansions and
# restructuring, absorbs losses.
#
# Building: a firm is first a construction site (vanilla district build time: city 16,
# rural 8 months; buyouts and restructuring half of that). One site per planet. New
# firms only where civilians are idle (4.x has no unemployed) and the planet's private
# jobs are at least 80% filled; otherwise investors buy state districts (Free Market,
# Laissez-faire), never the last one of a type. New firms and restructuring targets may
# not consume a resource in deficit. Every change of jobs re-runs the planet's employment.

# --- Monthly ------------------------------------------------------------------

pe_monthly_sites = {
	every_owned_colony = {
		limit = { has_planet_flag = pe_site is_occupied_flag = no }
		# A site the player cleared leaves the flag behind.
		if = {
			limit = { count_deposits = { category = pe_deposit_cat_sites count < 1 } }
			remove_planet_flag = pe_lab_site
			remove_planet_flag = pe_site
		}
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_sites }
			change_variable = { which = pe_left value = -1 }
		}
		# A finished site of a new firm opens only when there are workers for it: idle
		# civilians (civilians minus the jobs still open here) of at least 50. Otherwise it
		# waits (pe_hold), at most two years; then the investors give up and get their down
		# payment back. Restructured firms always reopen (their workers are the idle ones).
		export_trigger_value_to_variable = { trigger = num_assigned_jobs parameters = { job = civilian } variable = pe_civ }
		export_trigger_value_to_variable = { trigger = free_jobs variable = pe_fj }
		if = { limit = { NOT = { is_variable_set = pe_civ } } set_variable = { which = pe_civ value = 0 } }
		if = { limit = { NOT = { is_variable_set = pe_fj } } set_variable = { which = pe_fj value = 0 } }
		set_variable = { which = pe_sidle value = pe_civ }
		subtract_variable = { which = pe_sidle value = pe_fj }
		set_variable = { which = pe_scancel value = 0 }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_sites check_variable = { which = pe_kind value = 1 } }
			set_variable = { which = pe_hold value = 0 }
			if = {
				limit = {
					check_variable = { which = pe_left value <= 0 }
					prev = { check_variable = { which = pe_sidle value < 50 } }
				}
				set_variable = { which = pe_hold value = 1 }
			}
		}
		random_deposit = {
			limit = {
				has_deposit_category = pe_deposit_cat_sites
				check_variable = { which = pe_hold value = 1 }
				check_variable = { which = pe_left value <= -24 }
			}
			# Down payment = 40% of the cost; the debt is the other 60%.
			root = {
				set_variable = { which = pe_tmp value = prev.pe_debt }
				multiply_variable = { which = pe_tmp value = 0.6667 }
				change_variable = { which = pe_tmp value = prev.pe_fcap }
				change_variable = { which = pe_capital value = pe_tmp }
			}
			prev = { set_variable = { which = pe_scancel value = 1 } }
			remove_deposit = yes
		}
		if = {
			limit = { check_variable = { which = pe_scancel value = 1 } }
			save_event_target_as = pe_msg_planet
			set_timed_planet_flag = { flag = pe_cooldown days = 360 }
			if = {
				limit = { count_deposits = { category = pe_deposit_cat_sites count < 1 } }
				remove_planet_flag = pe_site
			}
			owner = {
				if = { limit = { NOT = { has_country_flag = pe_mute_crises } } create_message = {
					type = pe_private_economy_crises
					localization = pe_msg_site_cancel
					days = 30
					target = event_target:pe_msg_planet
					variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
				} }
			}
		}
		pe_site_complete = yes
	}
}

# --- Country level ------------------------------------------------------------

pe_init = {
	if = { limit = { NOT = { is_variable_set = pe_private_share } } set_variable = { which = pe_private_share value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_profit_share } } set_variable = { which = pe_profit_share value = 0 } }
	# Profit share by economic order (halved in 0.33; set here too, for running games).
	if = { limit = { has_policy_flag = pe_order_guided } set_variable = { which = pe_profit_share value = 0.075 } set_variable = { which = pe_profit_pct value = 7.5 } }
	else_if = { limit = { has_policy_flag = pe_order_mixed } set_variable = { which = pe_profit_share value = 0.1 } set_variable = { which = pe_profit_pct value = 10 } }
	else_if = { limit = { has_policy_flag = pe_order_free_market } set_variable = { which = pe_profit_share value = 0.125 } set_variable = { which = pe_profit_pct value = 12.5 } }
	else_if = { limit = { has_policy_flag = pe_order_laissez_faire } set_variable = { which = pe_profit_share value = 0.175 } set_variable = { which = pe_profit_pct value = 17.5 } }
	if = { limit = { NOT = { is_variable_set = pe_savings_rate } } set_variable = { which = pe_savings_rate value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_capital } } set_variable = { which = pe_capital value = 500 } }
	if = { limit = { NOT = { is_variable_set = pe_pool_income } } set_variable = { which = pe_pool_income value = 0 } }
	# Share of the profits the state paid last quarter (see pe_ledgers).
	if = { limit = { NOT = { is_variable_set = pe_lpay } } set_variable = { which = pe_lpay value = 1 } }
	if = {
		limit = { NOT = { is_variable_set = pe_seed } }
		set_variable_to_random_value = { which = pe_seed min = 0 max = 1008 rounded = yes }
	}
	# Spread the empires over the three months of a quarter.
	if = {
		limit = { NOT = { is_variable_set = pe_months } }
		set_variable_to_random_value = { which = pe_months min = 0 max = 2 rounded = yes }
	}
}

pe_quarterly = {
	pe_init = yes
	# Command economy: no private economy at all (firms that appeared anyway are removed).
	if = {
		limit = {
			has_policy_flag = pe_order_planned
			any_owned_colony = {
				OR = {
					count_deposits = { category = pe_deposit_cat_firms count > 0 }
					count_deposits = { category = pe_deposit_cat_sites count > 0 }
				}
			}
		}
		pe_command_economy = yes
	}
	set_variable = { which = pe_pool_income value = 0 }
	set_variable = { which = pe_profit_due value = 0 }
	set_variable = { which = pe_auto_p value = 0 }
	pe_savings = yes
	pe_census = yes
	pe_market = yes
	pe_job_values = yes
	pe_compute_values = yes
	# Planets with idle civilians first: founders start where workers are available.
	every_owned_colony = {
		limit = { pe_planet_eligible = yes pe_has_idle_pops = yes }
		set_timed_planet_flag = { flag = pe_stepped days = 1 }
		pe_planet_step = yes
	}
	every_owned_colony = {
		limit = { pe_planet_eligible = yes NOT = { has_planet_flag = pe_stepped } }
		pe_planet_step = yes
	}
	pe_supply_step = yes
	pe_relocate = yes
	pe_corp_step = yes
	pe_patron = yes
	pe_labour = yes
	pe_deal = yes
	pe_ledgers = yes
}

# Country scope: a firm moves to another planet with its workers, at most once a year per
# empire. Source: a planet that is stuck (no idle civilians, so no new firm can open; free
# district slots do not help without workers) or short of housing or amenities, without
# construction, and no firm moved away from it in the last three years (pe_rl_cd). The
# capital (its designation suits no firm) is stuck only once 2/3 of its district slots are
# used (pe_built_up); short of housing or amenities it always gives firms away. Firm: its
# main share does not suit
# the source's designation, or the source is short of housing or amenities. Destination:
# the firm could be founded there (free slot; rural firms a free district of their type
# with its deposits), no construction, free housing and amenities for all workers (200 x
# stage size), and a designation that suits the firm (if the source is short of housing
# or amenities, any destination will do when none suits). The firm pays the move like a
# restructuring (city 250 / rural 150 x stage size) from its own capital.
pe_relocate = {
	set_variable = { which = pe_rl_done value = 0 }
	if = {
		limit = { NOT = { has_country_flag = pe_relocated } }
		every_owned_colony = {
			limit = {
				pe_planet_eligible = yes
				NOT = { has_planet_flag = pe_site }
				NOT = { has_planet_flag = pe_rl_cd }
				count_deposits = { category = pe_deposit_cat_firms count > 0 }
				OR = {
					AND = {
						NOT = { pe_has_idle_pops = yes }
						OR = {
							is_capital = no
							check_variable = { which = pe_built_up value = 1 }
						}
					}
					free_amenities < 0
					free_housing < 0
				}
			}
			if = {
				limit = { root = { check_variable = { which = pe_rl_done value = 0 } } }
				pe_relocate_from = yes
			}
		}
	}
}

# Colony scope (the source planet): find a firm and the best destination, then move it.
# Only firms whose main share does not suit the source's designation move: a firm that
# suits its planet stays, even when the planet is short of housing or amenities.
pe_relocate_from = {
	save_event_target_as = pe_rl_src
	root = { set_variable = { which = pe_rl_srcdef value = 0 } }
	if = {
		limit = { OR = { free_amenities < 0 free_housing < 0 } }
		root = { set_variable = { which = pe_rl_srcdef value = 1 } }
	}
	every_deposit = {
		limit = {
			has_deposit_category = pe_deposit_cat_firms
			is_variable_set = pe_prim
		}
		set_variable = { which = pe_rl_cand value = 0 }
		if = {
			limit = { root = { check_variable = { which = pe_rl_done value = 0 } } }
			# Workers (200 x stage size) and the moving cost (city 250 / rural 150 x stage size).
			set_variable = { which = pe_tmp value = pe_stage }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_tmp value = 0.5 }
			if = {
				limit = { OR = { check_variable = { which = pe_cls value = 1 } check_variable = { which = pe_cls value = 5 } } }
				set_variable = { which = pe_rl_c value = 250 }
			}
			else = { set_variable = { which = pe_rl_c value = 150 } }
			multiply_variable = { which = pe_rl_c value = pe_tmp }
			root = {
				set_variable = { which = pe_rl_prim value = prev.pe_prim }
				set_variable = { which = pe_rl_code value = prev.pe_code }
				set_variable = { which = pe_rl_wf value = prev.pe_tmp }
				multiply_variable = { which = pe_rl_wf value = 200 }
				set_variable = { which = pe_rl_cost value = prev.pe_rl_c }
				set_variable = { which = pe_rl_best value = -1 }
			}
			if = {
				limit = {
					check_variable = { which = pe_fcap value >= pe_rl_c }
					planet = { NOT = { pe_desig_prim = yes } }
				}
				# The best destination that suits the firm's main share ...
				root = {
					set_variable = { which = pe_rl_pass value = 1 }
					every_owned_colony = {
						limit = {
							pe_planet_eligible = yes
							NOT = { is_same_value = event_target:pe_rl_src }
							NOT = { has_planet_flag = pe_site }
							pe_rl_target_fits = yes
							pe_desig_prim = yes
						}
						pe_relocate_check_dest = yes
					}
				}
				# ... or, from a planet short of housing or amenities, the best other one.
				if = {
					limit = {
						root = {
							check_variable = { which = pe_rl_best value < 0 }
							check_variable = { which = pe_rl_srcdef value = 1 }
						}
					}
					root = {
						set_variable = { which = pe_rl_pass value = 2 }
						every_owned_colony = {
							limit = {
								pe_planet_eligible = yes
								NOT = { is_same_value = event_target:pe_rl_src }
								NOT = { has_planet_flag = pe_site }
								pe_rl_target_fits = yes
							}
							pe_relocate_check_dest = yes
						}
					}
				}
				if = {
					limit = { root = { check_variable = { which = pe_rl_best value >= 0 } } }
					set_variable = { which = pe_rl_cand value = 1 }
					root = { set_variable = { which = pe_rl_done value = 1 } }
				}
			}
		}
	}
	if = {
		limit = { root = { check_variable = { which = pe_rl_done value = 1 } } }
		pe_relocate_exec = yes
		set_timed_planet_flag = { flag = pe_rl_cd days = 1080 }
	}
}

# Colony scope (a possible destination): enough free housing and amenities for all
# workers with a margin, so it is not short itself after the move. The destination with
# the most room left (the smaller of the two) wins.
pe_relocate_check_dest = {
	export_trigger_value_to_variable = { trigger = free_housing variable = pe_fh }
	export_trigger_value_to_variable = { trigger = free_amenities variable = pe_fa }
	subtract_variable = { which = pe_fh value = root.pe_rl_wf }
	subtract_variable = { which = pe_fa value = root.pe_rl_wf }
	set_variable = { which = pe_room value = pe_fh }
	if = { limit = { check_variable = { which = pe_fa value < pe_room } } set_variable = { which = pe_room value = pe_fa } }
	if = {
		limit = {
			check_variable = { which = pe_fh value >= 100 }
			check_variable = { which = pe_fa value >= 200 }
			check_variable = { which = pe_room value > root.pe_rl_best }
		}
		save_event_target_as = pe_rl_dst
		root = {
			set_variable = { which = pe_rl_best value = prev.pe_room }
			# 1 = designation, 2 = the source lacks housing or amenities (for the message)
			set_variable = { which = pe_rl_reason value = pe_rl_pass }
		}
	}
}

# Country scope: what the private economy costs the state, as monthly trade upkeep that
# shows in the budget: each planet's firms' profits (profit ledger) and the private savings
# (savings ledger on the capital), for the next three months.
# Cap (0.40): a progressive share of the net trade income the empire would have without
# these two ledgers (pe_lnet, per month): 40% of the first 100, 70% of the next 200, 85% of
# everything above 300. So the trade balance never turns negative because of them, and a
# rich trade economy may invest far more. Without a net trade income nothing is paid.
# What the state does not pay, the firms and the investor pool do not get either. The firms
# booked this quarter's profits at last quarter's share (pe_lpay, pe_firm_book); the
# difference to the share paid now is settled with the investor pool (pe_lcorr), so not a
# single trade is created or lost. Savings reach the pool only here, at the share paid.
pe_ledgers = {
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = trade } variable = pe_lnet }
	export_trigger_value_to_variable = { trigger = resource_expenses_compare parameters = { resource = trade } variable = pe_tmp }
	subtract_variable = { which = pe_lnet value = pe_tmp }
	# Last quarter's profit and savings ledgers are in the expenses: add them back.
	if = { limit = { NOT = { is_variable_set = pe_lpaid_m } } set_variable = { which = pe_lpaid_m value = 0 } }
	change_variable = { which = pe_lnet value = pe_lpaid_m }
	set_variable = { which = pe_lcap_m value = 0 }
	if = {
		limit = { check_variable = { which = pe_lnet value > 0 } }
		set_variable = { which = pe_tmp value = pe_lnet }
		if = { limit = { check_variable = { which = pe_tmp value > 100 } } set_variable = { which = pe_tmp value = 100 } }
		multiply_variable = { which = pe_tmp value = 0.4 }
		change_variable = { which = pe_lcap_m value = pe_tmp }
	}
	if = {
		limit = { check_variable = { which = pe_lnet value > 100 } }
		set_variable = { which = pe_tmp value = pe_lnet }
		if = { limit = { check_variable = { which = pe_tmp value > 300 } } set_variable = { which = pe_tmp value = 300 } }
		change_variable = { which = pe_tmp value = -100 }
		multiply_variable = { which = pe_tmp value = 0.7 }
		change_variable = { which = pe_lcap_m value = pe_tmp }
	}
	if = {
		limit = { check_variable = { which = pe_lnet value > 300 } }
		set_variable = { which = pe_tmp value = pe_lnet }
		change_variable = { which = pe_tmp value = -300 }
		multiply_variable = { which = pe_tmp value = 0.85 }
		change_variable = { which = pe_lcap_m value = pe_tmp }
	}
	set_variable = { which = pe_lcap value = pe_lcap_m }
	multiply_variable = { which = pe_lcap value = 3 }
	# What is due: profits (half after disaster relief, pe.226 gratitude; a quarter more under
	# a monopoly, pe.244; both inside the cap) and savings.
	set_variable = { which = pe_lfac value = 1 }
	if = { limit = { has_country_flag = pe_gratitude } multiply_variable = { which = pe_lfac value = 0.5 } }
	if = { limit = { has_country_flag = pe_monopoly } multiply_variable = { which = pe_lfac value = 1.25 } }
	set_variable = { which = pe_ltot value = pe_profit_due }
	multiply_variable = { which = pe_ltot value = pe_lfac }
	change_variable = { which = pe_ltot value = pe_savings_last }
	set_variable = { which = pe_lscale value = 1 }
	if = {
		limit = {
			check_variable = { which = pe_ltot value > pe_lcap }
			check_variable = { which = pe_ltot value > 0 }
		}
		set_variable = { which = pe_lscale value = pe_lcap }
		divide_variable = { which = pe_lscale value = pe_ltot }
	}
	set_variable = { which = pe_lprof value = pe_lscale }
	multiply_variable = { which = pe_lprof value = pe_lfac }
	# Settle the profits the firms booked at last quarter's share.
	set_variable = { which = pe_lcorr value = pe_lprof }
	subtract_variable = { which = pe_lcorr value = pe_lpay }
	multiply_variable = { which = pe_lcorr value = pe_profit_due }
	change_variable = { which = pe_capital value = pe_lcorr }
	change_variable = { which = pe_pool_income value = pe_lcorr }
	# The savings the state pays.
	set_variable = { which = pe_tmp value = pe_savings_last }
	multiply_variable = { which = pe_tmp value = pe_lscale }
	change_variable = { which = pe_capital value = pe_tmp }
	change_variable = { which = pe_pool_income value = pe_tmp }
	set_variable = { which = pe_lpay value = pe_lprof }
	every_owned_colony = { pe_ledger_planet = yes }
	set_variable = { which = pe_sav_month value = pe_savings_last }
	multiply_variable = { which = pe_sav_month value = pe_lscale }
	divide_variable = { which = pe_sav_month value = 3 }
	set_variable = { which = pe_profit_month value = pe_profit_due }
	multiply_variable = { which = pe_profit_month value = pe_lprof }
	divide_variable = { which = pe_profit_month value = 3 }
	set_variable = { which = pe_lpaid_m value = pe_profit_month }
	change_variable = { which = pe_lpaid_m value = pe_sav_month }
	# For the register (pe.276) and the log.
	set_variable = { which = pe_lpay_pct value = pe_lprof }
	multiply_variable = { which = pe_lpay_pct value = 100 }
	round_variable = pe_lpay_pct
	set_variable = { which = pe_lnet_r value = pe_lnet }
	round_variable = pe_lnet_r
	set_variable = { which = pe_lcap_r value = pe_lcap_m }
	round_variable = pe_lcap_r
	set_variable = { which = pe_ldue_r value = pe_ltot }
	divide_variable = { which = pe_ldue_r value = 3 }
	round_variable = pe_ldue_r
	set_variable = { which = pe_lcorr_r value = pe_lcorr }
	round_variable = pe_lcorr_r
	if = {
		limit = { is_ai = no }
		log = "private_economy: ledgers: net trade without them \\[Root.pe_lnet_r], cap \\[Root.pe_lcap_r], due \\[Root.pe_ldue_r] a month, profits paid \\[Root.pe_lpay_pct]%, pool settlement \\[Root.pe_lcorr_r]"
	}
	# State bonds (pe.242): the state repays the investor pool monthly, shown with the savings.
	if = {
		limit = { check_variable = { which = pe_bond_left value > 0 } }
		change_variable = { which = pe_sav_month value = pe_bond_month }
		set_variable = { which = pe_tmp value = pe_bond_month }
		multiply_variable = { which = pe_tmp value = 3 }
		change_variable = { which = pe_capital value = pe_tmp }
		change_variable = { which = pe_bond_left value = -3 }
	}
	# Shares (pe.241): 10 trade a month per share, paid by the investor pool as long as it can.
	set_variable = { which = pe_div_month value = 0 }
	if = {
		limit = { check_variable = { which = pe_shares value > 0 } }
		set_variable = { which = pe_div_month value = pe_shares }
		multiply_variable = { which = pe_div_month value = 10 }
		set_variable = { which = pe_tmp value = pe_div_month }
		multiply_variable = { which = pe_tmp value = 3 }
		if = {
			limit = { check_variable = { which = pe_capital value >= pe_tmp } }
			subtract_variable = { which = pe_capital value = pe_tmp }
		}
		else = { set_variable = { which = pe_div_month value = 0 } }
	}
	# Investment abroad (pe.265): taxes on the returns for the state (shown with the
	# dividends) and the returns for the investor pool, for ten years.
	if = {
		limit = { check_variable = { which = pe_abroad_q value > 0 } }
		change_variable = { which = pe_div_month value = pe_abroad_state }
		set_variable = { which = pe_tmp value = pe_abroad_pool }
		multiply_variable = { which = pe_tmp value = 3 }
		change_variable = { which = pe_capital value = pe_tmp }
		change_variable = { which = pe_abroad_q value = -1 }
		if = {
			limit = { check_variable = { which = pe_abroad_q value <= 0 } }
			set_variable = { which = pe_abroad_state value = 0 }
			set_variable = { which = pe_abroad_pool value = 0 }
		}
	}
	capital_scope = {
		root = { set_variable = { which = pe_lv value = pe_div_month } }
		pe_ledger_bits = { V = ld }
		if = {
			limit = { root = { check_variable = { which = pe_div_month value >= 0.5 } } }
			if = { limit = { NOT = { has_deposit = d_pe_ledger_dividend } } add_deposit = d_pe_ledger_dividend }
		}
		else_if = {
			limit = { has_deposit = d_pe_ledger_dividend }
			remove_deposit = d_pe_ledger_dividend
		}
		root = { set_variable = { which = pe_lv value = pe_sav_month } }
		pe_ledger_bits = { V = ls }
		if = {
			limit = { root = { check_variable = { which = pe_sav_month value >= 0.5 } } }
			if = { limit = { NOT = { has_deposit = d_pe_ledger_savings } } add_deposit = d_pe_ledger_savings }
		}
		else_if = {
			limit = { has_deposit = d_pe_ledger_savings }
			remove_deposit = d_pe_ledger_savings
		}
	}
}

# Colony scope: this planet's profit ledger for the next three months.
pe_ledger_planet = {
	if = {
		limit = { OR = { count_deposits = { category = pe_deposit_cat_firms count < 1 } is_occupied_flag = yes } }
		set_variable = { which = pe_pl_profit value = 0 }
	}
	if = { limit = { NOT = { is_variable_set = pe_pl_profit } } set_variable = { which = pe_pl_profit value = 0 } }
	root = {
		set_variable = { which = pe_lv value = prev.pe_pl_profit }
		multiply_variable = { which = pe_lv value = pe_lprof }
		divide_variable = { which = pe_lv value = 3 }
	}
	pe_ledger_bits = { V = lp }
	if = {
		limit = { check_variable = { which = pe_pl_profit value > 1.5 } }
		if = { limit = { NOT = { has_deposit = d_pe_ledger_profit } } add_deposit = d_pe_ledger_profit }
	}
	else_if = {
		limit = { has_deposit = d_pe_ledger_profit }
		remove_deposit = d_pe_ledger_profit
	}
}

# Country scope: private savings. A share of the gross trade income (pe_savings_rate, set by
# the economic order: 0/5/10/15/20%) goes to the investor pool each quarter; the state
# pays it monthly as trade upkeep (savings ledger on the capital, see pe_ledgers), and the
# pool gets what the state pays. Only if the stockpile can pay it, and only while the pool
# holds less than 4500 (otherwise it could not spend it).
pe_savings = {
	set_variable = { which = pe_savings_last value = 0 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = trade } variable = pe_sav }
	multiply_variable = { which = pe_sav value = pe_savings_rate }
	multiply_variable = { which = pe_sav value = 3 }
	export_resource_stockpile_to_variable = { resource = trade variable = pe_stock }
	if = {
		limit = {
			check_variable = { which = pe_sav value > 0 }
			check_variable = { which = pe_stock value >= pe_sav }
			# A full pool (it has nothing to spend it on) saves no more.
			check_variable = { which = pe_capital value < 4500 }
		}
		# The pool gets them at the end of the quarter, as far as the state pays them (pe_ledgers).
		set_variable = { which = pe_savings_last value = pe_sav }
	}
}

# Country scope: shadow price pe_s_<res> in trade per unit.
#   Investment goods: alloys and food for empires with bio-ships are worth their base
#   value (4 / 1) while the stock is below a quarter of the storage, falling to the selling
#   price at three quarters and to a third of it when nearly full. Unity (3) and research
#   (3.5 per point) have no stock. Alloys, research and unity are then balanced against
#   each other (pe_triad).
#   Flow goods: the buying price while short, falling to the selling price at the
#   target surplus (BAND, % of production); above it a dumping price (a third of the
#   selling price), with a full stock less still. A deficit adds a premium. Food only
#   needs to stay just above
#   zero (5%), unless alloys are made from it (Catalytic Processing: 20%).
#   Trade is money: worth 1, but a surplus only buys goods after market fees (0.7).
pe_market = {
	set_variable = { which = pe_emergency value = 0 }
	set_variable = { which = pe_is_invest value = 1 }
	pe_shadow = { RES = alloys BASE = 4 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	# No stock and no deficit for research and unity: fixed values.
	set_variable = { which = pe_s_unity value = 3 }
	set_variable = { which = pe_s_physics_research value = 3.5 }
	set_variable = { which = pe_s_society_research value = 3.5 }
	set_variable = { which = pe_s_engineering_research value = 3.5 }
	set_variable = { which = pe_is_invest value = 0 }
	if = {
		limit = { country_uses_bio_ships = yes }
		set_variable = { which = pe_is_invest value = 1 }
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
		set_variable = { which = pe_is_invest value = 0 }
	}
	else_if = {
		limit = { is_catalytic_empire = yes }
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	}
	else = {
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 5 MARKET = yes PEND = yes }
	}
	pe_shadow = { RES = consumer_goods BASE = 2 SELL = 0 BAND = 15 MARKET = yes PEND = yes }
	pe_shadow = { RES = minerals BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	pe_shadow = { RES = energy BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	pe_shadow = { RES = trade BASE = 1 SELL = 0.7 BAND = 20 PEND = yes }
	pe_triad = yes
}

# Country scope: the investment triad. Alloys, research and unity are all needed; each
# gets a target share of the empire's investment output (valued at 4 / 3.5 / 3 per unit):
# 35 / 45 / 20%, shifted by ethics (militarist or at war: alloys, pacifist: less alloys,
# materialist: research, spiritualist: unity). Below its target a good is worth more,
# above it less: factor 0.5 + 0.5 x target / share, between 0.5 and 2.
pe_triad = {
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = alloys } variable = pe_tri_a }
	multiply_variable = { which = pe_tri_a value = 4 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = physics_research } variable = pe_tri_r }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = society_research } variable = pe_tmp }
	change_variable = { which = pe_tri_r value = pe_tmp }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = engineering_research } variable = pe_tmp }
	change_variable = { which = pe_tri_r value = pe_tmp }
	multiply_variable = { which = pe_tri_r value = 3.5 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = unity } variable = pe_tri_u }
	multiply_variable = { which = pe_tri_u value = 3 }
	set_variable = { which = pe_tri_sum value = pe_tri_a }
	change_variable = { which = pe_tri_sum value = pe_tri_r }
	change_variable = { which = pe_tri_sum value = pe_tri_u }
	# Targets in percent.
	set_variable = { which = pe_tgt_a value = 35 }
	set_variable = { which = pe_tgt_r value = 45 }
	set_variable = { which = pe_tgt_u value = 20 }
	if = { limit = { is_militarist = yes } change_variable = { which = pe_tgt_a value = 10 } change_variable = { which = pe_tgt_r value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	if = { limit = { is_pacifist = yes } change_variable = { which = pe_tgt_a value = -10 } change_variable = { which = pe_tgt_r value = 5 } change_variable = { which = pe_tgt_u value = 5 } }
	if = { limit = { is_materialist = yes } change_variable = { which = pe_tgt_r value = 10 } change_variable = { which = pe_tgt_a value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	if = { limit = { is_spiritualist = yes } change_variable = { which = pe_tgt_u value = 10 } change_variable = { which = pe_tgt_a value = -5 } change_variable = { which = pe_tgt_r value = -5 } }
	if = { limit = { is_at_war = yes } change_variable = { which = pe_tgt_a value = 10 } change_variable = { which = pe_tgt_r value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	pe_triad_factor = { L = a }
	pe_triad_factor = { L = r }
	pe_triad_factor = { L = u }
	multiply_variable = { which = pe_s_alloys value = pe_tri_f_a }
	multiply_variable = { which = pe_s_physics_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_society_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_engineering_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_unity value = pe_tri_f_u }
}

# Country scope: share pe_tri_pct_<L> (percent of the triad) and factor pe_tri_f_<L>.
pe_triad_factor = {
	if = { limit = { check_variable = { which = pe_tgt_$L$ value < 10 } } set_variable = { which = pe_tgt_$L$ value = 10 } }
	set_variable = { which = pe_tri_f_$L$ value = 2 }
	set_variable = { which = pe_tri_pct_$L$ value = 0 }
	if = {
		limit = {
			check_variable = { which = pe_tri_sum value > 0 }
			check_variable = { which = pe_tri_$L$ value > 0 }
		}
		set_variable = { which = pe_tri_pct_$L$ value = pe_tri_$L$ }
		divide_variable = { which = pe_tri_pct_$L$ value = pe_tri_sum }
		multiply_variable = { which = pe_tri_pct_$L$ value = 100 }
		set_variable = { which = pe_tri_f_$L$ value = pe_tgt_$L$ }
		divide_variable = { which = pe_tri_f_$L$ value = pe_tri_pct_$L$ }
		multiply_variable = { which = pe_tri_f_$L$ value = 0.5 }
		change_variable = { which = pe_tri_f_$L$ value = 0.5 }
		if = { limit = { check_variable = { which = pe_tri_f_$L$ value > 2 } } set_variable = { which = pe_tri_f_$L$ value = 2 } }
		if = { limit = { check_variable = { which = pe_tri_f_$L$ value < 0.5 } } set_variable = { which = pe_tri_f_$L$ value = 0.5 } }
	}
}

pe_shadow = {
	export_resource_income_to_variable = { resource = $RES$ variable = pe_net }
	[[PEND]
	change_variable = { which = pe_net value = pe_pend_$RES$ }
	]
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = $RES$ } variable = pe_gross }
	# Shortage margin: below max(5, 5% of gross production), at most 50, a resource counts
	# as short, so the economy keeps a buffer instead of sliding into the minus.
	set_variable = { which = pe_margin value = pe_gross }
	multiply_variable = { which = pe_margin value = 0.05 }
	if = { limit = { check_variable = { which = pe_margin value < 5 } } set_variable = { which = pe_margin value = 5 } }
	if = { limit = { check_variable = { which = pe_margin value > 50 } } set_variable = { which = pe_margin value = 50 } }
	subtract_variable = { which = pe_net value = pe_margin }
	# Kept per resource: how much a restructuring may take away (pe_retool_guard).
	set_variable = { which = pe_netm_$RES$ value = pe_net }
	export_resource_stockpile_to_variable = { resource = $RES$ variable = pe_stock }
	export_resource_maximum_to_variable = { resource = $RES$ variable = pe_max }
	set_variable = { which = pe_scale value = pe_gross }
	if = { limit = { check_variable = { which = pe_scale value < 10 } } set_variable = { which = pe_scale value = 10 } }
	set_variable = { which = pe_rel value = pe_net }
	divide_variable = { which = pe_rel value = pe_scale }
	multiply_variable = { which = pe_rel value = 100 }
	[[MARKET]
	export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = $RES$ amount = 100 trade_type = market_buy } variable = pe_pbuy }
	divide_variable = { which = pe_pbuy value = 100 }
	if = { limit = { check_variable = { which = pe_pbuy value <= 0 } } set_variable = { which = pe_pbuy value = $BASE$ } multiply_variable = { which = pe_pbuy value = 1.3 } }
	export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = $RES$ amount = 100 trade_type = market_sell } variable = pe_psell }
	divide_variable = { which = pe_psell value = 100 }
	if = { limit = { check_variable = { which = pe_psell value <= 0 } } set_variable = { which = pe_psell value = $BASE$ } multiply_variable = { which = pe_psell value = 0.7 } }
	]
	[[!MARKET]
	set_variable = { which = pe_pbuy value = $BASE$ }
	set_variable = { which = pe_psell value = $SELL$ }
	]
	# pe_def_<res>: in deficit (pending supply included). Firms that consume it are not
	# founded and are the first to restructure.
	set_variable = { which = pe_def_$RES$ value = 0 }
	if = { limit = { check_variable = { which = pe_net value < 0 } } set_variable = { which = pe_def_$RES$ value = 1 } }
	# Investment goods start from their base value, not the market buying price with fees.
	if = { limit = { check_variable = { which = pe_is_invest value = 1 } } set_variable = { which = pe_pbuy value = $BASE$ } }
	set_variable = { which = pe_s_$RES$ value = pe_pbuy }
	if = {
		limit = {
			check_variable = { which = pe_is_invest value = 1 }
			check_variable = { which = pe_max value > 0 }
		}
		# Stock fraction f: pbuy - (pbuy - psell) * clamp((f - 0.25) / 0.5, 0, 1)
		set_variable = { which = pe_frac value = pe_stock }
		divide_variable = { which = pe_frac value = pe_max }
		set_variable = { which = pe_tmp value = pe_frac }
		change_variable = { which = pe_tmp value = -0.25 }
		multiply_variable = { which = pe_tmp value = 2 }
		if = { limit = { check_variable = { which = pe_tmp value < 0 } } set_variable = { which = pe_tmp value = 0 } }
		if = { limit = { check_variable = { which = pe_tmp value > 1 } } set_variable = { which = pe_tmp value = 1 } }
		set_variable = { which = pe_prem value = pe_pbuy }
		subtract_variable = { which = pe_prem value = pe_psell }
		multiply_variable = { which = pe_prem value = pe_tmp }
		subtract_variable = { which = pe_s_$RES$ value = pe_prem }
		if = {
			limit = { check_variable = { which = pe_frac value >= 0.95 } }
			set_variable = { which = pe_s_$RES$ value = pe_psell }
			multiply_variable = { which = pe_s_$RES$ value = 0.33 }
			# Alloys stay the economy's end goal: never below half their base value (0.38).
			set_variable = { which = pe_tmp value = $BASE$ }
			multiply_variable = { which = pe_tmp value = 0.5 }
			if = { limit = { check_variable = { which = pe_s_$RES$ value < pe_tmp } } set_variable = { which = pe_s_$RES$ value = pe_tmp } }
		}
	}
	if = {
		limit = {
			check_variable = { which = pe_is_invest value = 0 }
			check_variable = { which = pe_net value >= 0 }
		}
		if = {
			limit = { check_variable = { which = pe_rel value >= $BAND$ } }
			set_variable = { which = pe_s_$RES$ value = pe_psell }
			multiply_variable = { which = pe_s_$RES$ value = 0.33 }
		}
		else = {
			# pbuy - (pbuy - psell) * rel / BAND
			set_variable = { which = pe_tmp value = pe_pbuy }
			subtract_variable = { which = pe_tmp value = pe_psell }
			multiply_variable = { which = pe_tmp value = pe_rel }
			divide_variable = { which = pe_tmp value = $BAND$ }
			subtract_variable = { which = pe_s_$RES$ value = pe_tmp }
		}
		if = {
			limit = { check_variable = { which = pe_max value > 0 } }
			set_variable = { which = pe_full value = pe_max }
			multiply_variable = { which = pe_full value = 0.9 }
			if = {
				limit = { check_variable = { which = pe_stock value >= pe_full } }
				set_variable = { which = pe_s_$RES$ value = pe_psell }
				multiply_variable = { which = pe_s_$RES$ value = 0.15 }
			}
		}
	}
	if = {
		limit = { check_variable = { which = pe_net value < 0 } }
		# Any deficit +25%, plus 4% per 1% of production missing (up to +200%), plus
		# 50% if the stock lasts less than a year.
		set_variable = { which = pe_prem value = pe_rel }
		multiply_variable = { which = pe_prem value = -0.04 }
		change_variable = { which = pe_prem value = 0.25 }
		if = { limit = { check_variable = { which = pe_prem value > 2 } } set_variable = { which = pe_prem value = 2 } }
		set_variable = { which = pe_runway value = pe_net }
		multiply_variable = { which = pe_runway value = -12 }
		if = {
			limit = { check_variable = { which = pe_stock value < pe_runway } }
			change_variable = { which = pe_prem value = 0.5 }
			set_variable = { which = pe_emergency value = 1 }
		}
		change_variable = { which = pe_prem value = 1 }
		multiply_variable = { which = pe_s_$RES$ value = pe_prem }
	}
}

	root = { change_variable = { which = pe_fa value = prev.pe_tmp } }
	export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = $JOB$ } variable = pe_tmp }
	root = { change_variable = { which = pe_ff value = prev.pe_tmp } }
}

pe_planet_step = {
	# Assign new pops first, so idle civilians really are idle.
	check_planet_employment = yes
	# Event deposits that expired.
	if = {
		limit = { NOT = { has_planet_flag = pe_hydro } }
		if = { limit = { has_deposit = d_pe_hydroponics } remove_deposit = d_pe_hydroponics }
		if = { limit = { has_deposit = d_pe_hydroponics_2 } remove_deposit = d_pe_hydroponics_2 }
		if = { limit = { has_deposit = d_pe_hydroponics_3 } remove_deposit = d_pe_hydroponics_3 }
	}
	pe_planet_counts = yes
	pe_planet_prepare = yes
	# A corporate habitat: its corporation pays founding, buyouts and growth here.
	pe_chab_prep = yes
	# Idle pops: civilians minus the jobs still open here (they would take those first) and
	# minus the jobs of the construction sites here (200 x stage size each; a restructured
	# firm takes its workers back). At least 50 of them for a new firm or a growth step.
	root = { set_variable = { which = pe_unemp value = 0 } }
	export_trigger_value_to_variable = { trigger = num_assigned_jobs parameters = { job = civilian } variable = pe_civ }
	export_trigger_value_to_variable = { trigger = free_jobs variable = pe_fj }
	if = { limit = { NOT = { is_variable_set = pe_civ } } set_variable = { which = pe_civ value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_fj } } set_variable = { which = pe_fj value = 0 } }
	set_variable = { which = pe_idle value = pe_civ }
	subtract_variable = { which = pe_idle value = pe_fj }
	every_deposit = {
		limit = { has_deposit_category = pe_deposit_cat_sites }
		set_variable = { which = pe_tmp value = pe_tstage }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_tmp value = 100 }
		prev = { subtract_variable = { which = pe_idle value = prev.pe_tmp } }
	}
	if = { limit = { check_variable = { which = pe_idle value >= 50 } } root = { set_variable = { which = pe_unemp value = 1 } } }
	# Unemployment (0.38): while pops are idle and district slots free, the investors may go
	# beyond the economic order's share, one more firm per 200 idle pops (never the last free
	# slot). Firms founded beyond it (pe_xq) are not nationalised when the idle pops are gone;
	# only firms beyond the share and these are (pe_tbase = the share alone).
	set_variable = { which = pe_tbase value = pe_target }
	if = { limit = { NOT = { is_variable_set = pe_xq } } set_variable = { which = pe_xq value = 0 } }
	if = { limit = { root = { check_variable = { which = pe_private_share value <= 0 } } } set_variable = { which = pe_xq value = 0 } }
	set_variable = { which = pe_tmp value = pe_firms }
	subtract_variable = { which = pe_tmp value = pe_tbase }
	if = { limit = { check_variable = { which = pe_tmp value < 0 } } set_variable = { which = pe_tmp value = 0 } }
	if = { limit = { check_variable = { which = pe_xq value > pe_tmp } } set_variable = { which = pe_xq value = pe_tmp } }
	if = {
		limit = {
			NOT = { has_planet_flag = pe_corp_hab }
			root = { check_variable = { which = pe_private_share value > 0 } }
			check_variable = { which = pe_idle value >= 200 }
		}
		set_variable = { which = pe_xt value = pe_idle }
		divide_variable = { which = pe_xt value = 200 }
		floor_variable = pe_xt
		set_variable = { which = pe_tmp value = pe_free }
		change_variable = { which = pe_tmp value = -1 }
		if = { limit = { check_variable = { which = pe_xt value > pe_tmp } } set_variable = { which = pe_xt value = pe_tmp } }
		if = { limit = { check_variable = { which = pe_xt value > 0 } } change_variable = { which = pe_target value = pe_xt } }
	}
	pe_planet_fill = yes
	# Filled private jobs here (corporate benefits, pe_corp_social).
	set_variable = { which = pe_pfa value = root.pe_fa }
	pe_firm_accounting = yes
	# Firms per main product, sector group and corporation; cluster modifiers.
	pe_cluster_counts = yes
	pe_cluster_apply = yes
	# Corporate loans for growth and against losses.
	pe_corp_lending = yes
	root = { set_variable = { which = pe_chg_type value = 0 } }
	pe_firm_change = yes
	# A growth step (+100 jobs) uses idle pops too.
	if = {
		limit = { root = { check_variable = { which = pe_chg_type value = 1 } } }
		change_variable = { which = pe_idle value = -100 }
		if = { limit = { check_variable = { which = pe_idle value < 50 } } root = { set_variable = { which = pe_unemp value = 0 } } }
	}
	pe_automation = yes
	set_variable = { which = pe_tnat value = pe_tbase }
	change_variable = { which = pe_tnat value = pe_xq }
	# Why the investors did or did not do something here this quarter, for the register
	# (pe.276, 0.40), pe_why: 1 a site is running, 2 pause after a founding or buyout, 3 war
	# zone, 4 the economic order's share is taken, 5 no free district slot, 6 private jobs
	# below 80%, 7 no idle pops, 8 no firm worth it, 9 not enough private capital, 10 founded,
	# 11 bought a state district or offered to, 12 waiting for our decision beyond the share,
	# 13 we refused firms beyond the share, 14 supply check, 17 another request beyond the
	# share is open, 18 a firm is nationalised.
	set_variable = { which = pe_why value = 0 }
	if = { limit = { has_modifier = pe_warzone } set_variable = { which = pe_why value = 3 } }
	else_if = { limit = { has_planet_flag = pe_cooldown } set_variable = { which = pe_why value = 2 } }
	else_if = { limit = { has_planet_flag = pe_site } set_variable = { which = pe_why value = 1 } }
	# Many idle pops (600 more than the sites here will employ): a second site at once.
	if = {
		limit = {
			OR = {
				NOT = { has_planet_flag = pe_site }
				AND = {
					check_variable = { which = pe_idle value >= 600 }
					count_deposits = { category = pe_deposit_cat_sites count < 2 }
				}
			}
			NOT = { has_planet_flag = pe_cooldown }
		}
		export_trigger_value_to_variable = { trigger = count_deposits parameters = { category = pe_deposit_cat_sites } variable = pe_sc0 }
		if = {
			limit = { check_variable = { which = pe_firms value < pe_target } NOT = { has_modifier = pe_warzone } }
			if = {
				limit = {
					free_district_slots > 0
					root = {
						check_variable = { which = pe_unemp value = 1 }
						check_variable = { which = pe_fill_cur value >= 0.8 }
					}
				}
				# Option value: the last free slots are only used for clearly better firms.
				root = { set_variable = { which = pe_best_min value = 0 } set_variable = { which = pe_dmode value = 0 } }
				if = { limit = { free_district_slots = 1 } root = { set_variable = { which = pe_best_min value = 8 } } }
				else_if = { limit = { free_district_slots = 2 } root = { set_variable = { which = pe_best_min value = 4 } } }
				pe_choose_variant = { MODE = found }
				# Beyond the share in a human empire (not under Laissez-faire): we decide (pe.323;
				# one request at a time, a planet that was refused waits five years).
				if = {
					limit = {
						check_variable = { which = pe_firms value >= pe_tbase }
						owner = { is_ai = no NOT = { has_policy_flag = pe_order_laissez_faire } }
					}
					if = { limit = { root = { check_variable = { which = pe_best_variant value = 0 } } } set_variable = { which = pe_why value = 8 } }
					else_if = { limit = { has_planet_flag = pe_xq_no } set_variable = { which = pe_why value = 13 } }
					else_if = { limit = { owner = { has_country_flag = pe_xq_pending } } set_variable = { which = pe_why value = 17 } }
					else = {
						set_variable = { which = pe_why value = 12 }
						save_event_target_as = pe_xq_planet
						root = {
							set_variable = { which = pe_xq_code value = pe_best_variant }
							set_variable = { which = pe_xq_idle value = prev.pe_idle }
							set_timed_country_flag = { flag = pe_xq_pending days = 360 }
							country_event = { id = pe.323 }
						}
					}
				}
				else = {
					pe_found_variant = yes
					# Beyond the share: this firm stays when the idle pops are gone.
					export_trigger_value_to_variable = { trigger = count_deposits parameters = { category = pe_deposit_cat_sites } variable = pe_sc1 }
					if = {
						limit = { check_variable = { which = pe_sc1 value > pe_sc0 } check_variable = { which = pe_firms value >= pe_tbase } }
						change_variable = { which = pe_xq value = 1 }
					}
					if = { limit = { check_variable = { which = pe_sc1 value > pe_sc0 } } set_variable = { which = pe_why value = 10 } }
					else_if = { limit = { root = { check_variable = { which = pe_best_variant value = 0 } } } set_variable = { which = pe_why value = 8 } }
					else_if = { limit = { root = { check_variable = { which = pe_safe_ok value = 0 } } } set_variable = { which = pe_why value = 14 } }
					else = { set_variable = { which = pe_why value = 9 } }
				}
			}
			else = {
				if = { limit = { free_district_slots < 1 } set_variable = { which = pe_why value = 5 } }
				else_if = { limit = { root = { check_variable = { which = pe_unemp value = 0 } } } set_variable = { which = pe_why value = 7 } }
				else = { set_variable = { which = pe_why value = 6 } }
				if = {
					# At most one buyout (or offer) per planet every two years.
					limit = { OR = { owner = { pe_buyout_allowed = yes } has_planet_flag = pe_corp_hab } NOT = { has_planet_flag = pe_buy_cd } }
					# A firm must be worth clearly more than the state district it replaces.
					root = { set_variable = { which = pe_best_min value = 2 } set_variable = { which = pe_dmode value = 1 } }
					pe_choose_variant = { MODE = buy }
					pe_buy_or_offer = yes
					if = { limit = { has_planet_flag = pe_buy_cd } set_variable = { which = pe_why value = 11 } }
				}
			}
		}
		else_if = {
			limit = {
				check_variable = { which = pe_firms value > pe_target }
				check_variable = { which = pe_firms value > pe_tnat }
			}
			pe_nationalise_one = yes
			set_variable = { which = pe_why value = 18 }
		}
		else_if = {
			limit = { NOT = { has_modifier = pe_warzone } }
			set_variable = { which = pe_why value = 4 }
		}
	}
	# Nothing started on this planet this quarter, its private jobs are filled, and no
	# restructuring here in the last year: one firm may restructure.
	if = {
		limit = {
			NOT = { has_planet_flag = pe_site }
			NOT = { has_planet_flag = pe_cooldown }
			# A year between two restructurings on a planet; each firm has its own pause (pe_age,
			# pe_restructure: ten years, two for a firm consuming what the empire is short of).
			NOT = { has_planet_flag = pe_retool_cd }
			NOT = { has_modifier = pe_warzone }
			check_variable = { which = pe_firms_real value > 0 }
			root = { check_variable = { which = pe_fill_cur value >= 0.8 } }
		}
		pe_restructure = yes
	}
	pe_chab_pay = yes
}

# Every firm books its quarter.
pe_firm_accounting = {
	root = {
		set_variable = { which = pe_pl_acc value = 0 }
		set_variable = { which = pe_pend_any value = 0 }
		set_variable = { which = pe_here_m value = 0 }
		set_variable = { which = pe_here_e value = 0 }
		set_variable = { which = pe_here_f value = 0 }
	}
	every_deposit = {
		limit = { has_deposit_category = pe_deposit_cat_firms }
		if = { limit = { NOT = { is_variable_set = pe_code } } pe_firm_identify = yes }
		# Age in quarters (firms from before 0.16 count as old enough to restructure).
		if = { limit = { NOT = { is_variable_set = pe_age } } set_variable = { which = pe_age value = 40 } }
		change_variable = { which = pe_age value = 1 }
		if = { limit = { NOT = { is_variable_set = pe_corp } } set_variable = { which = pe_corp value = 0 } }
		if = { limit = { NOT = { is_variable_set = pe_loan } } set_variable = { which = pe_loan value = 0 } set_variable = { which = pe_lender value = 0 } set_variable = { which = pe_loan_q value = 0 } }
		# A firm number (supply contracts).
		if = {
			limit = { OR = { NOT = { is_variable_set = pe_fid } check_variable = { which = pe_fid value <= 0 } } }
			root = { change_variable = { which = pe_fid_next value = 1 } }
			set_variable = { which = pe_fid value = root.pe_fid_next }
		}
		pe_deposit_fetch = yes
		pe_deposit_fetch_scale = yes
		pe_safe_growth = yes
		pe_firm_book = yes
		# Rural classes present (a private mine frees its mining district when it restructures).
		if = { limit = { check_variable = { which = pe_cls value = 2 } } root = { set_variable = { which = pe_here_m value = 1 } } }
		if = { limit = { check_variable = { which = pe_cls value = 3 } } root = { set_variable = { which = pe_here_e value = 1 } } }
		if = { limit = { check_variable = { which = pe_cls value = 4 } } root = { set_variable = { which = pe_here_f value = 1 } } }
	}
	set_variable = { which = pe_pl_profit value = root.pe_pl_acc }
}

# Deposit scope. Quarterly profit = value added x stage size x fill x profit share,
# minus overhead (0.3 per stage and month). A firm in debt pays all of it back first: the
# investor pool's founding loan, then a corporate loan (3% interest a quarter). Lacking
# capital for growth, or in losses, it asks a corporation for a loan (pe_corp_lending).
# Then half goes to the firm and half to the investor pool (a corporation keeps it all),
# but a firm only keeps a reserve for its own restructuring and next growth step; the
# rest goes to the pool. Losses hit the firm. Then at most one stage change
# per planet and quarter: expand (+100 workforce, half a district) when the firm can pay
# half a district's cost (city 250, rural 150),
# its jobs are filled and the planet has idle civilians; shrink or go bankrupt when its
# capital runs out.
pe_firm_book = {
	set_variable = { which = pe_profit value = pe_va }
	set_variable = { which = pe_tmp value = pe_stage }
	change_variable = { which = pe_tmp value = 1 }
	multiply_variable = { which = pe_tmp value = 0.5 }
	multiply_variable = { which = pe_profit value = pe_tmp }
	multiply_variable = { which = pe_profit value = root.pe_fill_cur }
	multiply_variable = { which = pe_profit value = root.pe_profit_share }
	set_variable = { which = pe_tmp value = pe_stage }
	multiply_variable = { which = pe_tmp value = 0.3 }
	subtract_variable = { which = pe_profit value = pe_tmp }
	multiply_variable = { which = pe_profit value = 3 }
	set_variable = { which = pe_pending value = 0 }
	if = { limit = { NOT = { is_variable_set = pe_debt } } set_variable = { which = pe_debt value = 0 } }
	set_variable = { which = pe_want_loan value = 0 }
	set_variable = { which = pe_want_bridge value = 0 }
	# A corporate loan costs 3% interest a quarter, 1% from the firm's own corporation (0.38).
	if = {
		limit = { check_variable = { which = pe_loan value > 0 } }
		if = {
			limit = { check_variable = { which = pe_corp value > 0 } check_variable = { which = pe_lender value = pe_corp } }
			multiply_variable = { which = pe_loan value = 1.01 }
		}
		else = { multiply_variable = { which = pe_loan value = 1.03 } }
	}
	if = {
		limit = { check_variable = { which = pe_profit value > 0 } }
		# A price war of its corporation: 30% less profit.
		if = {
			limit = { check_variable = { which = pe_corp value > 0 } }
		if = { limit = { check_variable = { which = pe_corp value = 1 } root = { check_variable = { which = pe_c1_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 2 } root = { check_variable = { which = pe_c2_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 3 } root = { check_variable = { which = pe_c3_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 4 } root = { check_variable = { which = pe_c4_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 5 } root = { check_variable = { which = pe_c5_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 6 } root = { check_variable = { which = pe_c6_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 7 } root = { check_variable = { which = pe_c7_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		if = { limit = { check_variable = { which = pe_corp value = 8 } root = { check_variable = { which = pe_c8_war value > 0 } } } multiply_variable = { which = pe_profit value = 0.7 } }
		}
		# The state pays this profit (profit ledger, pe_ledgers). The firm books it at the share
		# the state paid last quarter (pe_lpay); pe_ledgers settles the difference with the
		# investor pool (0.40).
		root = {
			change_variable = { which = pe_profit_due value = prev.pe_profit }
			change_variable = { which = pe_pl_acc value = prev.pe_profit }
		}
		multiply_variable = { which = pe_profit value = root.pe_lpay }
	}
	# A corporate loan the firm cannot carry (its profit below the interest of 3%): after four
	# such quarters in a row its lender restructures it (pe_sanierung_step, pe.321 / pe.322).
	if = { limit = { NOT = { is_variable_set = pe_zq } } set_variable = { which = pe_zq value = 0 } }
	if = {
		limit = { check_variable = { which = pe_loan value > 0 } }
		set_variable = { which = pe_tmp value = pe_loan }
		multiply_variable = { which = pe_tmp value = 0.03 }
		if = { limit = { check_variable = { which = pe_profit value < pe_tmp } } change_variable = { which = pe_zq value = 1 } }
		else = { set_variable = { which = pe_zq value = 0 } }
	}
	else = { set_variable = { which = pe_zq value = 0 } }
	if = {
		# In debt: all profit pays the loan back; the rest of a final instalment is lost.
		limit = {
			check_variable = { which = pe_profit value > 0 }
			check_variable = { which = pe_debt value > 0 }
		}
		subtract_variable = { which = pe_debt value = pe_profit }
		if = { limit = { check_variable = { which = pe_debt value < 0 } } set_variable = { which = pe_debt value = 0 } }
	}
	else_if = {
		# A corporate loan: the profit pays it back first, to the lending corporation; what
		# is left stays with the firm.
		limit = {
			check_variable = { which = pe_profit value > 0 }
			check_variable = { which = pe_loan value > 0 }
		}
		set_variable = { which = pe_tmp value = pe_profit }
		if = { limit = { check_variable = { which = pe_tmp value > pe_loan } } set_variable = { which = pe_tmp value = pe_loan } }
		subtract_variable = { which = pe_loan value = pe_tmp }
		if = { limit = { check_variable = { which = pe_lender value = 1 } } root = { change_variable = { which = pe_c1_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 2 } } root = { change_variable = { which = pe_c2_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 3 } } root = { change_variable = { which = pe_c3_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 4 } } root = { change_variable = { which = pe_c4_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 5 } } root = { change_variable = { which = pe_c5_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 6 } } root = { change_variable = { which = pe_c6_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 7 } } root = { change_variable = { which = pe_c7_cash value = prev.pe_tmp } } }
		if = { limit = { check_variable = { which = pe_lender value = 8 } } root = { change_variable = { which = pe_c8_cash value = prev.pe_tmp } } }
		set_variable = { which = pe_tmp2 value = pe_profit }
		subtract_variable = { which = pe_tmp2 value = pe_tmp }
		change_variable = { which = pe_fcap value = pe_tmp2 }
		if = {
			limit = { check_variable = { which = pe_loan value < 0.5 } }
			set_variable = { which = pe_loan value = 0 }
			set_variable = { which = pe_lender value = 0 }
		}
	}
	else_if = {
		limit = { check_variable = { which = pe_profit value > 0 } }
		if = {
			limit = { check_variable = { which = pe_stage value < 3 } }
			set_variable = { which = pe_tmp value = pe_profit }
			multiply_variable = { which = pe_tmp value = 0.5 }
			change_variable = { which = pe_fcap value = pe_tmp }
			# A member of a corporation sends half of it to the corporation's cash.
			if = {
				limit = { check_variable = { which = pe_corp value > 0 } }
				multiply_variable = { which = pe_tmp value = 0.5 }
				if = { limit = { check_variable = { which = pe_corp value = 1 } } root = { change_variable = { which = pe_c1_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 2 } } root = { change_variable = { which = pe_c2_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 3 } } root = { change_variable = { which = pe_c3_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 4 } } root = { change_variable = { which = pe_c4_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 5 } } root = { change_variable = { which = pe_c5_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 6 } } root = { change_variable = { which = pe_c6_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 7 } } root = { change_variable = { which = pe_c7_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 8 } } root = { change_variable = { which = pe_c8_cash value = prev.pe_tmp } } }
			}
			root = {
				change_variable = { which = pe_capital value = prev.pe_tmp }
				change_variable = { which = pe_pool_income value = prev.pe_tmp }
			}
		}
		else = {
			change_variable = { which = pe_fcap value = pe_profit }
		}
		# Reserve: the firm keeps what its own restructuring would cost (city 250 / rural
		# 150 x stage size), plus its next growth step below a corporation. The rest goes
		# to the investor pool.
		if = { limit = { check_variable = { which = pe_cls value = 2 } } set_variable = { which = pe_res value = 225 } }
		else_if = { limit = { check_variable = { which = pe_cls value = 3 } } set_variable = { which = pe_res value = 225 } }
		else_if = { limit = { check_variable = { which = pe_cls value = 4 } } set_variable = { which = pe_res value = 225 } }
		else = { set_variable = { which = pe_res value = 375 } }
		set_variable = { which = pe_tmp value = pe_stage }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_tmp value = 0.5 }
		set_variable = { which = pe_res2 value = pe_res }
		multiply_variable = { which = pe_res2 value = pe_tmp }
		if = { limit = { check_variable = { which = pe_stage value < 3 } } change_variable = { which = pe_res2 value = pe_res } }
		if = {
			limit = { check_variable = { which = pe_fcap value > pe_res2 } }
			set_variable = { which = pe_tmp value = pe_fcap }
			subtract_variable = { which = pe_tmp value = pe_res2 }
			set_variable = { which = pe_fcap value = pe_res2 }
			if = {
				limit = { check_variable = { which = pe_corp value > 0 } }
				multiply_variable = { which = pe_tmp value = 0.5 }
				if = { limit = { check_variable = { which = pe_corp value = 1 } } root = { change_variable = { which = pe_c1_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 2 } } root = { change_variable = { which = pe_c2_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 3 } } root = { change_variable = { which = pe_c3_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 4 } } root = { change_variable = { which = pe_c4_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 5 } } root = { change_variable = { which = pe_c5_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 6 } } root = { change_variable = { which = pe_c6_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 7 } } root = { change_variable = { which = pe_c7_cash value = prev.pe_tmp } } }
				if = { limit = { check_variable = { which = pe_corp value = 8 } } root = { change_variable = { which = pe_c8_cash value = prev.pe_tmp } } }
			}
			root = {
				change_variable = { which = pe_capital value = prev.pe_tmp }
				change_variable = { which = pe_pool_income value = prev.pe_tmp }
			}
		}
	}
	else = {
		change_variable = { which = pe_fcap value = pe_profit }
	}
	if = {
		limit = { root = { check_variable = { which = pe_pend_any value = 0 } } }
		if = {
			# Losses: ask for a bridging loan (pe_corp_lending), at most two years in a row.
			limit = {
				check_variable = { which = pe_fcap value < 0 }
				check_variable = { which = pe_loan_q value < 8 }
			}
			set_variable = { which = pe_want_bridge value = pe_fcap }
			multiply_variable = { which = pe_want_bridge value = -1 }
		}
		else_if = {
			limit = { check_variable = { which = pe_fcap value < 0 } }
			set_variable = { which = pe_fcap value = 0 }
			if = { limit = { check_variable = { which = pe_stage value > 1 } } set_variable = { which = pe_pending value = 2 } }
			else = { set_variable = { which = pe_pending value = 3 } }
			root = { set_variable = { which = pe_pend_any value = 1 } }
		}
		else_if = {
			limit = {
				check_variable = { which = pe_stage value < 3 }
				NOT = { check_variable = { which = pe_bomb_hit value = 1 } }
				root = { check_variable = { which = pe_safe_ok value = 1 } }
				check_variable = { which = pe_profit value > 0 }
				check_variable = { which = pe_debt value <= 0 }
				check_variable = { which = pe_loan value <= 0 }
				root = {
					check_variable = { which = pe_fill_cur value >= 0.9 }
					check_variable = { which = pe_unemp value = 1 }
				}
			}
			if = { limit = { check_variable = { which = pe_cls value = 2 } } set_variable = { which = pe_tmp value = 225 } }
			else_if = { limit = { check_variable = { which = pe_cls value = 3 } } set_variable = { which = pe_tmp value = 225 } }
			else_if = { limit = { check_variable = { which = pe_cls value = 4 } } set_variable = { which = pe_tmp value = 225 } }
			else = { set_variable = { which = pe_tmp value = 375 } }
			# A corporate habitat: its corporation pays, 25% cheaper (pe_chab_prep).
			set_variable = { which = pe_chab_g value = pe_tmp }
			multiply_variable = { which = pe_chab_g value = 0.75 }
			if = {
				limit = { root = { check_variable = { which = pe_chab_pk value > 0 } check_variable = { which = pe_chab_cash value >= prev.pe_chab_g } } }
				root = {
					subtract_variable = { which = pe_chab_cash value = prev.pe_chab_g }
					change_variable = { which = pe_chab_due value = prev.pe_chab_g }
				}
				root = { change_variable = { which = pe_safe_reserved_energy value = pe_sd_energy } change_variable = { which = pe_safe_reserved_minerals value = pe_sd_minerals } change_variable = { which = pe_safe_reserved_food value = pe_sd_food } change_variable = { which = pe_safe_reserved_consumer_goods value = pe_sd_consumer_goods } change_variable = { which = pe_safe_reserved_trade value = pe_sd_trade } }
				set_variable = { which = pe_pending value = 1 }
				root = { set_variable = { which = pe_pend_any value = 1 } }
			}
			else_if = {
				limit = { check_variable = { which = pe_fcap value >= pe_tmp } }
				subtract_variable = { which = pe_fcap value = pe_tmp }
				root = { change_variable = { which = pe_safe_reserved_energy value = pe_sd_energy } change_variable = { which = pe_safe_reserved_minerals value = pe_sd_minerals } change_variable = { which = pe_safe_reserved_food value = pe_sd_food } change_variable = { which = pe_safe_reserved_consumer_goods value = pe_sd_consumer_goods } change_variable = { which = pe_safe_reserved_trade value = pe_sd_trade } }
				set_variable = { which = pe_pending value = 1 }
				root = { set_variable = { which = pe_pend_any value = 1 } }
			}
			# Not enough capital: ask a corporation for the rest (pe_corp_lending).
			else = {
				set_variable = { which = pe_want_cost value = pe_tmp }
				set_variable = { which = pe_want_loan value = pe_tmp }
				subtract_variable = { which = pe_want_loan value = pe_fcap }
			}
		}
	}
	# A profitable quarter with capital ends a series of bridging loans.
	if = {
		limit = {
			check_variable = { which = pe_profit value > 0 }
			check_variable = { which = pe_fcap value >= 0 }
		}
		set_variable = { which = pe_loan_q value = 0 }
	}
	# Restructured by its lender (pe_san_apply): 2 = one stage smaller, 3 = closed.
	if = {
		limit = { is_variable_set = pe_san check_variable = { which = pe_san value > 0 } }
		if = { limit = { check_variable = { which = pe_san value = 3 } } set_variable = { which = pe_pending value = 3 } root = { set_variable = { which = pe_pend_any value = 1 } } }
		else_if = { limit = { check_variable = { which = pe_stage value > 1 } } set_variable = { which = pe_pending value = 2 } root = { set_variable = { which = pe_pend_any value = 1 } } }
		set_variable = { which = pe_san value = 0 }
	}
	# Hit by a bombardment (pe_bombard_step): it loses a stage, at stage 1 it is destroyed
	# (ruins, see pe_firm_change). This replaces whatever else it planned.
	if = {
		limit = { is_variable_set = pe_bomb_hit check_variable = { which = pe_bomb_hit value = 1 } }
		if = { limit = { check_variable = { which = pe_stage value > 1 } } set_variable = { which = pe_pending value = 2 } }
		else = { set_variable = { which = pe_pending value = 3 } }
		root = { set_variable = { which = pe_pend_any value = 1 } }
	}
	# What the firm did this quarter and why, for the register (pe.276, 0.40): pe_lastp the
	# profit it kept, pe_fst the state (pe_corp_lending updates 10 to 13, 14 or 15):
	#   1 expands, 2 shrinks, 3 closes, 4 largest stage, 5 repays its founding debt, 6 repays
	#   a corporate loan, 7 losses, 8 growth waits: private jobs here below 90%, 9 growth
	#   stopped: it would consume what the empire is short of, 10 saves for the next stage (no
	#   corporate loan), 11 restructuring case, 12 growth waits: no idle pops, 13 growth waits:
	#   another firm here changes this quarter, 14 expands with a corporate loan, 15 losses
	#   bridged by a corporate loan.
	set_variable = { which = pe_lastp value = pe_profit }
	round_variable = pe_lastp
	if = { limit = { check_variable = { which = pe_pending value = 1 } } set_variable = { which = pe_fst value = 1 } }
	else_if = { limit = { check_variable = { which = pe_pending value = 2 } } set_variable = { which = pe_fst value = 2 } }
	else_if = { limit = { check_variable = { which = pe_pending value = 3 } } set_variable = { which = pe_fst value = 3 } }
	else_if = { limit = { check_variable = { which = pe_zq value >= 4 } } set_variable = { which = pe_fst value = 11 } }
	else_if = { limit = { OR = { check_variable = { which = pe_profit value <= 0 } check_variable = { which = pe_fcap value < 0 } } } set_variable = { which = pe_fst value = 7 } }
	else_if = { limit = { check_variable = { which = pe_debt value > 0 } } set_variable = { which = pe_fst value = 5 } }
	else_if = { limit = { check_variable = { which = pe_loan value > 0 } } set_variable = { which = pe_fst value = 6 } }
	else_if = { limit = { check_variable = { which = pe_stage value >= 3 } } set_variable = { which = pe_fst value = 4 } }
	else_if = { limit = { root = { check_variable = { which = pe_safe_ok value = 0 } } } set_variable = { which = pe_fst value = 9 } }
	else_if = { limit = { root = { check_variable = { which = pe_fill_cur value < 0.9 } } } set_variable = { which = pe_fst value = 8 } }
	else_if = { limit = { root = { check_variable = { which = pe_unemp value = 0 } } } set_variable = { which = pe_fst value = 12 } }
	else_if = { limit = { check_variable = { which = pe_want_loan value > 0 } } set_variable = { which = pe_fst value = 10 } }
	else = { set_variable = { which = pe_fst value = 13 } }
}

# Value of a variant on this planet: country value, plus entertainment need here, times
# the designation bonus of its main share, times 0.8-1.2 (entrepreneurial spirit). When
# buying, minus the value of the state district it replaces (weighted like a variant).
# Colony scope: private automation of the three worker jobs vanilla automates.
pe_automation = {
	set_variable = { which = pe_auto_pp value = 0 }
	# Levels per job (pe_al_<job>); older saves had one deposit per job and level.
	if = { limit = { has_deposit = d_pe_auto_miner_1 } remove_deposit = d_pe_auto_miner_1 set_variable = { which = pe_al_miner value = 1 } }
	if = { limit = { has_deposit = d_pe_auto_miner_2 } remove_deposit = d_pe_auto_miner_2 set_variable = { which = pe_al_miner value = 2 } }
	if = { limit = { NOT = { is_variable_set = pe_al_miner } } set_variable = { which = pe_al_miner value = 0 } }
	if = { limit = { has_deposit = d_pe_auto_technician_1 } remove_deposit = d_pe_auto_technician_1 set_variable = { which = pe_al_technician value = 1 } }
	if = { limit = { has_deposit = d_pe_auto_technician_2 } remove_deposit = d_pe_auto_technician_2 set_variable = { which = pe_al_technician value = 2 } }
	if = { limit = { NOT = { is_variable_set = pe_al_technician } } set_variable = { which = pe_al_technician value = 0 } }
	if = { limit = { has_deposit = d_pe_auto_farmer_1 } remove_deposit = d_pe_auto_farmer_1 set_variable = { which = pe_al_farmer value = 1 } }
	if = { limit = { has_deposit = d_pe_auto_farmer_2 } remove_deposit = d_pe_auto_farmer_2 set_variable = { which = pe_al_farmer value = 2 } }
	if = { limit = { NOT = { is_variable_set = pe_al_farmer } } set_variable = { which = pe_al_farmer value = 0 } }
	pe_auto_book = { JOB = miner }
	pe_auto_invest = { JOB = miner }
	pe_auto_book = { JOB = technician }
	pe_auto_invest = { JOB = technician }
	pe_auto_book = { JOB = farmer }
	pe_auto_invest = { JOB = farmer }
	# One deposit for all of it.
	if = {
		limit = { OR = { check_variable = { which = pe_al_miner value > 0 } check_variable = { which = pe_al_technician value > 0 } check_variable = { which = pe_al_farmer value > 0 } } }
		if = { limit = { NOT = { has_deposit = d_pe_automation } } add_deposit = d_pe_automation }
	}
	else_if = { limit = { has_deposit = d_pe_automation } remove_deposit = d_pe_automation }
	# The firms carry the energy upkeep of their automated jobs here (pe_auto_pp per month, 4 per
	# automated job; summed in pe_auto_p for pe_auto_cover_step). The planet modifier of 0.34.3
	# had no effect and goes.
	if = { limit = { has_modifier = pe_auto_cover } remove_modifier = pe_auto_cover }
}

# Colony scope. Like the vanilla automation buildings: level I +25% automated workforce
# (tech_assembly_pattern, 600), level II +50% (tech_construction_templates, 800 more). A
# joint investment of the firms with this job here, paid from their capital by capital
# share (the investor pool adds what they lack). The state pays the engine's energy upkeep
# for automated jobs (4 per automated worker job a month); the firms pay it back each
# quarter, valued at the energy shadow price. Automated jobs need no pops and no wages, so
# the firms keep their whole value: each quarter the automated value minus the upkeep is
# booked to the firms by workforce share; a loss is carried by their capital (by capital
# share), then by the investor pool. The level drops if neither can carry a loss (then no
# new automation of this job here for 3 years), or after two quarters with idle civilians
# on the planet (pops are there, automation is not needed; then 2 years).
pe_auto_book = {
	root = {
		set_variable = { which = pe_acap value = 0 }
		set_variable = { which = pe_apool value = 0 }
		set_variable = { which = pe_alvl value = 0 }
		set_variable = { which = pe_adown value = 0 }
	}
	every_deposit = {
		limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
		root = {
			change_variable = { which = pe_acap value = prev.pe_wf_$JOB$ }
			change_variable = { which = pe_apool value = prev.pe_fcap }
		}
	}
	# Leased machines (pe.212): the lease includes the energy the engine charges for the
	# automated jobs (25% of the private jobs x the lease size, 4 energy per job a month);
	# the investor pool pays for it, the state's upkeep is taken off by pe_auto_cover.
	if = {
		limit = { has_modifier = pe_leasing_$JOB$ }
		# Size of the lease (pe.212; 1 for leases from before 0.16).
		if = { limit = { NOT = { is_variable_set = pe_lease_m } } set_variable = { which = pe_lease_m value = 1 } }
		root = {
			set_variable = { which = pe_tmp value = pe_acap }
			multiply_variable = { which = pe_tmp value = 0.03 }
			multiply_variable = { which = pe_tmp value = prev.pe_lease_m }
			set_variable = { which = pe_tmp2 value = pe_tmp }
			divide_variable = { which = pe_tmp2 value = 3 }
			change_variable = { which = pe_auto_p value = pe_tmp2 }
			prev = { change_variable = { which = pe_auto_pp value = root.pe_tmp2 } }
			multiply_variable = { which = pe_tmp value = pe_s_energy }
			subtract_variable = { which = pe_capital value = pe_tmp }
		}
	}
	if = { limit = { check_variable = { which = pe_al_$JOB$ value = 1 } } root = { set_variable = { which = pe_alvl value = 1 } } }
	if = { limit = { check_variable = { which = pe_al_$JOB$ value = 2 } } root = { set_variable = { which = pe_alvl value = 2 } } }
	root = {
		# Automated value per month: jobs x 25% x level x value added per job.
		set_variable = { which = pe_aval value = pe_acap }
		multiply_variable = { which = pe_aval value = 0.0025 }
		multiply_variable = { which = pe_aval value = pe_alvl }
		multiply_variable = { which = pe_aval value = pe_vaj_pe_private_$JOB$ }
		# Energy upkeep per month: 4 per automated job = jobs x 0.01 x level.
		set_variable = { which = pe_aen value = pe_acap }
		multiply_variable = { which = pe_aen value = 0.01 }
		multiply_variable = { which = pe_aen value = pe_alvl }
		set_variable = { which = pe_aup value = pe_aen }
		multiply_variable = { which = pe_aup value = pe_s_energy }
		set_variable = { which = pe_anet value = pe_aval }
		subtract_variable = { which = pe_anet value = pe_aup }
		multiply_variable = { which = pe_anet value = 3 }
		set_variable = { which = pe_aloss value = pe_anet }
		multiply_variable = { which = pe_aloss value = -1 }
		# A loss is carried by the firms' capital first, then by the investor pool.
		set_variable = { which = pe_afloss value = pe_aloss }
		if = { limit = { check_variable = { which = pe_afloss value > pe_apool } } set_variable = { which = pe_afloss value = pe_apool } }
		set_variable = { which = pe_aploss value = pe_aloss }
		subtract_variable = { which = pe_aploss value = pe_afloss }
		set_variable = { which = pe_acover value = pe_apool }
		change_variable = { which = pe_acover value = pe_capital }
	}
	if = {
		limit = { root = { check_variable = { which = pe_alvl value > 0 } } }
		if = {
			limit = {
				root = {
					OR = {
						check_variable = { which = pe_anet value >= 0 }
						check_variable = { which = pe_acover value >= pe_aloss }
					}
				}
			}
			# The firms carry the energy upkeep (pe_aup in their accounts); the state's upkeep line
			# is taken off by the country modifier pe_auto_cover (pe_auto_cover_step).
			root = { change_variable = { which = pe_auto_p value = pe_aen } }
			change_variable = { which = pe_auto_pp value = root.pe_aen }
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
				if = {
					limit = { root = { check_variable = { which = pe_anet value >= 0 } } }
					set_variable = { which = pe_tmp value = root.pe_anet }
					multiply_variable = { which = pe_tmp value = pe_wf_$JOB$ }
					divide_variable = { which = pe_tmp value = root.pe_acap }
				}
				else_if = {
					limit = { root = { check_variable = { which = pe_apool value > 0 } } }
					set_variable = { which = pe_tmp value = root.pe_afloss }
					multiply_variable = { which = pe_tmp value = pe_fcap }
					divide_variable = { which = pe_tmp value = root.pe_apool }
					multiply_variable = { which = pe_tmp value = -1 }
				}
				else = { set_variable = { which = pe_tmp value = 0 } }
				change_variable = { which = pe_fcap value = pe_tmp }
			}
			if = {
				limit = { root = { check_variable = { which = pe_anet value < 0 } } }
				root = { subtract_variable = { which = pe_capital value = pe_aploss } }
			}
		}
		else = {
			root = { set_variable = { which = pe_adown value = 1 } }
			set_timed_planet_flag = { flag = pe_auto_cd_$JOB$ days = 1080 }
		}
		if = {
			limit = { root = { check_variable = { which = pe_unemp value = 1 } } }
			change_variable = { which = pe_idleq_$JOB$ value = 1 }
		}
		else = { set_variable = { which = pe_idleq_$JOB$ value = 0 } }
		if = {
			limit = { check_variable = { which = pe_idleq_$JOB$ value >= 2 } }
			set_variable = { which = pe_idleq_$JOB$ value = 0 }
			root = { set_variable = { which = pe_adown value = 1 } }
			set_timed_planet_flag = { flag = pe_auto_cd_$JOB$ days = 720 }
		}
	}
	# Drop one level.
	if = {
		limit = { root = { check_variable = { which = pe_adown value = 1 } } }
		save_event_target_as = pe_msg_planet
		if = { limit = { check_variable = { which = pe_al_$JOB$ value = 2 } } set_variable = { which = pe_al_$JOB$ value = 1 } }
		else = { set_variable = { which = pe_al_$JOB$ value = 0 } }
		check_planet_employment = yes
		owner = {
			if = { limit = { NOT = { has_country_flag = pe_mute_automation } } create_message = {
				type = pe_private_economy_automation
				localization = pe_msg_autodown_$JOB$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			} }
		}
	}
}

# Colony scope: invest in the next automation level of one job (uses pe_auto_book's
# numbers). Only for unfilled private jobs, without idle civilians, from 600 private jobs
# of the type, if the automated value covers the energy upkeep by 25%. A corporation whose
# firms have these jobs here pays the whole investment from its cash if it can (keeping a
# reserve of 300); otherwise the firms and the investor pool pay. The message for a
# corporation's automation comes from pe_corp_auto_msg (pe_acorp_<job> = its slot).
pe_auto_invest = {
	# Unfilled private jobs of this type here (automation only fills empty jobs).
	export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_$JOB$ } variable = pe_afree }
	if = { limit = { NOT = { is_variable_set = pe_afree } } set_variable = { which = pe_afree value = 0 } }
	root = { set_variable = { which = pe_afree value = prev.pe_afree } }
	root = {
		set_variable = { which = pe_anext value = pe_acap }
		multiply_variable = { which = pe_anext value = 0.0025 }
		set_variable = { which = pe_tmp value = pe_alvl }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_anext value = pe_tmp }
		multiply_variable = { which = pe_anext value = pe_vaj_pe_private_$JOB$ }
		set_variable = { which = pe_acost value = 600 }
		if = { limit = { check_variable = { which = pe_alvl value = 1 } } set_variable = { which = pe_acost value = 800 } }
		# A corporate habitat: its corporation pays (all firms there are its own), 25% cheaper.
		if = { limit = { prev = { has_planet_flag = pe_corp_hab } } multiply_variable = { which = pe_acost value = 0.75 } }
		# Upkeep of the next level: 4 energy per automated job.
		set_variable = { which = pe_aupn value = pe_acap }
		multiply_variable = { which = pe_aupn value = 0.01 }
		multiply_variable = { which = pe_aupn value = pe_tmp }
		multiply_variable = { which = pe_aupn value = pe_s_energy }
		# Safety margin: the value must cover the upkeep by 25%.
		multiply_variable = { which = pe_aupn value = 1.25 }
		# The firms pay what they can (by capital share), the investor pool the rest.
		set_variable = { which = pe_atot value = pe_apool }
		change_variable = { which = pe_atot value = pe_capital }
		set_variable = { which = pe_afirm value = pe_acost }
		if = { limit = { check_variable = { which = pe_afirm value > pe_apool } } set_variable = { which = pe_afirm value = pe_apool } }
		set_variable = { which = pe_arest value = pe_acost }
		subtract_variable = { which = pe_arest value = pe_afirm }
		set_variable = { which = pe_acorp value = 0 }

pe_consider_variant = {
	if = {
		limit = { pe_can_$MODE$_$V$ = yes root = { check_variable = { which = pe_safe_ok value = 1 } } }
		root = {
			# Its value here: habitat and void dwellers (pe_pf_u / pe_pf_r), the jobs of the
			# specialisation of its district type (pe_zb_<rural>, see pe_planet_prepare).
			set_variable = { which = pe_cand value = pe_val_$V$ }
			multiply_variable = { which = pe_cand value = pe_pf_$K$ }
			change_variable = { which = pe_cand value = $ZB$ }
			set_variable = { which = pe_tmp value = pe_ent_val }
			multiply_variable = { which = pe_tmp value = $ENT$ }
			change_variable = { which = pe_cand value = pe_tmp }
			multiply_variable = { which = pe_cand value = pe_spec_$PRIM$ }
			# Cluster: +5% for each firm with the same main product here (at most +15%).
			set_variable = { which = pe_tmp value = prev.pe_cn_$PRIM$ }
			if = { limit = { check_variable = { which = pe_tmp value > 3 } } set_variable = { which = pe_tmp value = 3 } }
			multiply_variable = { which = pe_tmp value = 0.05 }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_cand value = pe_tmp }
			# Pseudo-random 0.8-1.2: the engine's random value repeats within a day, so a
			# small linear congruential sequence (mod 1009) varies it per candidate.
			multiply_variable = { which = pe_seed value = 17 }
			change_variable = { which = pe_seed value = $CODE$ }
			change_variable = { which = pe_seed value = 23 }
			modulo_variable = { which = pe_seed value = 1009 }
			set_variable = { which = pe_rnd value = pe_seed }
			divide_variable = { which = pe_rnd value = 2522.5 }
			change_variable = { which = pe_rnd value = 0.8 }
			multiply_variable = { which = pe_cand value = pe_rnd }
			if = {
				limit = { check_variable = { which = pe_dmode value = 1 } }
				set_variable = { which = pe_tmp value = pe_dvz_$DCLS$ }
				multiply_variable = { which = pe_tmp value = 0.875 }
				subtract_variable = { which = pe_cand value = pe_tmp }
			}
			if = {
				limit = { check_variable = { which = pe_cand value > pe_best } }
				set_variable = { which = pe_best value = pe_cand }
				set_variable = { which = pe_best_variant value = $CODE$ }
			}
		}
	}
}

# The investor pool starts a construction site on a free slot. It costs what the district
# would cost the state: the pool pays 40% down, the firm owes the rest (pe_debt) and pays
# it back from its profit. The firm starts with 10 capital, enough to wait about three
# years for workers. On a corporate habitat its corporation pays instead, 25% cheaper and
# without a loan (pe_chab_prep); the firm is its own.
pe_found_firm = {
	root = {
		set_variable = { which = pe_down value = $COST$ }
		multiply_variable = { which = pe_down value = 0.4 }
		set_variable = { which = pe_loan value = $COST$ }
		subtract_variable = { which = pe_loan value = pe_down }
		set_variable = { which = pe_chab_c value = $COST$ }
		multiply_variable = { which = pe_chab_c value = 0.75 }
	}
	if = {
		limit = { root = { check_variable = { which = pe_chab_pk value > 0 } } }
		if = {
			limit = { root = { check_variable = { which = pe_chab_cash value >= pe_chab_c } check_variable = { which = pe_safe_ok value = 1 } } }
			root = { change_variable = { which = pe_safe_reserved_energy value = pe_safe_build_energy } change_variable = { which = pe_safe_reserved_minerals value = pe_safe_build_minerals } change_variable = { which = pe_safe_reserved_food value = pe_safe_build_food } change_variable = { which = pe_safe_reserved_consumer_goods value = pe_safe_build_consumer_goods } change_variable = { which = pe_safe_reserved_trade value = pe_safe_build_trade } }
		add_deposit = $SITE$
			last_added_deposit = {
				set_variable = { which = pe_target value = $CODE$ }
				set_variable = { which = pe_tstage value = 1 }
				set_variable = { which = pe_fcap value = 10 }
				set_variable = { which = pe_debt value = 0 }
				set_variable = { which = pe_left value = $MONTHS$ }
				set_variable = { which = pe_kind value = 1 }
				set_variable = { which = pe_corp value = root.pe_chab_pk }
			}
			set_planet_flag = pe_site
			root = {
				subtract_variable = { which = pe_chab_cash value = pe_chab_c }
				change_variable = { which = pe_chab_due value = pe_chab_c }
			}
		}
	}
	else_if = {
		limit = { root = { check_variable = { which = pe_capital value >= pe_down } check_variable = { which = pe_safe_ok value = 1 } } }
		root = { change_variable = { which = pe_safe_reserved_energy value = pe_safe_build_energy } change_variable = { which = pe_safe_reserved_minerals value = pe_safe_build_minerals } change_variable = { which = pe_safe_reserved_food value = pe_safe_build_food } change_variable = { which = pe_safe_reserved_consumer_goods value = pe_safe_build_consumer_goods } change_variable = { which = pe_safe_reserved_trade value = pe_safe_build_trade } }
		add_deposit = $SITE$
		last_added_deposit = {
			set_variable = { which = pe_target value = $CODE$ }
			set_variable = { which = pe_tstage value = 1 }
			set_variable = { which = pe_fcap value = 10 }
			set_variable = { which = pe_debt value = root.pe_loan }
			set_variable = { which = pe_left value = $MONTHS$ }
			set_variable = { which = pe_kind value = 1 }
		}
		set_planet_flag = pe_site
		root = { subtract_variable = { which = pe_capital value = pe_down } }
	}
}

# The investor pool buys a state district, which becomes the firm at once (no construction
# time). The state gets the full price; the pool pays 40% down, the firm owes the rest. Then
# a year's pause on the planet, and two years before the next buyout there. Only if the
# district really went away (pe_before > pe_after); otherwise nothing happens.
pe_buyout_execute = {
	set_variable = { which = pe_before value = 0 }
	set_variable = { which = pe_after value = 0 }
	owner = {
		set_variable = { which = pe_down value = $PRICE$ }
		multiply_variable = { which = pe_down value = 0.4 }
		# A corporate habitat: its corporation pays, 25% cheaper and without a loan (pe_chab_prep).
		set_variable = { which = pe_chab_c value = $PRICE$ }
		multiply_variable = { which = pe_chab_c value = 0.75 }
		set_variable = { which = pe_chab_buy value = 0 }
		if = {
			limit = { check_variable = { which = pe_chab_pk value > 0 } check_variable = { which = pe_chab_cash value >= pe_chab_c } }
			set_variable = { which = pe_chab_buy value = 1 }
		}
	}
	if = {
		limit = {
			has_district = $DISTRICT$
			root = { check_variable = { which = pe_safe_ok value = 1 } }
			owner = {
				OR = {
					check_variable = { which = pe_chab_buy value = 1 }
					AND = {
						NOT = { check_variable = { which = pe_chab_pk value > 0 } }
						check_variable = { which = pe_capital value >= pe_down }
					}
				}
			}
		}
		export_trigger_value_to_variable = { trigger = num_districts parameters = { type = $DISTRICT$ } variable = pe_before }
		remove_district = $DISTRICT$
		export_trigger_value_to_variable = { trigger = num_districts parameters = { type = $DISTRICT$ } variable = pe_after }
		if = { limit = { NOT = { is_variable_set = pe_after } } set_variable = { which = pe_after value = 0 } }
	}
	# Every sale logs its outcome (0.37.0: a sale on a habitat did not happen, cause unknown).
	if = {
		limit = { owner = { is_ai = no } }
		set_variable = { which = pe_buy_safe value = root.pe_safe_ok }
		owner = { prev = { set_variable = { which = pe_buy_cap value = prev.pe_capital } } }
		log = "private_economy: buyout $FIRM$ on \\[This.GetName]: $DISTRICT$ before \\[This.pe_before] after \\[This.pe_after], private capital \\[This.pe_buy_cap] (down payment $PRICE$ x 0.4), supply check \\[This.pe_buy_safe]"
	}
	if = {
		limit = { check_variable = { which = pe_before value > pe_after } }
		root = { change_variable = { which = pe_safe_reserved_energy value = pe_sd_energy } change_variable = { which = pe_safe_reserved_minerals value = pe_sd_minerals } change_variable = { which = pe_safe_reserved_food value = pe_sd_food } change_variable = { which = pe_safe_reserved_consumer_goods value = pe_sd_consumer_goods } change_variable = { which = pe_safe_reserved_trade value = pe_sd_trade } }
		add_deposit = $FIRM$
		last_added_deposit = {
			set_variable = { which = pe_code value = $CODE$ }
			set_variable = { which = pe_stage value = 1 }
			set_variable = { which = pe_fcap value = 10 }
			set_variable = { which = pe_debt value = $PRICE$ }
			multiply_variable = { which = pe_debt value = 0.6 }
			set_variable = { which = pe_age value = 0 }
			if = {
				limit = { root = { check_variable = { which = pe_chab_buy value = 1 } } }
				set_variable = { which = pe_debt value = 0 }
				set_variable = { which = pe_corp value = root.pe_chab_pk }
			}
		}
		set_timed_planet_flag = { flag = pe_cooldown days = 360 }
		set_timed_planet_flag = { flag = pe_buy_cd days = 720 }
		check_planet_employment = yes
		save_event_target_as = pe_msg_planet
		owner = {
			add_resource = { $RES$ = $MINERALS$ trade = $TRADE$ }
			if = {
				limit = { check_variable = { which = pe_chab_buy value = 1 } }
				subtract_variable = { which = pe_chab_cash value = pe_chab_c }
				change_variable = { which = pe_chab_due value = pe_chab_c }
			}
			else = { subtract_variable = { which = pe_capital value = pe_down } }
			if = { limit = { NOT = { has_country_flag = pe_mute_firms } } create_message = {
				type = pe_private_economy_firms
				localization = pe_msg_buy_$FIRM$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			} }
		}
	}
}

# Contracts" (pe_cluster_apply). See tools/generate-firms.mjs (SUP_*).
pe_supply_step = {
	every_owned_colony = {
		limit = { count_deposits = { category = pe_deposit_cat_firms count > 0 } }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_sup_fid value > 0 } }
			set_variable = { which = pe_sup_ok value = 0 }
		}
	}
	# Each supplier counts its buyers and marks them.
	every_owned_colony = {
		limit = { check_variable = { which = pe_sch value > 0 } }
		every_deposit = {
			limit = {
				has_deposit_category = pe_deposit_cat_firms
				check_variable = { which = pe_fid value > 0 }
				OR = { check_variable = { which = pe_prim value = 9 } check_variable = { which = pe_prim value = 2 } check_variable = { which = pe_prim value = 11 } }
			}
			root = {
				set_variable = { which = pe_cur_fid value = prev.pe_fid }
				set_variable = { which = pe_cur_n value = 0 }
				every_owned_colony = {
					limit = { count_deposits = { category = pe_deposit_cat_firms count > 0 } }
					every_deposit = {
						limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_sup_fid value = root.pe_cur_fid } }
						set_variable = { which = pe_sup_ok value = 1 }
						root = { change_variable = { which = pe_cur_n value = 1 } }
					}
				}
			}
			set_variable = { which = pe_sup_n value = root.pe_cur_n }
		}
	}
	# Contracts without a supplier end; the buyer waits a year before it looks again.
	every_owned_colony = {
		limit = { count_deposits = { category = pe_deposit_cat_firms count > 0 } }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_firms }

				set_variable = { which = pe_sup_cd value = 4 }

				}
			}
		}
	}
	# A new contract with a private mine.
	if = {
		limit = {
			any_owned_colony = { check_variable = { which = pe_scn_m value > 0 } }
			any_owned_colony = { check_variable = { which = pe_scs_m value > 0 } }

					OR = { AND = { check_variable = { which = pe_prim value = 1 } root = { is_catalytic_empire = no } } AND = { check_variable = { which = pe_prim value = 2 }  } }
				}
				root = { set_variable = { which = pe_new_cons value = prev.pe_fid } }
			}
			change_variable = { which = pe_scn_m value = -1 }
		}
		if = {
			limit = { check_variable = { which = pe_new_cons value > 0 } }
			random_owned_colony = {
				limit = { check_variable = { which = pe_scs_m value > 0 } }
				save_event_target_as = pe_sc_seller
				random_deposit = {
					limit = {
						has_deposit_category = pe_deposit_cat_firms
						check_variable = { which = pe_fid value > 0 }
						check_variable = { which = pe_prim value = 9 }
						check_variable = { which = pe_sup_n value < 2 }
					}
					change_variable = { which = pe_sup_n value = 1 }
					root = { set_variable = { which = pe_new_sup value = prev.pe_fid } }
				}
				change_variable = { which = pe_scs_m value = -1 }
			}
		}
		if = {
			limit = { check_variable = { which = pe_new_sup value > 0 } }
			event_target:pe_sc_buyer = {
				random_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_fid value = root.pe_new_cons } }
					set_variable = { which = pe_sup_fid value = root.pe_new_sup }
				}
				save_event_target_as = pe_msg_planet
			}
			if = {
				limit = { is_ai = no }
				if = { limit = { NOT = { has_country_flag = pe_mute_supply } } create_message = { type = pe_private_economy_supply localization = pe_msg_supply_m days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } variable = { type = name localization = FROM scope = event_target:pe_sc_seller } } }
			}
		}
	}
	# A new contract with a private factory.
	if = {
		limit = {
			any_owned_colony = { check_variable = { which = pe_scn_c value > 0 } }
			any_owned_colony = { check_variable = { which = pe_scs_c value > 0 } }
		}
		set_variable = { which = pe_new_cons value = 0 }
		set_variable = { which = pe_new_sup value = 0 }
		random_owned_colony = {
			limit = { check_variable = { which = pe_scn_c value > 0 } }
			save_event_target_as = pe_sc_buyer
			random_deposit = {
				limit = {
					has_deposit_category = pe_deposit_cat_firms
					check_variable = { which = pe_fid value > 0 }
					check_variable = { which = pe_sup_fid value <= 0 }
					check_variable = { which = pe_sup_cd value <= 0 }
					OR = { AND = { check_variable = { which = pe_prim value = 3 }  } AND = { check_variable = { which = pe_prim value = 5 }  } AND = { check_variable = { which = pe_prim value = 6 }  } AND = { check_variable = { which = pe_prim value = 7 }  } AND = { check_variable = { which = pe_prim value = 8 }  } }
				}
				root = { set_variable = { which = pe_new_cons value = prev.pe_fid } }
			}
			change_variable = { which = pe_scn_c value = -1 }
		}
		if = {
			limit = { check_variable = { which = pe_new_cons value > 0 } }
			random_owned_colony = {
				limit = { check_variable = { which = pe_scs_c value > 0 } }
				save_event_target_as = pe_sc_seller
				random_deposit = {
					limit = {
						has_deposit_category = pe_deposit_cat_firms
						check_variable = { which = pe_fid value > 0 }
						check_variable = { which = pe_prim value = 2 }
						check_variable = { which = pe_sup_n value < 2 }
					}
					change_variable = { which = pe_sup_n value = 1 }
					root = { set_variable = { which = pe_new_sup value = prev.pe_fid } }
				}
				change_variable = { which = pe_scs_c value = -1 }
			}
		}
		if = {
			limit = { check_variable = { which = pe_new_sup value > 0 } }
			event_target:pe_sc_buyer = {
				random_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_fid value = root.pe_new_cons } }
					set_variable = { which = pe_sup_fid value = root.pe_new_sup }

	# A new contract with a private farm.
	if = {

# --- Bombardment -------------------------------------------------------------------------
# Country scope, monthly (pe.100): our colonies under orbital bombardment. The damage follows
# the devastation of the planet (how hard the bombardment really hits: ships, technology,
# shields), only while it is bombarded:
# - each firm loses 0.2% of its capital a month per point of devastation (a random 0.1-0.3%
#   per point, per planet and month);
# - rarely a firm is hit (1% a month, up to 5% at high devastation): it loses a stage at its
#   next accounts, at stage 1 it is destroyed and leaves ruins (pe_firm_book, pe_firm_change);
# - the planet is a war zone for three years after the last bombardment (pe_warzone): no new
#   firms, no buyouts, no restructuring there.
pe_bombard_step = {
	every_owned_colony = {
		limit = { has_orbital_bombardment = yes }
		if = { limit = { has_modifier = pe_warzone } remove_modifier = pe_warzone }
		add_modifier = { modifier = pe_warzone days = 1080 }
		if = {
			limit = { count_deposits = { category = pe_deposit_cat_firms count > 0 } }
			export_trigger_value_to_variable = { trigger = planet_devastation variable = pe_dev }
			set_variable_to_random_value = { which = pe_bfac min = 0.001 max = 0.003 }
			set_variable = { which = pe_brate value = pe_dev }
			multiply_variable = { which = pe_brate value = pe_bfac }
			if = { limit = { check_variable = { which = pe_brate value > 0.9 } } set_variable = { which = pe_brate value = 0.9 } }
			set_variable = { which = pe_bkeep value = 1 }
			subtract_variable = { which = pe_bkeep value = pe_brate }
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms is_variable_set = pe_fcap }
				if = { limit = { check_variable = { which = pe_fcap value > 0 } } multiply_variable = { which = pe_fcap value = prev.pe_bkeep } }
				if = {
					limit = { prev = { check_variable = { which = pe_dev value < 20 } } }
					random = { chance = 1 set_variable = { which = pe_bomb_hit value = 1 } }
				}
				else_if = {
					limit = { prev = { check_variable = { which = pe_dev value < 40 } } }
					random = { chance = 2 set_variable = { which = pe_bomb_hit value = 1 } }
				}
				else_if = {
					limit = { prev = { check_variable = { which = pe_dev value < 60 } } }
					random = { chance = 3 set_variable = { which = pe_bomb_hit value = 1 } }
				}
				else_if = {
					limit = { prev = { check_variable = { which = pe_dev value < 80 } } }
					random = { chance = 4 set_variable = { which = pe_bomb_hit value = 1 } }
				}
				else = {
					random = { chance = 5 set_variable = { which = pe_bomb_hit value = 1 } }
				}
			}
		}
	}
}

# Country scope, quarterly: ruins of private firms on planets no longer bombarded (one site
# per planet as always). A corporation rebuilds its own firm's ruins from its cash (the
# founding cost, keeping 300); for other ruins we get an offer at most once a year (pe.296):
# a subsidy of half the founding cost, the investor pool lends the rest.
pe_ruins_step = {
	every_owned_colony = {
		limit = {
			count_deposits = { category = pe_deposit_cat_ruins count > 0 }
			has_orbital_bombardment = no
			NOT = { has_planet_flag = pe_site }
		}
		root = { set_variable = { which = pe_rb_done value = 0 } }
		random_deposit = {
			limit = { is_deposit_type = d_pe_ruins is_variable_set = pe_ruin_corp check_variable = { which = pe_ruin_corp value > 0 } }
			root = {
				set_variable = { which = pe_rb_code value = prev.pe_ruin_code }
				set_variable = { which = pe_rb_corp value = prev.pe_ruin_corp }
				pe_ruin_cost = yes
				set_variable = { which = pe_rb_need value = pe_rb_cost }
				change_variable = { which = pe_rb_need value = 300 }
				if = { limit = { check_variable = { which = pe_rb_corp value = 1 } check_variable = { which = pe_c1_on value = 1 } check_variable = { which = pe_c1_cash value >= pe_rb_need } } subtract_variable = { which = pe_c1_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c1_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 2 } check_variable = { which = pe_c2_on value = 1 } check_variable = { which = pe_c2_cash value >= pe_rb_need } } subtract_variable = { which = pe_c2_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c2_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 3 } check_variable = { which = pe_c3_on value = 1 } check_variable = { which = pe_c3_cash value >= pe_rb_need } } subtract_variable = { which = pe_c3_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c3_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 4 } check_variable = { which = pe_c4_on value = 1 } check_variable = { which = pe_c4_cash value >= pe_rb_need } } subtract_variable = { which = pe_c4_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c4_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 5 } check_variable = { which = pe_c5_on value = 1 } check_variable = { which = pe_c5_cash value >= pe_rb_need } } subtract_variable = { which = pe_c5_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c5_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 6 } check_variable = { which = pe_c6_on value = 1 } check_variable = { which = pe_c6_cash value >= pe_rb_need } } subtract_variable = { which = pe_c6_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c6_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 7 } check_variable = { which = pe_c7_on value = 1 } check_variable = { which = pe_c7_cash value >= pe_rb_need } } subtract_variable = { which = pe_c7_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c7_name } }
				else_if = { limit = { check_variable = { which = pe_rb_corp value = 8 } check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_cash value >= pe_rb_need } } subtract_variable = { which = pe_c8_cash value = pe_rb_cost } set_variable = { which = pe_rb_done value = 1 } set_variable = { which = pe_msg_name value = pe_c8_name } }
			}
			if = { limit = { root = { check_variable = { which = pe_rb_done value = 1 } } } remove_deposit = yes }
		}
		if = {
			limit = { root = { check_variable = { which = pe_rb_done value = 1 } } }
			root = { set_variable = { which = pe_rb_fcap value = 10 } set_variable = { which = pe_rb_debt value = 0 } }
			pe_ruin_rebuild = yes
			save_event_target_as = pe_msg_planet
			if = { limit = { owner = { is_ai = no } } root = { if = { limit = { NOT = { has_country_flag = pe_mute_corporations } } pe_corp_msg = { KEY = pe_msg_corp_rebuild CATEGORY = corporations } } } }
		}
		# Other ruins: an offer for a subsidy (humans), the investors rebuild them (AI).
		else_if = {
			limit = { root = { NOT = { has_country_flag = pe_rebuild_cd } } }
			random_deposit = {
				limit = { is_deposit_type = d_pe_ruins }
				root = {
					set_variable = { which = pe_rb_code value = prev.pe_ruin_code }
					set_variable = { which = pe_rb_corp value = 0 }
					pe_ruin_cost = yes
					set_variable = { which = pe_rb_sub value = pe_rb_cost }
					multiply_variable = { which = pe_rb_sub value = 0.5 }
				}
				save_event_target_as = pe_rb_ruin
			}
			save_event_target_as = pe_rb_planet
			root = {
				set_timed_country_flag = { flag = pe_rebuild_cd days = 360 }
				if = {
					limit = { is_ai = yes check_variable = { which = pe_capital value >= pe_rb_cost } }
					subtract_variable = { which = pe_capital value = pe_rb_cost }
					event_target:pe_rb_ruin = { remove_deposit = yes }
					set_variable = { which = pe_rb_fcap value = 10 }
					set_variable = { which = pe_rb_debt value = 0 }
					event_target:pe_rb_planet = { pe_ruin_rebuild = yes }
				}
				else_if = {
					limit = { is_ai = no }
					country_event = { id = pe.296 }
				}
			}
		}
	}
}

# Country scope (pe.296): we pay half the founding cost, the investor pool lends the rest; the
# firm is rebuilt as an independent firm (site of its old variant).
pe_ruin_subsidy = {
	add_resource = { trade = -1 mult = pe_rb_sub }
	set_variable = { which = pe_rb_fcap value = 10 }
	set_variable = { which = pe_rb_debt value = pe_rb_cost }
	subtract_variable = { which = pe_rb_debt value = pe_rb_sub }
	event_target:pe_rb_ruin = { remove_deposit = yes }
	event_target:pe_rb_planet = { pe_ruin_rebuild = yes }
}

# Colony scope: every firm here becomes a state district of its type, sites and private
# automation and the planet's ledgers go (the Command Economy, a conqueror without a private
# economy).
pe_firms_to_state = {
	while = {
		limit = { count_deposits = { category = pe_deposit_cat_firms count > 0 } }
		root = { set_variable = { which = pe_cmd_cls value = 0 } }
		random_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_firms }
			if = { limit = { NOT = { is_variable_set = pe_code } } pe_firm_identify = yes }
			if = { limit = { NOT = { is_variable_set = pe_cls } } pe_deposit_fetch = yes }
			root = { set_variable = { which = pe_cmd_cls value = prev.pe_cls } }
			remove_deposit = yes
		}
		root = { set_variable = { which = pe_sd_cls value = pe_cmd_cls } if = { limit = { check_variable = { which = pe_sd_cls value = 0 } } set_variable = { which = pe_sd_cls value = 1 } } }
		pe_state_district = yes
	}
	while = {
		limit = { count_deposits = { category = pe_deposit_cat_sites count > 0 } }
		root = { set_variable = { which = pe_cmd_cls value = 0 } }
		random_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_sites }
			if = {
				limit = { check_variable = { which = pe_kind value = 3 } }
				root = { set_variable = { which = pe_cmd_cls value = 1 } }
				if = { limit = { is_deposit_type = d_pe_site_mining } root = { set_variable = { which = pe_cmd_cls value = 2 } } }
				if = { limit = { is_deposit_type = d_pe_site_generator } root = { set_variable = { which = pe_cmd_cls value = 3 } } }
				if = { limit = { is_deposit_type = d_pe_site_farming } root = { set_variable = { which = pe_cmd_cls value = 4 } } }
			}
			remove_deposit = yes
		}
		if = {
			limit = { root = { check_variable = { which = pe_cmd_cls value > 0 } } }
			root = { set_variable = { which = pe_sd_cls value = pe_cmd_cls } }
			pe_state_district = yes
		}
	}
	while = {
		limit = { count_deposits = { category = pe_deposit_cat_automation count > 0 } }
		random_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_automation }
			remove_deposit = yes
		}
	}
	set_variable = { which = pe_al_miner value = 0 }
	set_variable = { which = pe_al_technician value = 0 }
	set_variable = { which = pe_al_farmer value = 0 }
	if = { limit = { has_deposit = d_pe_ledger_profit } remove_deposit = d_pe_ledger_profit }

# Colony scope: a state district of the class root.pe_sd_cls (1 city, 2 mining, 3 generator,
# 4 farming, 5 lab) instead of a firm; on a habitat the habitat district (housing, mining,
# energy for power plants and farms, research for labs).
pe_state_district = {
	if = {
		limit = { uses_district_set = habitat }
		if = { limit = { root = { check_variable = { which = pe_sd_cls value = 2 } } } add_district = { district_type = district_hab_mining ignore_cap = yes } }
		else_if = { limit = { root = { OR = { check_variable = { which = pe_sd_cls value = 3 } check_variable = { which = pe_sd_cls value = 4 } } } } add_district = { district_type = district_hab_energy ignore_cap = yes } }
		else_if = { limit = { root = { check_variable = { which = pe_sd_cls value = 5 } } } add_district = { district_type = district_hab_science ignore_cap = yes } }
		else = { add_district = { district_type = district_hab_housing ignore_cap = yes } }
	}
	else = {
		if = { limit = { root = { check_variable = { which = pe_sd_cls value = 2 } } } add_district = { district_type = district_mining ignore_cap = yes } }
		else_if = { limit = { root = { check_variable = { which = pe_sd_cls value = 3 } } } add_district = { district_type = district_generator ignore_cap = yes } }
		else_if = { limit = { root = { check_variable = { which = pe_sd_cls value = 4 } } } add_district = { district_type = district_farming ignore_cap = yes } }
		else = { add_district = { district_type = district_city ignore_cap = yes } }
	}
}

# --- Empire size ---------------------------------------------------------------------------
# Country scope, monthly (pe.100) and right after a new perk, tradition, policy or technology
# (pe.36): the empire size of the private economy, as two country modifiers
# with a multiplier (pe_sprawl_firms +0.5, pe_sprawl_corp_hab -20 each):
# - every firm counts like the district whose slot it holds: 1 + the planet's
#   empire_size_districts_mult (pe_spr_d), times its planetary ascension (vanilla -5% per
#   tier, times 1 + planetary_ascension_effect_mult);
# - every corporate habitat (planet flag pe_corp_hab; pe_corp_hab_test for the test events)
#   takes its colony out again: 1 + empire_size_colonies_mult, times its ascension; and its
#   districts (pe_sprawl_corp_hab_d -0.5 each, like the firms above); its firms do not count.
# pe_spr_firms / pe_spr_hab / pe_spr_hd = the multipliers (for the log).
pe_sprawl_step = {
	set_variable = { which = pe_spr_firms value = 0 }
	set_variable = { which = pe_spr_hab value = 0 }
	set_variable = { which = pe_spr_hd value = 0 }
	export_modifier_to_variable = { modifier = empire_size_colonies_mult variable = pe_spr_cm }
	export_modifier_to_variable = { modifier = planetary_ascension_effect_mult variable = pe_spr_am }
	every_owned_colony = {
		limit = {
			OR = {
				count_deposits = { category = pe_deposit_cat_firms count > 0 }
				has_planet_flag = pe_corp_hab
				has_planet_flag = pe_corp_hab_test
			}
		}
		# Ascension factor: 1 - 0.05 x tier x (1 + planetary_ascension_effect_mult).
		set_variable = { which = pe_spr_asc value = 0 }
		if = { limit = { has_planetary_ascension_tier >= 1 } export_trigger_value_to_variable = { trigger = has_planetary_ascension_tier variable = pe_spr_asc } }
		multiply_variable = { which = pe_spr_asc value = -0.05 }
		set_variable = { which = pe_tmp value = root.pe_spr_am }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_spr_asc value = pe_tmp }
		change_variable = { which = pe_spr_asc value = 1 }
		if = { limit = { check_variable = { which = pe_spr_asc value < 0 } } set_variable = { which = pe_spr_asc value = 0 } }
		export_modifier_to_variable = { modifier = empire_size_districts_mult variable = pe_spr_d }
		# A corporate habitat: no colony, its districts out again, its firms not counted.
		if = {
			limit = { OR = { has_planet_flag = pe_corp_hab has_planet_flag = pe_corp_hab_test } }
			set_variable = { which = pe_tmp value = root.pe_spr_cm }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_tmp value = pe_spr_asc }
			root = { change_variable = { which = pe_spr_hab value = prev.pe_tmp } }
			export_trigger_value_to_variable = { trigger = num_districts parameters = { type = any } variable = pe_spr_n }
			if = { limit = { NOT = { is_variable_set = pe_spr_n } } set_variable = { which = pe_spr_n value = 0 } }
			set_variable = { which = pe_tmp value = pe_spr_d }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_tmp value = pe_spr_n }
			multiply_variable = { which = pe_tmp value = pe_spr_asc }
			root = { change_variable = { which = pe_spr_hd value = prev.pe_tmp } }
		}
		# The firms here, like districts.
		else = {
			set_variable = { which = pe_spr_n value = 0 }
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms }
				prev = { change_variable = { which = pe_spr_n value = 1 } }
			}
			set_variable = { which = pe_tmp value = pe_spr_d }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_tmp value = pe_spr_n }
			multiply_variable = { which = pe_tmp value = pe_spr_asc }
			root = { change_variable = { which = pe_spr_firms value = prev.pe_tmp } }
		}
	}
	if = { limit = { has_modifier = pe_sprawl_firms } remove_modifier = pe_sprawl_firms }
	if = { limit = { has_modifier = pe_sprawl_corp_hab } remove_modifier = pe_sprawl_corp_hab }
	if = { limit = { check_variable = { which = pe_spr_firms value > 0 } } add_modifier = { modifier = pe_sprawl_firms days = -1 multiplier = pe_spr_firms } }
	if = { limit = { check_variable = { which = pe_spr_hab value > 0 } } add_modifier = { modifier = pe_sprawl_corp_hab days = -1 multiplier = pe_spr_hab } }
	if = { limit = { has_modifier = pe_sprawl_corp_hab_d } remove_modifier = pe_sprawl_corp_hab_d }
	if = { limit = { check_variable = { which = pe_spr_hd value > 0 } } add_modifier = { modifier = pe_sprawl_corp_hab_d days = -1 multiplier = pe_spr_hd } }
}

# Colony scope, root = country: a lease accepted in pe.212 is added to this quarter's estimate
# of the automation upkeep at once (as pe_auto_book does each quarter: 1% of the private
# workforce of the job x the lease size, per month), so the refund follows next month.
pe_lease_est = {
	root = { set_variable = { which = pe_acap value = 0 } }
	every_deposit = {
		limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
		root = { change_variable = { which = pe_acap value = prev.pe_wf_$JOB$ } }
	}
	root = {
		set_variable = { which = pe_tmp2 value = pe_acap }
		multiply_variable = { which = pe_tmp2 value = 0.01 }
		multiply_variable = { which = pe_tmp2 value = pe_lease_mult }
		if = { limit = { NOT = { is_variable_set = pe_auto_p } } set_variable = { which = pe_auto_p value = 0 } }
		change_variable = { which = pe_auto_p value = pe_tmp2 }
		prev = { change_variable = { which = pe_auto_pp value = root.pe_tmp2 } }
	}
}

# Country scope, monthly (pe.100, after the quarter): the energy upkeep the engine charges for
# the automated jobs of private automation (pe_auto_p, per month: 4 per automated job, summed
# by pe_auto_book each quarter, a new lease added at once by pe_lease_est) is carried by the
# firms (pe_aup in their accounts). They pay it back to us as a budget income line (deposit
# d_pe_ledger_auto on the capital, category pe_auto_refund, energy in bits la), never more
# than the line itself (pe_auto_r). Monthly, because the line grows a month after a lease.
# No modifier lowers that upkeep (0.34.1-0.35.1 tried three; the old ones go).
pe_auto_cover_step = {
	if = { limit = { has_modifier = pe_auto_cover } remove_modifier = pe_auto_cover }
	export_trigger_value_to_variable = { trigger = resource_expenses_compare parameters = { resource = energy category = planet_automated_jobs } variable = pe_auto_e }
	if = { limit = { NOT = { is_variable_set = pe_auto_e } } set_variable = { which = pe_auto_e value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_auto_p } } set_variable = { which = pe_auto_p value = 0 } }
	# Never more than the line itself.
	set_variable = { which = pe_auto_r value = pe_auto_p }
	if = { limit = { check_variable = { which = pe_auto_r value > pe_auto_e } } set_variable = { which = pe_auto_r value = pe_auto_e } }
	set_variable = { which = pe_lv value = pe_auto_r }
	capital_scope = {
		pe_ledger_bits = { V = la }
		if = {
			limit = { root = { check_variable = { which = pe_auto_r value >= 0.5 } } }
			if = { limit = { NOT = { has_deposit = d_pe_ledger_auto } } add_deposit = d_pe_ledger_auto }
		}
		else_if = { limit = { has_deposit = d_pe_ledger_auto } remove_deposit = d_pe_ledger_auto }
	}
	if = {
		limit = { check_variable = { which = pe_months value = 0 } }
		log = "private_economy: automation refund: private \\[Root.pe_auto_r] of \\[Root.pe_auto_p] energy a month, budget line \\[Root.pe_auto_e]"
	}
}

# Colony scope, root = country (pe.323): found the firm root.pe_xq_code beyond the economic
# order's share. Only with a free district slot besides the last one. MODE 1 allowed, 2 with
# a licence fee (pe_xq_fee trade from the investor pool to us, only if the firm is founded).
pe_xq_found = {
	if = {
		limit = { free_district_slots > 1 }
		root = {
			set_variable = { which = pe_best_variant value = pe_xq_code }
			set_variable = { which = pe_chab_pk value = 0 }
		}
		export_trigger_value_to_variable = { trigger = count_deposits parameters = { category = pe_deposit_cat_sites } variable = pe_sc0 }
		pe_found_variant = yes
		export_trigger_value_to_variable = { trigger = count_deposits parameters = { category = pe_deposit_cat_sites } variable = pe_sc1 }
		if = {
			limit = { check_variable = { which = pe_sc1 value > pe_sc0 } }
			if = { limit = { NOT = { is_variable_set = pe_xq } } set_variable = { which = pe_xq value = 0 } }
			change_variable = { which = pe_xq value = 1 }
			if = {
				limit = { root = { check_variable = { which = pe_xq_mode value = 2 } } }
				root = {
					add_resource = { trade = 1 mult = pe_xq_fee }
					subtract_variable = { which = pe_capital value = pe_xq_fee }
				}
			}
		}
	}
	log = "private_economy: firm beyond the share on \\[This.GetName]: mode \\[Root.pe_xq_mode], sites before \\[This.pe_sc0] after \\[This.pe_sc1]"
}
