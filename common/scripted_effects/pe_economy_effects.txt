# Private economy logic. Runs quarterly per country (pe.100); construction sites
# advance monthly.
# Call depth stays at 4 or less: the engine initialises only 5 nested levels.
#   pe_monthly_sites (1) -> pe_site_complete (2)
#   pe_quarterly (1) -> pe_init (2)
#                    -> pe_census (2)
#                    -> pe_market (2) -> pe_shadow (3)
#                    -> pe_job_values (2) -> pe_job_count (3), pe_job_res (3)
#                    -> pe_compute_values (2)
#                    -> pe_planet_step (2) -> pe_planet_counts (3)
#                                          -> pe_planet_prepare (3)
#                                          -> pe_planet_fill (3)      -> pe_count_planet_job (4)
#                                          -> pe_firm_accounting (3)  -> pe_firm_identify (4)
#                                                                     -> pe_deposit_fetch (4)
#                                                                     -> pe_firm_book (4)
#                                          -> pe_firm_change (3)
#                                          -> pe_choose_variant (3)   -> pe_consider_variant (4)
#                                          -> pe_found_variant (3)    -> pe_found_firm (4)
#                                          -> pe_buy_or_offer (3)     -> pe_buyout_execute (4)
#                                          -> pe_nationalise_one (3)
#                                          -> pe_restructure (3)      -> pe_retool_consider (4)
#                    -> pe_relocate (2) -> pe_relocate_from (3) -> pe_relocate_exec (4)
#                    -> pe_ledgers (2) -> pe_ledger_planet (3) -> pe_ledger_bits (4)
#                                      -> pe_ledger_bits (3)
# Variant lists and dispatchers: pe_firm_effects_generated.txt (tools/generate-firms.mjs).
#
# The loop: shadow prices (market price plus need) -> value added per private job ->
# the firms keep a profit share of it (the economic order lowers private output by the
# same share) -> firm capital and investor pool -> firms grow in stages, the pool builds
# new firms and buys districts -> more supply (construction sites count in advance)
# lowers the price -> less profit -> restructuring or bankruptcy.
#
# Capital is counted in trade. Investor pool (country pe_capital): starts at 500, gets
# private savings (a share of the trade income, by economic order) and half of each
# debt-free firm's profit (all of it from corporations). It pays 40% down on foundings
# (the state's district cost: city 500, rural 300) and buyouts; the firm owes the rest
# and repays it before it pays out or grows. It also pays restructuring when the firm
# cannot. Firm capital (deposit pe_fcap): the other half; pays expansions and
# restructuring, absorbs losses.
#
# Building: a firm is first a construction site (vanilla district build time: city 16,
# rural 8 months; buyouts and restructuring half of that). One site per planet. New
# firms only where civilians are idle (4.x has no unemployed) and the planet's private
# jobs are at least 80% filled; otherwise investors buy state districts (Free Market,
# Laissez-faire), never the last one of a type. New firms and restructuring targets may
# not consume a resource in deficit. Every change of jobs re-runs the planet's employment.

# --- Monthly ------------------------------------------------------------------

pe_monthly_sites = {
	every_owned_colony = {
		limit = { has_planet_flag = pe_site }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_sites }
			change_variable = { which = pe_left value = -1 }
		}
		pe_site_complete = yes
	}
}

# --- Country level ------------------------------------------------------------

pe_init = {
	if = { limit = { NOT = { is_variable_set = pe_private_share } } set_variable = { which = pe_private_share value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_profit_share } } set_variable = { which = pe_profit_share value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_savings_rate } } set_variable = { which = pe_savings_rate value = 0 } }
	if = { limit = { NOT = { is_variable_set = pe_capital } } set_variable = { which = pe_capital value = 500 } }
	if = { limit = { NOT = { is_variable_set = pe_pool_income } } set_variable = { which = pe_pool_income value = 0 } }
	if = {
		limit = { NOT = { is_variable_set = pe_seed } }
		set_variable_to_random_value = { which = pe_seed min = 0 max = 1008 rounded = yes }
	}
	# Spread the empires over the three months of a quarter.
	if = {
		limit = { NOT = { is_variable_set = pe_months } }
		set_variable_to_random_value = { which = pe_months min = 0 max = 2 rounded = yes }
	}
}

pe_quarterly = {
	pe_init = yes
	set_variable = { which = pe_pool_income value = 0 }
	set_variable = { which = pe_profit_due value = 0 }
	pe_savings = yes
	pe_census = yes
	pe_market = yes
	pe_job_values = yes
	pe_compute_values = yes
	# Planets with idle civilians first: founders start where workers are available.
	every_owned_colony = {
		limit = { pe_planet_eligible = yes pe_has_idle_pops = yes }
		set_timed_planet_flag = { flag = pe_stepped days = 1 }
		pe_planet_step = yes
	}
	every_owned_colony = {
		limit = { pe_planet_eligible = yes NOT = { has_planet_flag = pe_stepped } }
		pe_planet_step = yes
	}
	pe_relocate = yes
	pe_ledgers = yes
}

# Country scope: when a planet is stuck (no free district slot, no idle civilians, no
# construction), one city firm or lab whose main share does not suit the planet's
# designation may move to a planet that suits it, with its workers. At most once a year
# per empire. The destination needs a free slot, no construction site, and free housing
# and amenities for all workers; the firm pays the move like a restructuring (250 x stage
# size) from its own capital.
pe_relocate = {
	set_variable = { which = pe_rl_done value = 0 }
	if = {
		limit = { NOT = { has_country_flag = pe_relocated } }
		every_owned_colony = {
			limit = {
				pe_planet_eligible = yes
				free_district_slots = 0
				NOT = { pe_has_idle_pops = yes }
				NOT = { has_planet_flag = pe_site }
				count_deposits = { category = pe_deposit_cat_firms count > 0 }
			}
			if = {
				limit = { root = { check_variable = { which = pe_rl_done value = 0 } } }
				pe_relocate_from = yes
			}
		}
	}
}

# Colony scope (a stuck planet): find a firm and a destination, then move it.
pe_relocate_from = {
	save_event_target_as pe_rl_src
	every_deposit = {
		limit = {
			has_deposit_category = pe_deposit_cat_firms
			is_variable_set = pe_prim
			OR = {
				check_variable = { which = pe_cls value = 1 }
				check_variable = { which = pe_cls value = 5 }
			}
		}
		set_variable = { which = pe_rl_cand value = 0 }
		if = {
			limit = { root = { check_variable = { which = pe_rl_done value = 0 } } }
			# Workers (200 x stage size) and the moving cost (250 x stage size).
			set_variable = { which = pe_tmp value = pe_stage }
			change_variable = { which = pe_tmp value = 1 }
			multiply_variable = { which = pe_tmp value = 0.5 }
			root = {
				set_variable = { which = pe_rl_prim value = prev.pe_prim }
				set_variable = { which = pe_rl_wf value = prev.pe_tmp }
				multiply_variable = { which = pe_rl_wf value = 200 }
				set_variable = { which = pe_rl_cost value = prev.pe_tmp }
				multiply_variable = { which = pe_rl_cost value = 250 }
			}
			if = {
				limit = {
					check_variable = { which = pe_fcap value >= root.pe_rl_cost }
					event_target:pe_rl_src = { NOT = { pe_desig_prim = yes } }
				}
				root = {
					every_owned_colony = {
						limit = {
							pe_planet_eligible = yes
							NOT = { is_same_value = event_target:pe_rl_src }
							free_district_slots > 0
							NOT = { has_planet_flag = pe_site }
							pe_desig_prim = yes
						}
						export_trigger_value_to_variable = { trigger = free_housing variable = pe_fh }
						export_trigger_value_to_variable = { trigger = free_amenities variable = pe_fa }
						if = {
							limit = {
								root = { check_variable = { which = pe_rl_done value = 0 } }
								check_variable = { which = pe_fh value >= root.pe_rl_wf }
								check_variable = { which = pe_fa value >= root.pe_rl_wf }
							}
							save_event_target_as pe_rl_dst
							root = { set_variable = { which = pe_rl_done value = 1 } }
						}
					}
				}
				if = {
					limit = { root = { check_variable = { which = pe_rl_done value = 1 } } }
					set_variable = { which = pe_rl_cand value = 1 }
				}
			}
		}
	}
	if = {
		limit = { root = { check_variable = { which = pe_rl_done value = 1 } } }
		pe_relocate_exec = yes
	}
}

# Country scope: what the private economy costs the state, as monthly trade upkeep that
# shows in the budget: each planet's firms' profits (profit ledger) and the private savings
# (savings ledger on the capital), for the next three months. Together at most half of
# the gross trade income.
pe_ledgers = {
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = trade } variable = pe_lcap }
	multiply_variable = { which = pe_lcap value = 1.5 }
	set_variable = { which = pe_ltot value = pe_profit_due }
	change_variable = { which = pe_ltot value = pe_savings_last }
	set_variable = { which = pe_lscale value = 1 }
	if = {
		limit = {
			check_variable = { which = pe_ltot value > pe_lcap }
			check_variable = { which = pe_ltot value > 0 }
		}
		set_variable = { which = pe_lscale value = pe_lcap }
		divide_variable = { which = pe_lscale value = pe_ltot }
	}
	every_owned_colony = { pe_ledger_planet = yes }
	set_variable = { which = pe_sav_month value = pe_savings_last }
	multiply_variable = { which = pe_sav_month value = pe_lscale }
	divide_variable = { which = pe_sav_month value = 3 }
	set_variable = { which = pe_profit_month value = pe_profit_due }
	multiply_variable = { which = pe_profit_month value = pe_lscale }
	divide_variable = { which = pe_profit_month value = 3 }
	capital_scope = {
		root = { set_variable = { which = pe_lv value = pe_sav_month } }
		pe_ledger_bits = { V = ls }
		if = {
			limit = { root = { check_variable = { which = pe_sav_month value >= 0.5 } } }
			if = { limit = { NOT = { has_deposit = d_pe_ledger_savings } } add_deposit = d_pe_ledger_savings }
		}
		else_if = {
			limit = { has_deposit = d_pe_ledger_savings }
			remove_deposit = d_pe_ledger_savings
		}
	}
}

# Colony scope: this planet's profit ledger for the next three months.
pe_ledger_planet = {
	if = {
		limit = { count_deposits = { category = pe_deposit_cat_firms count < 1 } }
		set_variable = { which = pe_pl_profit value = 0 }
	}
	if = { limit = { NOT = { is_variable_set = pe_pl_profit } } set_variable = { which = pe_pl_profit value = 0 } }
	root = {
		set_variable = { which = pe_lv value = prev.pe_pl_profit }
		multiply_variable = { which = pe_lv value = pe_lscale }
		divide_variable = { which = pe_lv value = 3 }
	}
	pe_ledger_bits = { V = lp }
	if = {
		limit = { check_variable = { which = pe_pl_profit value > 1.5 } }
		if = { limit = { NOT = { has_deposit = d_pe_ledger_profit } } add_deposit = d_pe_ledger_profit }
	}
	else_if = {
		limit = { has_deposit = d_pe_ledger_profit }
		remove_deposit = d_pe_ledger_profit
	}
}

# Country scope: private savings. A share of the gross trade income (pe_savings_rate, set by
# the economic order: 0/5/10/15/20%) goes to the investor pool each quarter; the state
# pays it monthly as trade upkeep (savings ledger on the capital, see pe_ledgers). Only if the stockpile can pay it, and only while the pool holds less than
# 3000 (otherwise it could not spend it).
pe_savings = {
	set_variable = { which = pe_savings_last value = 0 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = trade } variable = pe_sav }
	multiply_variable = { which = pe_sav value = pe_savings_rate }
	multiply_variable = { which = pe_sav value = 3 }
	export_resource_stockpile_to_variable = { resource = trade variable = pe_stock }
	if = {
		limit = {
			check_variable = { which = pe_sav value > 0 }
			check_variable = { which = pe_stock value >= pe_sav }
			# A full pool (it has nothing to spend it on) saves no more.
			check_variable = { which = pe_capital value < 3000 }
		}
		change_variable = { which = pe_capital value = pe_sav }
		change_variable = { which = pe_pool_income value = pe_sav }
		set_variable = { which = pe_savings_last value = pe_sav }
	}
}

# Country scope: shadow price pe_s_<res> in trade per unit.
#   Investment goods: alloys and food for empires with bio-ships are worth their base
#   value (4 / 1) while the stock is below a quarter of the storage, falling to the selling
#   price at three quarters and to a third of it when nearly full. Unity (3) and research
#   (3.5 per point) have no stock. Alloys, research and unity are then balanced against
#   each other (pe_triad).
#   Flow goods: the buying price while short, falling to the selling price at the
#   target surplus (BAND, % of production); above it a dumping price (a third of the
#   selling price), with a full stock less still. A deficit adds a premium. Food only
#   needs to stay just above
#   zero (5%), unless alloys are made from it (Catalytic Processing: 20%).
#   Trade is money: worth 1, but a surplus only buys goods after market fees (0.7).
pe_market = {
	set_variable = { which = pe_emergency value = 0 }
	set_variable = { which = pe_is_invest value = 1 }
	pe_shadow = { RES = alloys BASE = 4 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	# No stock and no deficit for research and unity: fixed values.
	set_variable = { which = pe_s_unity value = 3 }
	set_variable = { which = pe_s_physics_research value = 3.5 }
	set_variable = { which = pe_s_society_research value = 3.5 }
	set_variable = { which = pe_s_engineering_research value = 3.5 }
	set_variable = { which = pe_is_invest value = 0 }
	if = {
		limit = { country_uses_bio_ships = yes }
		set_variable = { which = pe_is_invest value = 1 }
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
		set_variable = { which = pe_is_invest value = 0 }
	}
	else_if = {
		limit = { is_catalytic_empire = yes }
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	}
	else = {
		pe_shadow = { RES = food BASE = 1 SELL = 0 BAND = 5 MARKET = yes PEND = yes }
	}
	pe_shadow = { RES = consumer_goods BASE = 2 SELL = 0 BAND = 15 MARKET = yes PEND = yes }
	pe_shadow = { RES = minerals BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	pe_shadow = { RES = energy BASE = 1 SELL = 0 BAND = 20 MARKET = yes PEND = yes }
	pe_shadow = { RES = trade BASE = 1 SELL = 0.7 BAND = 20 PEND = yes }
	pe_triad = yes
}

# Country scope: the investment triad. Alloys, research and unity are all needed; each
# gets a target share of the empire's investment output (valued at 4 / 3.5 / 3 per unit):
# 35 / 45 / 20%, shifted by ethics (militarist or at war: alloys, pacifist: less alloys,
# materialist: research, spiritualist: unity). Below its target a good is worth more,
# above it less: factor 0.5 + 0.5 x target / share, between 0.5 and 2.
pe_triad = {
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = alloys } variable = pe_tri_a }
	multiply_variable = { which = pe_tri_a value = 4 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = physics_research } variable = pe_tri_r }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = society_research } variable = pe_tmp }
	change_variable = { which = pe_tri_r value = pe_tmp }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = engineering_research } variable = pe_tmp }
	change_variable = { which = pe_tri_r value = pe_tmp }
	multiply_variable = { which = pe_tri_r value = 3.5 }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = unity } variable = pe_tri_u }
	multiply_variable = { which = pe_tri_u value = 3 }
	set_variable = { which = pe_tri_sum value = pe_tri_a }
	change_variable = { which = pe_tri_sum value = pe_tri_r }
	change_variable = { which = pe_tri_sum value = pe_tri_u }
	# Targets in percent.
	set_variable = { which = pe_tgt_a value = 35 }
	set_variable = { which = pe_tgt_r value = 45 }
	set_variable = { which = pe_tgt_u value = 20 }
	if = { limit = { is_militarist = yes } change_variable = { which = pe_tgt_a value = 10 } change_variable = { which = pe_tgt_r value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	if = { limit = { is_pacifist = yes } change_variable = { which = pe_tgt_a value = -10 } change_variable = { which = pe_tgt_r value = 5 } change_variable = { which = pe_tgt_u value = 5 } }
	if = { limit = { is_materialist = yes } change_variable = { which = pe_tgt_r value = 10 } change_variable = { which = pe_tgt_a value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	if = { limit = { is_spiritualist = yes } change_variable = { which = pe_tgt_u value = 10 } change_variable = { which = pe_tgt_a value = -5 } change_variable = { which = pe_tgt_r value = -5 } }
	if = { limit = { is_at_war = yes } change_variable = { which = pe_tgt_a value = 10 } change_variable = { which = pe_tgt_r value = -5 } change_variable = { which = pe_tgt_u value = -5 } }
	pe_triad_factor = { L = a }
	pe_triad_factor = { L = r }
	pe_triad_factor = { L = u }
	multiply_variable = { which = pe_s_alloys value = pe_tri_f_a }
	multiply_variable = { which = pe_s_physics_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_society_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_engineering_research value = pe_tri_f_r }
	multiply_variable = { which = pe_s_unity value = pe_tri_f_u }
}

# Country scope: share pe_tri_pct_<L> (percent of the triad) and factor pe_tri_f_<L>.
pe_triad_factor = {
	if = { limit = { check_variable = { which = pe_tgt_$L$ value < 10 } } set_variable = { which = pe_tgt_$L$ value = 10 } }
	set_variable = { which = pe_tri_f_$L$ value = 2 }
	set_variable = { which = pe_tri_pct_$L$ value = 0 }
	if = {
		limit = {
			check_variable = { which = pe_tri_sum value > 0 }
			check_variable = { which = pe_tri_$L$ value > 0 }
		}
		set_variable = { which = pe_tri_pct_$L$ value = pe_tri_$L$ }
		divide_variable = { which = pe_tri_pct_$L$ value = pe_tri_sum }
		multiply_variable = { which = pe_tri_pct_$L$ value = 100 }
		set_variable = { which = pe_tri_f_$L$ value = pe_tgt_$L$ }
		divide_variable = { which = pe_tri_f_$L$ value = pe_tri_pct_$L$ }
		multiply_variable = { which = pe_tri_f_$L$ value = 0.5 }
		change_variable = { which = pe_tri_f_$L$ value = 0.5 }
		if = { limit = { check_variable = { which = pe_tri_f_$L$ value > 2 } } set_variable = { which = pe_tri_f_$L$ value = 2 } }
		if = { limit = { check_variable = { which = pe_tri_f_$L$ value < 0.5 } } set_variable = { which = pe_tri_f_$L$ value = 0.5 } }
	}
}

pe_shadow = {
	export_resource_income_to_variable = { resource = $RES$ variable = pe_net }
	[[PEND]
	change_variable = { which = pe_net value = pe_pend_$RES$ }
	]
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = $RES$ } variable = pe_gross }
	# Shortage margin: below max(5, 5% of gross production), at most 50, a resource counts
	# as short, so the economy keeps a buffer instead of sliding into the minus.
	set_variable = { which = pe_margin value = pe_gross }
	multiply_variable = { which = pe_margin value = 0.05 }
	if = { limit = { check_variable = { which = pe_margin value < 5 } } set_variable = { which = pe_margin value = 5 } }
	if = { limit = { check_variable = { which = pe_margin value > 50 } } set_variable = { which = pe_margin value = 50 } }
	subtract_variable = { which = pe_net value = pe_margin }
	export_resource_stockpile_to_variable = { resource = $RES$ variable = pe_stock }
	export_resource_maximum_to_variable = { resource = $RES$ variable = pe_max }
	set_variable = { which = pe_scale value = pe_gross }
	if = { limit = { check_variable = { which = pe_scale value < 10 } } set_variable = { which = pe_scale value = 10 } }
	set_variable = { which = pe_rel value = pe_net }
	divide_variable = { which = pe_rel value = pe_scale }
	multiply_variable = { which = pe_rel value = 100 }
	[[MARKET]
	export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = $RES$ amount = 100 trade_type = market_buy } variable = pe_pbuy }
	divide_variable = { which = pe_pbuy value = 100 }
	if = { limit = { check_variable = { which = pe_pbuy value <= 0 } } set_variable = { which = pe_pbuy value = $BASE$ } multiply_variable = { which = pe_pbuy value = 1.3 } }
	export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = $RES$ amount = 100 trade_type = market_sell } variable = pe_psell }
	divide_variable = { which = pe_psell value = 100 }
	if = { limit = { check_variable = { which = pe_psell value <= 0 } } set_variable = { which = pe_psell value = $BASE$ } multiply_variable = { which = pe_psell value = 0.7 } }
	]
	[[!MARKET]
	set_variable = { which = pe_pbuy value = $BASE$ }
	set_variable = { which = pe_psell value = $SELL$ }
	]
	# pe_def_<res>: in deficit (pending supply included). Firms that consume it are not
	# founded and are the first to restructure.
	set_variable = { which = pe_def_$RES$ value = 0 }
	if = { limit = { check_variable = { which = pe_net value < 0 } } set_variable = { which = pe_def_$RES$ value = 1 } }
	# Investment goods start from their base value, not the market buying price with fees.
	if = { limit = { check_variable = { which = pe_is_invest value = 1 } } set_variable = { which = pe_pbuy value = $BASE$ } }
	set_variable = { which = pe_s_$RES$ value = pe_pbuy }
	if = {
		limit = {
			check_variable = { which = pe_is_invest value = 1 }
			check_variable = { which = pe_max value > 0 }
		}
		# Stock fraction f: pbuy - (pbuy - psell) * clamp((f - 0.25) / 0.5, 0, 1)
		set_variable = { which = pe_frac value = pe_stock }
		divide_variable = { which = pe_frac value = pe_max }
		set_variable = { which = pe_tmp value = pe_frac }
		change_variable = { which = pe_tmp value = -0.25 }
		multiply_variable = { which = pe_tmp value = 2 }
		if = { limit = { check_variable = { which = pe_tmp value < 0 } } set_variable = { which = pe_tmp value = 0 } }
		if = { limit = { check_variable = { which = pe_tmp value > 1 } } set_variable = { which = pe_tmp value = 1 } }
		set_variable = { which = pe_prem value = pe_pbuy }
		subtract_variable = { which = pe_prem value = pe_psell }
		multiply_variable = { which = pe_prem value = pe_tmp }
		subtract_variable = { which = pe_s_$RES$ value = pe_prem }
		if = {
			limit = { check_variable = { which = pe_frac value >= 0.95 } }
			set_variable = { which = pe_s_$RES$ value = pe_psell }
			multiply_variable = { which = pe_s_$RES$ value = 0.33 }
		}
	}
	if = {
		limit = {
			check_variable = { which = pe_is_invest value = 0 }
			check_variable = { which = pe_net value >= 0 }
		}
		if = {
			limit = { check_variable = { which = pe_rel value >= $BAND$ } }
			set_variable = { which = pe_s_$RES$ value = pe_psell }
			multiply_variable = { which = pe_s_$RES$ value = 0.33 }
		}
		else = {
			# pbuy - (pbuy - psell) * rel / BAND
			set_variable = { which = pe_tmp value = pe_pbuy }
			subtract_variable = { which = pe_tmp value = pe_psell }
			multiply_variable = { which = pe_tmp value = pe_rel }
			divide_variable = { which = pe_tmp value = $BAND$ }
			subtract_variable = { which = pe_s_$RES$ value = pe_tmp }
		}
		if = {
			limit = { check_variable = { which = pe_max value > 0 } }
			set_variable = { which = pe_full value = pe_max }
			multiply_variable = { which = pe_full value = 0.9 }
			if = {
				limit = { check_variable = { which = pe_stock value >= pe_full } }
				set_variable = { which = pe_s_$RES$ value = pe_psell }
				multiply_variable = { which = pe_s_$RES$ value = 0.15 }
			}
		}
	}
	if = {
		limit = { check_variable = { which = pe_net value < 0 } }
		# Any deficit +25%, plus 4% per 1% of production missing (up to +200%), plus
		# 50% if the stock lasts less than a year.
		set_variable = { which = pe_prem value = pe_rel }
		multiply_variable = { which = pe_prem value = -0.04 }
		change_variable = { which = pe_prem value = 0.25 }
		if = { limit = { check_variable = { which = pe_prem value > 2 } } set_variable = { which = pe_prem value = 2 } }
		set_variable = { which = pe_runway value = pe_net }
		multiply_variable = { which = pe_runway value = -12 }
		if = {
			limit = { check_variable = { which = pe_stock value < pe_runway } }
			change_variable = { which = pe_prem value = 0.5 }
			set_variable = { which = pe_emergency value = 1 }
		}
		change_variable = { which = pe_prem value = 1 }
		multiply_variable = { which = pe_s_$RES$ value = pe_prem }
	}
}

pe_planet_step = {
	# Assign new pops first, so idle civilians really are idle.
	check_planet_employment = yes
	pe_planet_counts = yes
	pe_planet_prepare = yes
	root = { set_variable = { which = pe_unemp value = 0 } }
	if = { limit = { pe_has_idle_pops = yes } root = { set_variable = { which = pe_unemp value = 1 } } }
	pe_planet_fill = yes
	pe_firm_accounting = yes
	pe_firm_change = yes
	pe_automation = yes
	if = {
		limit = {
			NOT = { has_planet_flag = pe_site }
			NOT = { has_planet_flag = pe_cooldown }
		}
		if = {
			limit = { check_variable = { which = pe_firms value < pe_target } }
			if = {
				limit = {
					free_district_slots > 0
					pe_has_idle_pops = yes
					root = { check_variable = { which = pe_fill_cur value >= 0.8 } }
				}
				# Option value: the last free slots are only used for clearly better firms.
				root = { set_variable = { which = pe_best_min value = 0 } set_variable = { which = pe_dmode value = 0 } }
				if = { limit = { free_district_slots = 1 } root = { set_variable = { which = pe_best_min value = 8 } } }
				else_if = { limit = { free_district_slots = 2 } root = { set_variable = { which = pe_best_min value = 4 } } }
				pe_choose_variant = { MODE = found }
				pe_found_variant = yes
			}
			else_if = {
				limit = { owner = { pe_buyout_allowed = yes } }
				# A firm must be worth clearly more than the state district it replaces.
				root = { set_variable = { which = pe_best_min value = 2 } set_variable = { which = pe_dmode value = 1 } }
				pe_choose_variant = { MODE = buy }
				pe_buy_or_offer = yes
			}
		}
		else_if = {
			limit = { check_variable = { which = pe_firms value > pe_target } }
			pe_nationalise_one = yes
		}
	}
	# Nothing started on this planet this quarter, its private jobs are filled, and no
	# restructuring in the last two years: one firm may restructure.
	if = {
		limit = {
			NOT = { has_planet_flag = pe_site }
			NOT = { has_planet_flag = pe_cooldown }
			NOT = { has_planet_flag = pe_retool_cd }
			check_variable = { which = pe_firms_real value > 0 }
			root = { check_variable = { which = pe_fill_cur value >= 0.8 } }
		}
		pe_restructure = yes
	}
}

# Deposit scope. Quarterly profit = value added x stage size x fill x profit share,
# minus overhead (0.3 per stage and month). A firm in debt pays all of it back first.
# Then half goes to the firm and half to the investor pool (a corporation keeps it all),
# but a firm only keeps a reserve for its own restructuring and next growth step; the
# rest goes to the pool. Losses hit the firm. Then at most one stage change
# per planet and quarter: expand (+100 workforce, half a district) when the firm can pay
# half a district's cost (city 250, rural 150),
# its jobs are filled and the planet has idle civilians; shrink or go bankrupt when its
# capital runs out.
pe_firm_book = {
	set_variable = { which = pe_profit value = pe_va }
	set_variable = { which = pe_tmp value = pe_stage }
	change_variable = { which = pe_tmp value = 1 }
	multiply_variable = { which = pe_tmp value = 0.5 }
	multiply_variable = { which = pe_profit value = pe_tmp }
	multiply_variable = { which = pe_profit value = root.pe_fill_cur }
	multiply_variable = { which = pe_profit value = root.pe_profit_share }
	set_variable = { which = pe_tmp value = pe_stage }
	multiply_variable = { which = pe_tmp value = 0.3 }
	subtract_variable = { which = pe_profit value = pe_tmp }
	multiply_variable = { which = pe_profit value = 3 }
	set_variable = { which = pe_pending value = 0 }
	if = { limit = { NOT = { is_variable_set = pe_debt } } set_variable = { which = pe_debt value = 0 } }
	if = {
		limit = { check_variable = { which = pe_profit value > 0 } }
		root = {
			change_variable = { which = pe_profit_due value = prev.pe_profit }
			change_variable = { which = pe_pl_acc value = prev.pe_profit }
		}
	}
	if = {
		# In debt: all profit pays the loan back; the rest of a final instalment is lost.
		limit = {
			check_variable = { which = pe_profit value > 0 }
			check_variable = { which = pe_debt value > 0 }
		}
		subtract_variable = { which = pe_debt value = pe_profit }
		if = { limit = { check_variable = { which = pe_debt value < 0 } } set_variable = { which = pe_debt value = 0 } }
	}
	else_if = {
		limit = { check_variable = { which = pe_profit value > 0 } }
		if = {
			limit = { check_variable = { which = pe_stage value < 3 } }
			set_variable = { which = pe_tmp value = pe_profit }
			multiply_variable = { which = pe_tmp value = 0.5 }
			change_variable = { which = pe_fcap value = pe_tmp }
			root = {
				change_variable = { which = pe_capital value = prev.pe_tmp }
				change_variable = { which = pe_pool_income value = prev.pe_tmp }
			}
		}
		else = {
			change_variable = { which = pe_fcap value = pe_profit }
		}
		# Reserve: the firm keeps what its own restructuring would cost (city 250 / rural
		# 150 x stage size), plus its next growth step below a corporation. The rest goes
		# to the investor pool.
		if = { limit = { check_variable = { which = pe_cls value = 2 } } set_variable = { which = pe_res value = 150 } }
		else_if = { limit = { check_variable = { which = pe_cls value = 3 } } set_variable = { which = pe_res value = 150 } }
		else_if = { limit = { check_variable = { which = pe_cls value = 4 } } set_variable = { which = pe_res value = 150 } }
		else = { set_variable = { which = pe_res value = 250 } }
		set_variable = { which = pe_tmp value = pe_stage }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_tmp value = 0.5 }
		set_variable = { which = pe_res2 value = pe_res }
		multiply_variable = { which = pe_res2 value = pe_tmp }
		if = { limit = { check_variable = { which = pe_stage value < 3 } } change_variable = { which = pe_res2 value = pe_res } }
		if = {
			limit = { check_variable = { which = pe_fcap value > pe_res2 } }
			set_variable = { which = pe_tmp value = pe_fcap }
			subtract_variable = { which = pe_tmp value = pe_res2 }
			set_variable = { which = pe_fcap value = pe_res2 }
			root = {
				change_variable = { which = pe_capital value = prev.pe_tmp }
				change_variable = { which = pe_pool_income value = prev.pe_tmp }
			}
		}
	}
	else = {
		change_variable = { which = pe_fcap value = pe_profit }
	}
	if = {
		limit = { root = { check_variable = { which = pe_pend_any value = 0 } } }
		if = {
			limit = { check_variable = { which = pe_fcap value < 0 } }
			set_variable = { which = pe_fcap value = 0 }
			if = { limit = { check_variable = { which = pe_stage value > 1 } } set_variable = { which = pe_pending value = 2 } }
			else = { set_variable = { which = pe_pending value = 3 } }
			root = { set_variable = { which = pe_pend_any value = 1 } }
		}
		else_if = {
			limit = {
				check_variable = { which = pe_stage value < 3 }
				check_variable = { which = pe_profit value > 0 }
				check_variable = { which = pe_debt value <= 0 }
				root = {
					check_variable = { which = pe_fill_cur value >= 0.9 }
					check_variable = { which = pe_unemp value = 1 }
				}
			}
			if = { limit = { check_variable = { which = pe_cls value = 2 } } set_variable = { which = pe_tmp value = 150 } }
			else_if = { limit = { check_variable = { which = pe_cls value = 3 } } set_variable = { which = pe_tmp value = 150 } }
			else_if = { limit = { check_variable = { which = pe_cls value = 4 } } set_variable = { which = pe_tmp value = 150 } }
			else = { set_variable = { which = pe_tmp value = 250 } }
			if = {
				limit = { check_variable = { which = pe_fcap value >= pe_tmp } }
				subtract_variable = { which = pe_fcap value = pe_tmp }
				set_variable = { which = pe_pending value = 1 }
				root = { set_variable = { which = pe_pend_any value = 1 } }
			}
		}
	}
}

# Value of a variant on this planet: country value, plus entertainment need here, times
# the designation bonus of its main share, times 0.8-1.2 (entrepreneurial spirit). When
# buying, minus the value of the state district it replaces (weighted like a variant).
# Colony scope: private automation of the three worker jobs vanilla automates.
pe_automation = {
	pe_auto_book = { JOB = miner }
	pe_auto_invest = { JOB = miner }
	pe_auto_book = { JOB = technician }
	pe_auto_invest = { JOB = technician }
	pe_auto_book = { JOB = farmer }
	pe_auto_invest = { JOB = farmer }
}

# Colony scope. Like the vanilla automation buildings: level I +25% automated workforce
# (tech_assembly_pattern, 600), level II +50% (tech_construction_templates, 800 more). A
# joint investment of the firms with this job here, paid from their capital by capital
# share (the investor pool adds what they lack). The state pays the engine's energy upkeep
# for automated jobs (4 per automated worker job a month); the firms pay it back each
# quarter, valued at the energy shadow price. Automated jobs need no pops and no wages, so
# the firms keep their whole value: each quarter the automated value minus the upkeep is
# booked to the firms by workforce share; a loss is carried by their capital (by capital
# share), then by the investor pool. The level drops if neither can carry a loss, or
# after two quarters with idle civilians on the planet
# (pops are there, automation is not needed).
pe_auto_book = {
	root = {
		set_variable = { which = pe_acap value = 0 }
		set_variable = { which = pe_apool value = 0 }
		set_variable = { which = pe_alvl value = 0 }
		set_variable = { which = pe_adown value = 0 }
	}
	every_deposit = {
		limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
		root = {
			change_variable = { which = pe_acap value = prev.pe_wf_$JOB$ }
			change_variable = { which = pe_apool value = prev.pe_fcap }
		}
	}
	if = { limit = { has_deposit = d_pe_auto_$JOB$_1 } root = { set_variable = { which = pe_alvl value = 1 } } }
	if = { limit = { has_deposit = d_pe_auto_$JOB$_2 } root = { set_variable = { which = pe_alvl value = 2 } } }
	root = {
		# Automated value per month: jobs x 25% x level x value added per job.
		set_variable = { which = pe_aval value = pe_acap }
		multiply_variable = { which = pe_aval value = 0.0025 }
		multiply_variable = { which = pe_aval value = pe_alvl }
		multiply_variable = { which = pe_aval value = pe_vaj_pe_private_$JOB$ }
		# Energy upkeep per month: 4 per automated job = jobs x 0.01 x level.
		set_variable = { which = pe_aen value = pe_acap }
		multiply_variable = { which = pe_aen value = 0.01 }
		multiply_variable = { which = pe_aen value = pe_alvl }
		set_variable = { which = pe_aup value = pe_aen }
		multiply_variable = { which = pe_aup value = pe_s_energy }
		set_variable = { which = pe_anet value = pe_aval }
		subtract_variable = { which = pe_anet value = pe_aup }
		multiply_variable = { which = pe_anet value = 3 }
		set_variable = { which = pe_aloss value = pe_anet }
		multiply_variable = { which = pe_aloss value = -1 }
		# A loss is carried by the firms' capital first, then by the investor pool.
		set_variable = { which = pe_afloss value = pe_aloss }
		if = { limit = { check_variable = { which = pe_afloss value > pe_apool } } set_variable = { which = pe_afloss value = pe_apool } }
		set_variable = { which = pe_aploss value = pe_aloss }
		subtract_variable = { which = pe_aploss value = pe_afloss }
		set_variable = { which = pe_acover value = pe_apool }
		change_variable = { which = pe_acover value = pe_capital }
	}
	if = {
		limit = { root = { check_variable = { which = pe_alvl value > 0 } } }
		if = {
			limit = {
				root = {
					OR = {
						check_variable = { which = pe_anet value >= 0 }
						check_variable = { which = pe_acover value >= pe_aloss }
					}
				}
			}
			# The firms reimburse the state's energy upkeep of this quarter.
			root = {
				set_variable = { which = pe_tmp value = pe_aen }
				multiply_variable = { which = pe_tmp value = 3 }
				add_resource = { energy = 1 mult = pe_tmp }
			}
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
				if = {
					limit = { root = { check_variable = { which = pe_anet value >= 0 } } }
					set_variable = { which = pe_tmp value = root.pe_anet }
					multiply_variable = { which = pe_tmp value = pe_wf_$JOB$ }
					divide_variable = { which = pe_tmp value = root.pe_acap }
				}
				else_if = {
					limit = { root = { check_variable = { which = pe_apool value > 0 } } }
					set_variable = { which = pe_tmp value = root.pe_afloss }
					multiply_variable = { which = pe_tmp value = pe_fcap }
					divide_variable = { which = pe_tmp value = root.pe_apool }
					multiply_variable = { which = pe_tmp value = -1 }
				}
				else = { set_variable = { which = pe_tmp value = 0 } }
				change_variable = { which = pe_fcap value = pe_tmp }
			}
			if = {
				limit = { root = { check_variable = { which = pe_anet value < 0 } } }
				root = { subtract_variable = { which = pe_capital value = pe_aploss } }
			}
		}
		else = {
			root = { set_variable = { which = pe_adown value = 1 } }
		}
		if = {
			limit = { root = { check_variable = { which = pe_unemp value = 1 } } }
			change_variable = { which = pe_idleq_$JOB$ value = 1 }
		}
		else = { set_variable = { which = pe_idleq_$JOB$ value = 0 } }
		if = {
			limit = { check_variable = { which = pe_idleq_$JOB$ value >= 2 } }
			set_variable = { which = pe_idleq_$JOB$ value = 0 }
			root = { set_variable = { which = pe_adown value = 1 } }
		}
	}
	# Drop one level.
	if = {
		limit = { root = { check_variable = { which = pe_adown value = 1 } } }
		save_event_target_as = pe_msg_planet
		if = { limit = { has_deposit = d_pe_auto_$JOB$_2 } remove_deposit = d_pe_auto_$JOB$_2 add_deposit = d_pe_auto_$JOB$_1 }
		else = { remove_deposit = d_pe_auto_$JOB$_1 }
		check_planet_employment = yes
		owner = {
			create_message = {
				type = pe_private_economy
				localization = pe_msg_autodown_$JOB$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			}
		}
	}
}

# numbers). Only for unfilled private jobs, without idle civilians, from 200 private jobs
# of the type, if the automated value covers the energy upkeep by 25%.
pe_auto_invest = {
	# Unfilled private jobs of this type here (automation only fills empty jobs).
	export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_$JOB$ } variable = pe_afree }
	if = { limit = { NOT = { is_variable_set = pe_afree } } set_variable = { which = pe_afree value = 0 } }
	root = { set_variable = { which = pe_afree value = prev.pe_afree } }
	root = {
		set_variable = { which = pe_anext value = pe_acap }
		multiply_variable = { which = pe_anext value = 0.0025 }
		set_variable = { which = pe_tmp value = pe_alvl }
		change_variable = { which = pe_tmp value = 1 }
		multiply_variable = { which = pe_anext value = pe_tmp }
		multiply_variable = { which = pe_anext value = pe_vaj_pe_private_$JOB$ }
		set_variable = { which = pe_acost value = 600 }
		if = { limit = { check_variable = { which = pe_alvl value = 1 } } set_variable = { which = pe_acost value = 800 } }
		# Upkeep of the next level: 4 energy per automated job.
		set_variable = { which = pe_aupn value = pe_acap }
		multiply_variable = { which = pe_aupn value = 0.01 }
		multiply_variable = { which = pe_aupn value = pe_tmp }
		multiply_variable = { which = pe_aupn value = pe_s_energy }
		# Safety margin: the value must cover the upkeep by 25%.
		multiply_variable = { which = pe_aupn value = 1.25 }
		# The firms pay what they can (by capital share), the investor pool the rest.
		set_variable = { which = pe_atot value = pe_apool }
		change_variable = { which = pe_atot value = pe_capital }
		set_variable = { which = pe_afirm value = pe_acost }
		if = { limit = { check_variable = { which = pe_afirm value > pe_apool } } set_variable = { which = pe_afirm value = pe_apool } }
		set_variable = { which = pe_arest value = pe_acost }
		subtract_variable = { which = pe_arest value = pe_afirm }
	}
	if = {
		limit = {
			root = {
				check_variable = { which = pe_unemp value = 0 }
				check_variable = { which = pe_adown value = 0 }
				check_variable = { which = pe_afree value > 0 }
				check_variable = { which = pe_acap value >= 200 }
				check_variable = { which = pe_atot value >= pe_acost }
				check_variable = { which = pe_anext value >= pe_aupn }
			}
			OR = {
				AND = {
					root = { check_variable = { which = pe_alvl value = 0 } }
					owner = { has_technology = tech_assembly_pattern }
				}
				AND = {
					root = { check_variable = { which = pe_alvl value = 1 } }
					owner = { has_technology = tech_construction_templates }
				}
			}
		}
		if = {
			limit = { root = { check_variable = { which = pe_apool value > 0 } } }
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_$JOB$ value > 0 } }
				set_variable = { which = pe_tmp value = root.pe_afirm }
				multiply_variable = { which = pe_tmp value = pe_fcap }
				divide_variable = { which = pe_tmp value = root.pe_apool }
				subtract_variable = { which = pe_fcap value = pe_tmp }
			}
		}
		root = { subtract_variable = { which = pe_capital value = pe_arest } }
		save_event_target_as = pe_msg_planet
		if = {
			limit = { root = { check_variable = { which = pe_alvl value = 0 } } }
			add_deposit = d_pe_auto_$JOB$_1
			owner = { create_message = { type = pe_private_economy localization = pe_msg_auto_$JOB$_1 days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } } }
		}
		else = {
			remove_deposit = d_pe_auto_$JOB$_1
			add_deposit = d_pe_auto_$JOB$_2
			owner = { create_message = { type = pe_private_economy localization = pe_msg_auto_$JOB$_2 days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } } }
		}
		check_planet_employment = yes
	}
}

pe_consider_variant = {
	if = {
		limit = { pe_can_$MODE$_$V$ = yes pe_ok_$V$ = yes }
		root = {
			set_variable = { which = pe_cand value = pe_val_$V$ }
			set_variable = { which = pe_tmp value = pe_ent_val }
			multiply_variable = { which = pe_tmp value = $ENT$ }
			change_variable = { which = pe_cand value = pe_tmp }
			multiply_variable = { which = pe_cand value = pe_spec_$PRIM$ }
			# Pseudo-random 0.8-1.2: the engine's random value repeats within a day, so a
			# small linear congruential sequence (mod 1009) varies it per candidate.
			multiply_variable = { which = pe_seed value = 17 }
			change_variable = { which = pe_seed value = $CODE$ }
			change_variable = { which = pe_seed value = 23 }
			modulo_variable = { which = pe_seed value = 1009 }
			set_variable = { which = pe_rnd value = pe_seed }
			divide_variable = { which = pe_rnd value = 2522.5 }
			change_variable = { which = pe_rnd value = 0.8 }
			multiply_variable = { which = pe_cand value = pe_rnd }
			if = {
				limit = { check_variable = { which = pe_dmode value = 1 } }
				set_variable = { which = pe_tmp value = pe_dval_$DCLS$ }
				multiply_variable = { which = pe_tmp value = 0.875 }
				subtract_variable = { which = pe_cand value = pe_tmp }
			}
			if = {
				limit = { check_variable = { which = pe_cand value > pe_best } }
				set_variable = { which = pe_best value = pe_cand }
				set_variable = { which = pe_best_variant value = $CODE$ }
			}
		}
	}
}

# The investor pool starts a construction site on a free slot. It costs what the district
# would cost the state: the pool pays 40% down, the firm owes the rest (pe_debt) and pays
# it back from its profit. The firm starts with 10 capital, enough to wait about three
# years for workers.
pe_found_firm = {
	root = {
		set_variable = { which = pe_down value = $COST$ }
		multiply_variable = { which = pe_down value = 0.4 }
		set_variable = { which = pe_loan value = $COST$ }
		subtract_variable = { which = pe_loan value = pe_down }
	}
	if = {
		limit = { root = { check_variable = { which = pe_capital value >= pe_down } } }
		add_deposit = $SITE$
		last_added_deposit = {
			set_variable = { which = pe_target value = $CODE$ }
			set_variable = { which = pe_tstage value = 1 }
			set_variable = { which = pe_fcap value = 10 }
			set_variable = { which = pe_debt value = root.pe_loan }
			set_variable = { which = pe_left value = $MONTHS$ }
			set_variable = { which = pe_kind value = 1 }
		}
		set_planet_flag = pe_site
		root = { subtract_variable = { which = pe_capital value = pe_down } }
	}
}

# The investor pool buys a state district, which becomes the firm at once (no construction
# time). The state gets the full price; the pool pays 40% down, the firm owes the rest. Then
# a year's pause on the planet. Only if the
# district really went away (pe_before > pe_after); otherwise nothing happens.
pe_buyout_execute = {
	set_variable = { which = pe_before value = 0 }
	set_variable = { which = pe_after value = 0 }
	owner = {
		set_variable = { which = pe_down value = $PRICE$ }
		multiply_variable = { which = pe_down value = 0.4 }
	}
	if = {
		limit = {
			has_district = $DISTRICT$
			owner = { check_variable = { which = pe_capital value >= pe_down } }
		}
		export_trigger_value_to_variable = { trigger = num_districts parameters = { type = $DISTRICT$ } variable = pe_before }
		remove_district = $DISTRICT$
		export_trigger_value_to_variable = { trigger = num_districts parameters = { type = $DISTRICT$ } variable = pe_after }
		if = { limit = { NOT = { is_variable_set = pe_after } } set_variable = { which = pe_after value = 0 } }
	}
	if = {
		limit = { check_variable = { which = pe_before value > pe_after } }
		add_deposit = $FIRM$
		last_added_deposit = {
			set_variable = { which = pe_code value = $CODE$ }
			set_variable = { which = pe_stage value = 1 }
			set_variable = { which = pe_fcap value = 10 }
			set_variable = { which = pe_debt value = $PRICE$ }
			multiply_variable = { which = pe_debt value = 0.6 }
		}
		set_timed_planet_flag = { flag = pe_cooldown days = 360 }
		check_planet_employment = yes
		save_event_target_as = pe_msg_planet
		owner = {
			add_resource = { minerals = $MINERALS$ trade = $TRADE$ }
			subtract_variable = { which = pe_capital value = pe_down }
			create_message = {
				type = pe_private_economy
				localization = pe_msg_buy_$FIRM$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			}
		}
	}
}

			}
		}
	}
}
