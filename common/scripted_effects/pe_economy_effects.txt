# Private economy logic. Runs quarterly per country (pe.100).
# Call depth stays at 4 or less: the engine initialises only 5 nested levels.
#   pe_quarterly (1) -> pe_capital_income (2) -> pe_count_job (3)
#                    -> pe_scores (2) -> pe_score_resource (3)
#                    -> pe_compute_values (2)
#                    -> pe_planet_step (2) -> pe_planet_counts (3)
#                                          -> pe_choose_variant (3)  -> pe_consider_variant (4)
#                                          -> pe_found_variant (3)   -> pe_found_firm (4)
#                                          -> pe_buy_or_offer (3)    -> pe_buyout_execute (4)
#                                          -> pe_nationalise_one (3) -> pe_nat_consider (4)
#                                                                    -> pe_nationalise_firm (4)
#                                          -> pe_restructure (3)     -> pe_retool_consider (4)
#                                                                    -> pe_retool_gain (4)
# Variant lists and dispatchers: pe_firm_effects_generated.txt (tools/generate-firms.mjs).
#
# Private capital (country variable pe_capital): starts at 1000. Each private job pays
# 0.5 energy upkeep per month as profit, which flows into the capital. Founding costs
# 500 (city) / 300 (rural); a buyout costs the price paid to the state (600 / 360);
# restructuring costs half the founding cost; nationalisation pays the price back in.

# --- Country level ------------------------------------------------------------

pe_init = {
	if = {
		limit = { NOT = { is_variable_set = pe_private_share } }
		set_variable = { which = pe_private_share value = 0 }
	}
	if = {
		limit = { NOT = { is_variable_set = pe_capital } }
		set_variable = { which = pe_capital value = 1000 }
	}
}

pe_quarterly = {
	pe_init = yes
	pe_capital_income = yes
	change_variable = { which = pe_capital value = pe_capital_income }
	pe_scores = yes
	pe_compute_values = yes
	every_owned_colony = {
		limit = { pe_planet_eligible = yes }
		pe_planet_step = yes
	}
}

pe_scores = {
	pe_score_resource = { RES = alloys MARKET = yes ALWAYS = yes }
	pe_score_resource = { RES = consumer_goods MARKET = yes }
	pe_score_resource = { RES = minerals MARKET = yes }
	pe_score_resource = { RES = energy MARKET = yes }
	pe_score_resource = { RES = food MARKET = yes }
	pe_score_resource = { RES = trade }
}

# Country scope: profit of the last quarter = filled private jobs x 0.5 energy x 3
# months (100 workforce = one job).
pe_capital_income = {
	set_variable = { which = pe_jobs value = 0 }
	pe_count_job = { JOB = pe_private_metallurgist }
	pe_count_job = { JOB = pe_private_artisan }
	pe_count_job = { JOB = pe_private_trader }
	pe_count_job = { JOB = pe_private_miner }
	pe_count_job = { JOB = pe_private_technician }
	pe_count_job = { JOB = pe_private_farmer }
	set_variable = { which = pe_capital_income value = pe_jobs }
	multiply_variable = { which = pe_capital_income value = 0.015 }
}

pe_count_job = {
	export_trigger_value_to_variable = { trigger = num_assigned_jobs parameters = { job = $JOB$ } variable = pe_tmp }
	change_variable = { which = pe_jobs value = pe_tmp }
}

# Country scope. pe_score_<res>: how urgently the economy needs more of a resource.
#   deficit:  50 + deficit in % of gross production, +30 if the stock lasts < 12 months
#   surplus:  25 - 0.75 x surplus in % of production (a small surplus still invites
#             investment; zero at a 33% surplus)
#   stock at 90% of its maximum: -50 (not for ALWAYS resources)
#   ALWAYS (alloys): +40, always needed
#   MARKET: +1 per 1% the market price is above the first price this empire saw
pe_score_resource = {
	export_resource_income_to_variable = { resource = $RES$ variable = pe_net }
	export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = $RES$ } variable = pe_gross }
	export_resource_stockpile_to_variable = { resource = $RES$ variable = pe_stock }
	export_resource_maximum_to_variable = { resource = $RES$ variable = pe_max }
	set_variable = { which = pe_score_$RES$ value = 0 }
	set_variable = { which = pe_scale value = pe_gross }
	if = {
		limit = { check_variable = { which = pe_scale value < 10 } }
		set_variable = { which = pe_scale value = 10 }
	}
	# Net relative to production, in percent.
	set_variable = { which = pe_rel value = pe_net }
	divide_variable = { which = pe_rel value = pe_scale }
	multiply_variable = { which = pe_rel value = 100 }
	if = {
		limit = { check_variable = { which = pe_net value < 0 } }
		change_variable = { which = pe_score_$RES$ value = 50 }
		subtract_variable = { which = pe_score_$RES$ value = pe_rel }
		set_variable = { which = pe_runway value = pe_net }
		multiply_variable = { which = pe_runway value = -12 }
		if = {
			limit = { check_variable = { which = pe_stock value < pe_runway } }
			change_variable = { which = pe_score_$RES$ value = 30 }
		}
	}
	else = {
		change_variable = { which = pe_score_$RES$ value = 25 }
		multiply_variable = { which = pe_rel value = 0.75 }
		subtract_variable = { which = pe_score_$RES$ value = pe_rel }
	}
	[[ALWAYS]
	change_variable = { which = pe_score_$RES$ value = 40 }
	]
	[[!ALWAYS]
	if = {
		limit = { check_variable = { which = pe_max value > 0 } }
		set_variable = { which = pe_full value = pe_max }
		multiply_variable = { which = pe_full value = 0.9 }
		if = {
			limit = { check_variable = { which = pe_stock value >= pe_full } }
			change_variable = { which = pe_score_$RES$ value = -50 }
		}
	}
	]
	[[MARKET]
	export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = $RES$ amount = 1 trade_type = market_buy } variable = pe_price }
	if = {
		limit = {
			check_variable = { which = pe_price value > 0 }
			NOT = { is_variable_set = pe_price_ref_$RES$ }
		}
		set_variable = { which = pe_price_ref_$RES$ value = pe_price }
	}
	if = {
		limit = {
			check_variable = { which = pe_price value > 0 }
			check_variable = { which = pe_price_ref_$RES$ value > 0 }
		}
		set_variable = { which = pe_ratio value = pe_price }
		divide_variable = { which = pe_ratio value = pe_price_ref_$RES$ }
		change_variable = { which = pe_ratio value = -1 }
		multiply_variable = { which = pe_ratio value = 100 }
		change_variable = { which = pe_score_$RES$ value = pe_ratio }
	}
	]
}

# --- Planet level (colony scope, ROOT = owner) --------------------------------

# District slots, firms, and how many firms the economic order allows.
pe_planet_counts = {
	export_trigger_value_to_variable = { trigger = num_districts parameters = { type = any } variable = pe_districts }
	export_trigger_value_to_variable = { trigger = free_district_slots variable = pe_free }
	export_trigger_value_to_variable = { trigger = count_deposits parameters = { category = pe_deposit_cat_firms } variable = pe_firms }
	set_variable = { which = pe_target value = pe_districts }
	change_variable = { which = pe_target value = pe_free }
	change_variable = { which = pe_target value = pe_firms }
	multiply_variable = { which = pe_target value = root.pe_private_share }
	divide_variable = { which = pe_target value = 100 }
	floor_variable = pe_target
}

pe_planet_step = {
	pe_planet_counts = yes
	if = {
		limit = { check_variable = { which = pe_firms value < pe_target } }
		if = {
			limit = { free_district_slots > 0 }
			pe_choose_variant = { MODE = found }
			pe_found_variant = yes
		}
		else_if = {
			limit = { owner = { pe_buyout_allowed = yes } }
			pe_choose_variant = { MODE = buy }
			pe_buy_or_offer = yes
		}
	}
	else_if = {
		limit = { check_variable = { which = pe_firms value > pe_target } }
		pe_nationalise_one = yes
	}
	# No action this year, and no room or no need to grow: firms may restructure.
	if = {
		limit = {
			NOT = { has_planet_flag = pe_cooldown }
			check_variable = { which = pe_firms value > 0 }
			OR = {
				check_variable = { which = pe_firms value >= pe_target }
				free_district_slots = 0
			}
		}
		pe_restructure = yes
	}
}

pe_consider_variant = {
	if = {
		limit = { pe_can_$MODE$_$V$ = yes }
		root = {
			if = {
				limit = { check_variable = { which = pe_val_$V$ value > pe_best } }
				set_variable = { which = pe_best value = pe_val_$V$ }
				set_variable = { which = pe_best_variant value = $CODE$ }
			}
		}
	}
}

pe_found_firm = {
	if = {
		limit = { owner = { check_variable = { which = pe_capital value >= $COST$ } } }
		add_deposit = $FIRM$
		set_timed_planet_flag = { flag = pe_cooldown days = 360 }
		save_event_target_as = pe_msg_planet
		owner = {
			change_variable = { which = pe_capital value = -$COST$ }
			create_message = {
				type = pe_private_economy
				localization = pe_msg_found_$FIRM$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			}
		}
	}
}

# The firm takes over a state district and pays the price from the private capital.
pe_buyout_execute = {
	if = {
		limit = {
			has_district = $DISTRICT$
			owner = { check_variable = { which = pe_capital value >= $PRICE$ } }
		}
		remove_district = $DISTRICT$
		add_deposit = $FIRM$
		set_timed_planet_flag = { flag = pe_cooldown days = 360 }
		save_event_target_as = pe_msg_planet
		owner = {
			add_resource = { minerals = $MINERALS$ trade = $TRADE$ }
			change_variable = { which = pe_capital value = -$PRICE$ }
			create_message = {
				type = pe_private_economy
				localization = pe_msg_buy_$FIRM$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			}
		}
	}
}

pe_nat_consider = {
	if = {
		limit = { has_deposit = $V$ }
		root = {
			if = {
				limit = { check_variable = { which = pe_val_$V$ value < pe_worst } }
				set_variable = { which = pe_worst value = pe_val_$V$ }
				set_variable = { which = pe_worst_variant value = $CODE$ }
			}
		}
	}
}

# The state buys a firm back at the buyout price (if it can pay) and rebuilds the district.
pe_nationalise_firm = {
	if = {
		limit = {
			owner = {
				has_resource = { type = minerals amount >= $MINERALS$ }
				has_resource = { type = trade amount >= $TRADE$ }
			}
		}
		remove_deposit = $FIRM$
		add_district = $DISTRICT$
		set_timed_planet_flag = { flag = pe_cooldown days = 360 }
		save_event_target_as = pe_msg_planet
		owner = {
			add_resource = { minerals = -$MINERALS$ trade = -$TRADE$ }
			change_variable = { which = pe_capital value = $PRICE$ }
			create_message = {
				type = pe_private_economy
				localization = pe_msg_nat_$FIRM$
				days = 30
				target = event_target:pe_msg_planet
				variable = { type = name localization = PLANET scope = event_target:pe_msg_planet }
			}
		}
	}
}

# Best variant per rural class on this planet (the side share depends on its deposits).
pe_retool_consider = {
	if = {
		limit = { pe_can_retool_$V$ = yes }
		root = {
			if = {
				limit = { check_variable = { which = pe_val_$V$ value > pe_cls_val_$CLS$ } }
				set_variable = { which = pe_cls_val_$CLS$ value = pe_val_$V$ }
				set_variable = { which = pe_cls_var_$CLS$ value = $CODE$ }
			}
		}
	}
}

# Gain if this firm switched to the best variant of its class.
pe_retool_gain = {
	if = {
		limit = { has_deposit = $V$ }
		root = {
			set_variable = { which = pe_tmp value = pe_cls_val_$CLS$ }
			subtract_variable = { which = pe_tmp value = pe_val_$V$ }
			if = {
				limit = { check_variable = { which = pe_tmp value > pe_best_gain } }
				set_variable = { which = pe_best_gain value = pe_tmp }
				set_variable = { which = pe_from value = $CODE$ }
				set_variable = { which = pe_to value = pe_cls_var_$CLS$ }
			}
		}
	}
}
