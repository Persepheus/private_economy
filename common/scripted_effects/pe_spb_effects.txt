# Specialisation buildings of corporations (0.45). A corporation with a firm of a rural type on a
# planet builds the matching building of the specialisation there (mineral purification plant,
# energy grid, food processing facility), with the same technology the state needs:
#   1 into a zone of the type with a free building slot,
#   2 else into the state's district of the type, adding the zone (1000 minerals),
#   3 else onto a free district slot of the type, adding the district (300 minerals) and zone.
# The corporation pays everything from its cash (minerals at the market buying price, pe_pb_minerals,
# since 0.45.4; before at their shadow price, which a mineral surplus pushes far down) and leases
# it to the state for ten years: the cost plus 50% profit over 120 months (0.45.1, budget line
# d_pe_ledger_lease), then the building is ours; the planet decision Pay Off Lease ends it
# earlier at its remaining value. So it is always worth it for the corporation; it only needs
# a firm of the type there that benefits. Among the candidates of an empire the one whose
# private jobs gain most against the cost comes first. Its corporation dissolved, the building gone or
# the planet lost: the lease ends. The player is always asked (pe.333, also under
# Laissez-faire, because of the lease); AI empires accept. One building per empire and
# quarter; not on habitats or other special planets.
# A building of the type also adds its bonus to the value added of the firms here
# (pe_spb_va), so it shows in their profit.
# Call depths: pe_quarterly (1) -> pe_spb_step (2) -> pe_spb_eval_<t> (3), pe_spb_build (3)
#   -> pe_corp_msg (4); pe.100 -> pe_spb_month (1) -> pe_spb_month_<t> (2) -> pe_corp_msg (3),
#   pe_ledger_bits (2); pe_firm_accounting (3) -> pe_spb_va (4); pe.131 -> pe_spb_clear (1);
#   decision -> pe_spb_bb_calc (1); pe.333 -> pe_spb_build (1); pe.334 -> pe_spb_bb_pay (1).

# Country scope, quarterly.
pe_spb_step = {
	if = {
		limit = {
			NOT = { has_country_flag = pe_spb_offer }
			NOT = { has_policy_flag = pe_order_planned }
			OR = {
				check_variable = { which = pe_c1_on value = 1 }
				check_variable = { which = pe_c2_on value = 1 }
				check_variable = { which = pe_c3_on value = 1 }
				check_variable = { which = pe_c4_on value = 1 }
				check_variable = { which = pe_c5_on value = 1 }
				check_variable = { which = pe_c6_on value = 1 }
				check_variable = { which = pe_c7_on value = 1 }
				check_variable = { which = pe_c8_on value = 1 }
			}
		}
		set_variable = { which = pe_spb_best value = -1000000 }
		set_variable = { which = pe_spb_t value = 0 }
		set_variable = { which = pe_spb_pick value = 0 }
		every_owned_colony = {
			limit = {
				uses_district_set = standard
				NOT = { has_modifier = pe_warzone }
				count_deposits = { category = pe_deposit_cat_firms count > 0 }
			}
			pe_spb_eval_m = yes
			pe_spb_eval_e = yes
			pe_spb_eval_f = yes
		}
		if = {
			limit = { check_variable = { which = pe_spb_t value > 0 } }
			# The lease for the offer (pe_spb_build checks again).
			set_variable = { which = pe_spb_rent value = pe_spb_cc }
			multiply_variable = { which = pe_spb_rent value = 0.025 }
			ceiling_variable = pe_spb_rent
			multiply_variable = { which = pe_spb_rent value = 0.5 }
			# The cost, rounded for the offer.
			set_variable = { which = pe_spb_ccr value = pe_spb_cc }
			round_variable = pe_spb_ccr
			if = {
				limit = { is_ai = yes }
				event_target:pe_spb_planet = { pe_spb_build = yes }
			}
			else = {
				set_timed_country_flag = { flag = pe_spb_offer days = 360 }
				country_event = { id = pe.333 }
			}
		}
	}
}

# root.pe_spb_* (pe_spb_t = 1) and event_target:pe_spb_planet.

		limit = {
			owner = { has_technology = tech_mineral_purification_1 }
			NOT = { OR = { has_building = building_mineral_purification_plant has_building = building_mineral_purification_hub } }
			NOT = { has_building_construction = building_mineral_purification_plant }

			any_deposit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_cls value = 2 } check_variable = { which = pe_corp value > 0 } }
		}
		# Private workforce with the building's job here (pe_wf_miner, set by the accounts).
		root = { set_variable = { which = pe_spb_wf value = 0 } set_variable = { which = pe_spb_lc value = 0 } }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_wf_miner value > 0 } }
			root = { change_variable = { which = pe_spb_wf value = prev.pe_wf_miner } }
		}
		# Where it would go (see the top of this file).
		if = { limit = { OR = { num_zones = { type = zone_minerals value > 0 } num_zones = { type = zone_betharian value > 0 } num_zones = { type = zone_rare_crystals value > 0 } num_zones = { type = zone_minerals_physics value > 0 } } free_building_slots > 0 } root = { set_variable = { which = pe_spb_lc value = 1 } } }

		# The corporation: one with a firm of the type here, the one with the most cash.
		root = { set_variable = { which = pe_spb_kc value = -1 } set_variable = { which = pe_spb_kx value = 0 } }
		every_deposit = {
			limit = { has_deposit_category = pe_deposit_cat_firms check_variable = { which = pe_cls value = 2 } check_variable = { which = pe_corp value > 0 } }
			if = { limit = { check_variable = { which = pe_corp value = 1 } root = { check_variable = { which = pe_c1_on value = 1 } check_variable = { which = pe_c1_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c1_cash } set_variable = { which = pe_spb_kx value = 1 } } }
			if = { limit = { check_variable = { which = pe_corp value = 2 } root = { check_variable = { which = pe_c2_on value = 1 } check_variable = { which = pe_c2_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c2_cash } set_variable = { which = pe_spb_kx value = 2 } } }
			if = { limit = { check_variable = { which = pe_corp value = 3 } root = { check_variable = { which = pe_c3_on value = 1 } check_variable = { which = pe_c3_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c3_cash } set_variable = { which = pe_spb_kx value = 3 } } }
			if = { limit = { check_variable = { which = pe_corp value = 4 } root = { check_variable = { which = pe_c4_on value = 1 } check_variable = { which = pe_c4_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c4_cash } set_variable = { which = pe_spb_kx value = 4 } } }
			if = { limit = { check_variable = { which = pe_corp value = 5 } root = { check_variable = { which = pe_c5_on value = 1 } check_variable = { which = pe_c5_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c5_cash } set_variable = { which = pe_spb_kx value = 5 } } }
			if = { limit = { check_variable = { which = pe_corp value = 6 } root = { check_variable = { which = pe_c6_on value = 1 } check_variable = { which = pe_c6_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c6_cash } set_variable = { which = pe_spb_kx value = 6 } } }
			if = { limit = { check_variable = { which = pe_corp value = 7 } root = { check_variable = { which = pe_c7_on value = 1 } check_variable = { which = pe_c7_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c7_cash } set_variable = { which = pe_spb_kx value = 7 } } }
			if = { limit = { check_variable = { which = pe_corp value = 8 } root = { check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c8_cash } set_variable = { which = pe_spb_kx value = 8 } } }
		}
		root = {
			# Ten years of the extra output (+0.5 per job of 100 workforce and month; pe_spb_v)
			# against the cost (building, zone, district: pe_spb_xt). The output at shadow prices,
			# the cost at the market buying price. The corporation must afford the whole cost.
			set_variable = { which = pe_spb_v value = pe_spb_wf }
			multiply_variable = { which = pe_spb_v value = 0.6 }

			multiply_variable = { which = pe_spb_xt value = pe_pb_minerals }
			set_variable = { which = pe_spb_md value = 0 }
			if = {
				limit = { check_variable = { which = pe_spb_lc value > 0 } check_variable = { which = pe_spb_kx value > 0 } check_variable = { which = pe_spb_wf value > 0 } check_variable = { which = pe_spb_kc value >= pe_spb_xt } }
				set_variable = { which = pe_spb_md value = 1 }
				set_variable = { which = pe_spb_xc value = pe_spb_xt }
			}
			if = {
				limit = { check_variable = { which = pe_spb_md value > 0 } }
				set_variable = { which = pe_tmp value = pe_spb_v }
				subtract_variable = { which = pe_tmp value = pe_spb_xc }
				if = {
					limit = { check_variable = { which = pe_tmp value > pe_spb_best } }
					set_variable = { which = pe_spb_best value = pe_tmp }
					set_variable = { which = pe_spb_t value = 1 }
					set_variable = { which = pe_spb_lv value = pe_spb_lc }
					set_variable = { which = pe_spb_k value = pe_spb_kx }
					set_variable = { which = pe_spb_cc value = pe_spb_xc }
					set_variable = { which = pe_spb_pick value = 1 }
				}
			}
		}
		if = {
			limit = { root = { check_variable = { which = pe_spb_pick value = 1 } } }
			save_event_target_as = pe_spb_planet
			root = { set_variable = { which = pe_spb_pick value = 0 } }
		}
	}
}

			if = { limit = { check_variable = { which = pe_corp value = 8 } root = { check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_cash value > pe_spb_kc } } } root = { set_variable = { which = pe_spb_kc value = pe_c8_cash } set_variable = { which = pe_spb_kx value = 8 } } }
		}
		root = {
			# Ten years of the extra output (+0.5 per job of 100 workforce and month; pe_spb_v)
			# against the cost (building, zone, district: pe_spb_xt). The output at shadow prices,
			# the cost at the market buying price. The corporation must afford the whole cost.
			set_variable = { which = pe_spb_v value = pe_spb_wf }
			multiply_variable = { which = pe_spb_v value = 0.6 }
			multiply_variable = { which = pe_spb_v value = pe_s_food }
			set_variable = { which = pe_spb_xt value = 400 }
			if = { limit = { check_variable = { which = pe_spb_lc value >= 2 } } change_variable = { which = pe_spb_xt value = 1000 } }
			if = { limit = { check_variable = { which = pe_spb_lc value = 3 } } change_variable = { which = pe_spb_xt value = 300 } }
			multiply_variable = { which = pe_spb_xt value = pe_pb_minerals }
			set_variable = { which = pe_spb_md value = 0 }
			if = {
				limit = { check_variable = { which = pe_spb_lc value > 0 } check_variable = { which = pe_spb_kx value > 0 } check_variable = { which = pe_spb_wf value > 0 } check_variable = { which = pe_spb_kc value >= pe_spb_xt } }
				set_variable = { which = pe_spb_md value = 1 }
				set_variable = { which = pe_spb_xc value = pe_spb_xt }
			}
			if = {
				limit = { check_variable = { which = pe_spb_md value > 0 } }
				set_variable = { which = pe_tmp value = pe_spb_v }
				subtract_variable = { which = pe_tmp value = pe_spb_xc }
				if = {
					limit = { check_variable = { which = pe_tmp value > pe_spb_best } }
					set_variable = { which = pe_spb_best value = pe_tmp }
					set_variable = { which = pe_spb_t value = 3 }
					set_variable = { which = pe_spb_lv value = pe_spb_lc }
					set_variable = { which = pe_spb_k value = pe_spb_kx }
					set_variable = { which = pe_spb_cc value = pe_spb_xc }
					set_variable = { which = pe_spb_pick value = 1 }
				}
			}
		}
		if = {
			limit = { root = { check_variable = { which = pe_spb_pick value = 1 } } }
			save_event_target_as = pe_spb_planet
			root = { set_variable = { which = pe_spb_pick value = 0 } }
		}
	}
}

# Colony scope (decision Pay Off Lease): what is left of the leases here, cost x months left /
# 120 -> the owner's pe_spb_bb, rounded up; pe_spb_bb_ok = 1 if we have that much trade.
pe_spb_bb_calc = {
	owner = { set_variable = { which = pe_spb_bb value = 0 } }
	if = {
		limit = { check_variable = { which = pe_spb_m_m value > 0 } }
		set_variable = { which = pe_tmp value = pe_spb_m_c }
		multiply_variable = { which = pe_tmp value = pe_spb_m_m }
		divide_variable = { which = pe_tmp value = 120 }
		owner = { change_variable = { which = pe_spb_bb value = prev.pe_tmp } }
	}
	if = {
		limit = { check_variable = { which = pe_spb_e_m value > 0 } }
		set_variable = { which = pe_tmp value = pe_spb_e_c }
		multiply_variable = { which = pe_tmp value = pe_spb_e_m }
		divide_variable = { which = pe_tmp value = 120 }
		owner = { change_variable = { which = pe_spb_bb value = prev.pe_tmp } }
	}
	if = {
		limit = { check_variable = { which = pe_spb_f_m value > 0 } }
		set_variable = { which = pe_tmp value = pe_spb_f_c }
		multiply_variable = { which = pe_tmp value = pe_spb_f_m }
		divide_variable = { which = pe_tmp value = 120 }
		owner = { change_variable = { which = pe_spb_bb value = prev.pe_tmp } }
	}
	owner = {
		ceiling_variable = pe_spb_bb
		set_variable = { which = pe_spb_bb_ok value = 0 }
		export_resource_stockpile_to_variable = { resource = trade variable = pe_tmp }
		if = { limit = { check_variable = { which = pe_tmp value >= pe_spb_bb } } set_variable = { which = pe_spb_bb_ok value = 1 } }
	}
}
