# Arc furnace prototype (events pe.22-27 in pe_proto_events.txt). Planet scope: the molten
# planet of the holdings' furnace. Like vanilla cybernetics.1000-1015: every planet in the
# system without a research deposit, not a star, not habitable gets the furnace deposit of
# stage $N$ (minerals; from stage III by overclock) and the station bonus of that stage.
pe_proto_af_deposits = {
	solar_system = {
		every_system_planet = {
			limit = {
				NOT = { has_deposit_for = shipclass_research_station }
				is_star = no
				is_astral_scar = no
				colonizable_planet = no
			}
			dismantle_arc_furnace_effect = yes
			remove_trade_deposits = yes
			add_deposit = d_arc_furnace_$N$
			add_modifier = { modifier = orbital_arc_furnace_$N$_mod days = -1 }
			set_planet_flag = pe_proto_af_dep
		}
	}
}

# Country scope: event_target:pe_hold (the Private Holdings of root), if there are any.
pe_proto_af_hold = {
	random_country = {
		limit = { is_country_type = pe_holdings has_relation_flag = { who = root flag = pe_holdings_of } }
		save_event_target_as = pe_hold
	}
}
