# Story events of the private economy.
#   Patrons (pe_patron_events.txt): rich private people spend the investor pool. They are
#   recurring figures (pe_mag_id, reputation pe_mag_rep); a magnate retires after four
#   events and a new one appears.
#   Labour (pe_labour_events.txt): private worker jobs stay empty; firms and workers react.
#   Market (pe_market_events.txt): shares, bonds, takeovers, monopolies, crashes ...
#   Flair (pe_flair_events.txt): supply deals, modernisation, rationalisation, capital
#   levy, and the neighbours (investment abroad, foreign branches, licences, dumping).
#   pe_quarterly (1) -> pe_patron (2) -> pe_patron_fire (3) -> pe_pat_amounts (4)
#                    -> pe_labour (2) -> pe_labour_fire (3) -> pe_pat_amounts (4)
#                    -> pe_deal (2) -> pe_deal_fire (3)
#   pe_patron_fire (3) -> pe_pick_sector (4), pe_pick_free_spec (4)

# --- Patrons ------------------------------------------------------------------------

# Country scope, quarterly: with a full investor pool (4500 or more) and no patron event
# in the last two years, a 25% chance per quarter of a patron event.
pe_patron = {
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 4500 }
			NOT = { has_country_flag = pe_patron_cd }
		}
		random = {
			chance = 25
			pe_patron_fire = yes
		}
	}
}

# Country scope: a patron event. The magnate puts 30% of the investor pool into it

		if = { limit = { check_variable = { which = pe_cnt_t value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_u value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }

		if = { limit = { check_variable = { which = pe_cnt_z value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_b value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_m value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }

				save_event_target_as = pe_pat_country
			}

		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root } } }
			modifier = { factor = 0 NOT = { OR = { check_variable = { which = pe_cnt_a value >= 2 } check_variable = { which = pe_cnt_c value >= 2 } } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.268 }
		}
		# Market and corporate events (pe_market_events.txt).
		6 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_n_corp value > 0 } } }
			country_event = { id = pe.241 }
		}
		6 = { country_event = { id = pe.242 } }
		6 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } } } }
			random_owned_colony = {

# Country scope: the magnate (a new one after four events), the amount (30% of the
# investor pool, taken from it) and what the options give, all rounded.
pe_pat_amounts = {
	if = {
		limit = {
			OR = {
				NOT = { is_variable_set = pe_mag_id }
				check_variable = { which = pe_mag_events value >= 4 }
			}
		}
		# A magnate retires: their farewell (pe.240) a month later.
		if = {
			limit = { is_variable_set = pe_mag_id }
			set_variable = { which = pe_mag_old value = pe_mag_id }
			set_variable = { which = pe_mag_old_rep value = pe_mag_rep }
			country_event = { id = pe.240 days = 30 }
		}
		set_variable_to_random_value = { which = pe_mag_id min = 1 max = 16 rounded = yes }
		set_variable = { which = pe_mag_events value = 0 }
		set_variable = { which = pe_mag_rep value = 0 }
	}
	change_variable = { which = pe_mag_events value = 1 }
	set_variable = { which = pe_pat_spend value = pe_capital }
	multiply_variable = { which = pe_pat_spend value = 0.3 }
	round_variable = pe_pat_spend
	subtract_variable = { which = pe_capital value = pe_pat_spend }
	set_variable = { which = pe_pat_unity value = pe_pat_spend }
	multiply_variable = { which = pe_pat_unity value = 0.3 }
	round_variable = pe_pat_unity
	set_variable = { which = pe_pat_tax value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tax value = 0.25 }
	round_variable = pe_pat_tax
	set_variable = { which = pe_pat_res value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res value = 0.4 }
	round_variable = pe_pat_res
	set_variable = { which = pe_pat_res_free value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res_free value = 0.5 }
	round_variable = pe_pat_res_free
	set_variable = { which = pe_pat_proto_full value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_full value = 0.6 }
	round_variable = pe_pat_proto_full
	set_variable = { which = pe_pat_proto_part value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_part value = 0.2 }
	round_variable = pe_pat_proto_part
	set_variable = { which = pe_pat_art value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art value = 0.2 }
	round_variable = pe_pat_art
	set_variable = { which = pe_pat_art_state value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art_state value = 0.4 }
	round_variable = pe_pat_art_state
	set_variable = { which = pe_pat_tour value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tour value = 0.2 }
	round_variable = pe_pat_tour
	set_variable = { which = pe_pat_half value = pe_pat_spend }
	multiply_variable = { which = pe_pat_half value = 0.5 }
	round_variable = pe_pat_half
	set_variable = { which = pe_pat_influence value = pe_pat_spend }
	multiply_variable = { which = pe_pat_influence value = 0.05 }
	round_variable = pe_pat_influence
	if = { limit = { check_variable = { which = pe_pat_influence value > 150 } } set_variable = { which = pe_pat_influence value = 150 } }
	# Scale of the results that are no amounts (modifiers, deposits, settlers): the spend in
	# units of 1500 (30% of a pool of 5000), between 0.5 and 3, to one decimal.
	set_variable = { which = pe_pat_mult value = pe_pat_spend }
	divide_variable = { which = pe_pat_mult value = 150 }
	round_variable = pe_pat_mult
	divide_variable = { which = pe_pat_mult value = 10 }
	if = { limit = { check_variable = { which = pe_pat_mult value < 0.5 } } set_variable = { which = pe_pat_mult value = 0.5 } }
	if = { limit = { check_variable = { which = pe_pat_mult value > 3 } } set_variable = { which = pe_pat_mult value = 3 } }
	# Follow-ups and small amounts that used to be fixed.
	set_variable = { which = pe_pat_u50 value = pe_pat_mult }
	multiply_variable = { which = pe_pat_u50 value = 50 }
	round_variable = pe_pat_u50
	set_variable = { which = pe_pat_u100 value = pe_pat_mult }
	multiply_variable = { which = pe_pat_u100 value = 100 }
	round_variable = pe_pat_u100
	# Settlers of a private colony (pe.223): 100 x scale.
	set_variable = { which = pe_col_pops value = pe_pat_u100 }
	# Deposits of a prospecting find (pe.214) and size of the hydroponic farms (pe.215):
	# 1, 2 from scale 1.5, 3 from scale 2.5.
	set_variable = { which = pe_find_n value = 1 }
	if = { limit = { check_variable = { which = pe_pat_mult value >= 1.5 } } set_variable = { which = pe_find_n value = 2 } }
	if = { limit = { check_variable = { which = pe_pat_mult value >= 2.5 } } set_variable = { which = pe_find_n value = 3 } }
	set_variable = { which = pe_hydro_food value = pe_find_n }
	multiply_variable = { which = pe_hydro_food value = 15 }
	# Company housing (pe.230): 500 x the same size.
	set_variable = { which = pe_house value = pe_find_n }
	multiply_variable = { which = pe_house value = 500 }
	# Guest workers (pe.211): 200 x scale.
	set_variable = { which = pe_gw_pops value = pe_pat_u100 }
	multiply_variable = { which = pe_gw_pops value = 2 }
	# Leased machines (pe.212): at most twice the base lease (25% -> 50% automated).
	set_variable = { which = pe_lease_mult value = pe_pat_mult }
	if = { limit = { check_variable = { which = pe_lease_mult value > 2 } } set_variable = { which = pe_lease_mult value = 2 } }
	set_variable = { which = pe_pat_mult_half value = pe_pat_mult }
	multiply_variable = { which = pe_pat_mult_half value = 0.5 }
	# Investment abroad (pe.265): per month, taxes for the state and returns for the pool.
	set_variable = { which = pe_abroad_new_state value = pe_pat_spend }
	multiply_variable = { which = pe_abroad_new_state value = 0.004 }
	round_variable = pe_abroad_new_state
	set_variable = { which = pe_abroad_new_pool value = pe_pat_spend }
	multiply_variable = { which = pe_abroad_new_pool value = 0.008 }
	round_variable = pe_abroad_new_pool
	# Technology licence (pe.267): the fee and the research.
	set_variable = { which = pe_lic_cost value = pe_pat_spend }
	multiply_variable = { which = pe_lic_cost value = 0.3 }
	round_variable = pe_lic_cost
	set_variable = { which = pe_lic_res_half value = pe_pat_res }
	multiply_variable = { which = pe_lic_res_half value = 0.5 }
	round_variable = pe_lic_res_half
	# Dumping (pe.268): the tariff in influence.
	set_variable = { which = pe_tariff value = pe_pat_mult }
	multiply_variable = { which = pe_tariff value = 25 }
	round_variable = pe_tariff
}

# --- Labour -------------------------------------------------------------------------

# Colony scope: private worker jobs stand empty here (at least one job's worth).
# (Trigger in pe_triggers.txt: pe_rural_short.)

# Country scope, quarterly: when private worker jobs stand empty somewhere and the pool
# can pay wages (1000 or more), a 25% chance per quarter of a labour event; then two
# years' pause.
pe_labour = {
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 1500 }
			NOT = { has_country_flag = pe_labour_cd }
			any_owned_colony = { pe_planet_eligible = yes pe_rural_short = yes }
		}
		random = {
			chance = 25
			pe_labour_fire = yes
		}
	}
}

# Country scope: a labour event on a planet with empty private worker jobs. pe_lab_job:
# the job with the most empty places there (1 miner, 2 technician, 3 farmer).
pe_labour_fire = {
	set_timed_country_flag = { flag = pe_labour_cd days = 730 }
	pe_pat_amounts = yes
	random_owned_colony = {
		limit = { pe_planet_eligible = yes pe_rural_short = yes }
		save_event_target_as = pe_pat_planet
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_miner } variable = pe_lf_m }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_technician } variable = pe_lf_e }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_farmer } variable = pe_lf_f }
		root = {
			set_variable = { which = pe_lab_job value = 1 }
			set_variable = { which = pe_tmp value = prev.pe_lf_m }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_e } } set_variable = { which = pe_lab_job value = 2 } set_variable = { which = pe_tmp value = prev.pe_lf_e } }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_f } } set_variable = { which = pe_lab_job value = 3 } }
		}
	}
	random_list = {
		30 = { country_event = { id = pe.210 } }
		# Guest workers: an empire we know, whose founder species finds the planet
		# habitable and would not be purged here.
		25 = {
			modifier = {
				factor = 0
				NOT = {
					any_country = {
						is_country_type = default
						NOT = { is_same_value = root }
						has_communications = root
					}
				}
			}
			random_country = {
				limit = {
					is_country_type = default
					NOT = { is_same_value = root }
					has_communications = root
				}
				founder_species = { save_event_target_as = pe_gw_species }
			}
			country_event = { id = pe.211 }
		}
		25 = { country_event = { id = pe.212 } }
		20 = { country_event = { id = pe.213 } }
		# Company housing where housing is short.
		20 = {
			modifier = { factor = 0 NOT = { event_target:pe_pat_planet = { free_housing < 300 } } }
			country_event = { id = pe.230 }
		}
		# Robot start-up: the empire builds robots.
		20 = {
			modifier = { factor = 0 NOT = { has_technology = tech_robotic_workers } }
			modifier = { factor = 0 has_policy_flag = robots_outlawed }
			country_event = { id = pe.231 }
		}
		30 = { country_event = { id = pe.232 } }
		15 = { country_event = { id = pe.233 } }
	}
}

# Country scope, quarterly: when a market good is short, a private firm that makes it
# offers a deal (pe.260): it delivers a year's shortfall in exchange for a good we have
# plenty of, cheaper than the market would be for us. Its margin goes to the investors.
# At most every three years, a 30% chance per quarter.
pe_deal = {
	if = {
		limit = {
			NOT = { has_country_flag = pe_deal_cd }
			OR = {
				AND = { check_variable = { which = pe_def_alloys value = 1 } check_variable = { which = pe_cnt_a value > 0 } }
				AND = { check_variable = { which = pe_def_consumer_goods value = 1 } check_variable = { which = pe_cnt_c value > 0 } }
				AND = { check_variable = { which = pe_def_minerals value = 1 } check_variable = { which = pe_cnt_m value > 0 } }
				AND = { check_variable = { which = pe_def_energy value = 1 } check_variable = { which = pe_cnt_e value > 0 } }
				AND = { check_variable = { which = pe_def_food value = 1 } check_variable = { which = pe_cnt_f value > 0 } }
			}
		}
		random = {
			chance = 30
			pe_deal_fire = yes
		}
	}
}

# Country scope: pick the short good X (pe_deal_x) and the plentiful good Y (pe_deal_y: the
# fullest stock, at least 40%, not short), then the amounts: X = a year's shortfall (at
# least 100), Y worth 80% of what X costs on the market (at the price we would get for Y),
# at most half of the Y stock. The firm's margin: a quarter of X's market value.
pe_deal_fire = {
	set_timed_country_flag = { flag = pe_deal_cd days = 1080 }
	set_variable = { which = pe_deal_x value = 0 }
	random_list = {
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_alloys value = 1 } } }
			modifier = { factor = 0 NOT = { check_variable = { which = pe_cnt_a value > 0 } } }
			set_variable = { which = pe_deal_x value = 1 }
		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_consumer_goods value = 1 } } }
			modifier = { factor = 0 NOT = { check_variable = { which = pe_cnt_c value > 0 } } }
			set_variable = { which = pe_deal_x value = 2 }
		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_minerals value = 1 } } }
			modifier = { factor = 0 NOT = { check_variable = { which = pe_cnt_m value > 0 } } }
			set_variable = { which = pe_deal_x value = 3 }
		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_energy value = 1 } } }
			modifier = { factor = 0 NOT = { check_variable = { which = pe_cnt_e value > 0 } } }
			set_variable = { which = pe_deal_x value = 4 }
		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_food value = 1 } } }
			modifier = { factor = 0 NOT = { check_variable = { which = pe_cnt_f value > 0 } } }
			set_variable = { which = pe_deal_x value = 5 }
		}
	}
	set_variable = { which = pe_deal_y value = 0 }
	set_variable = { which = pe_deal_best value = 0.4 }
	export_resource_stockpile_to_variable = { resource = alloys variable = pe_ds_1 }
	export_resource_maximum_to_variable = { resource = alloys variable = pe_dm }
	if = {
		limit = {
			check_variable = { which = pe_dm value > 0 }
			NOT = { check_variable = { which = pe_deal_x value = 1 } }
			NOT = { check_variable = { which = pe_def_alloys value = 1 } }
		}
		set_variable = { which = pe_df value = pe_ds_1 }
		divide_variable = { which = pe_df value = pe_dm }
		if = {
			limit = { check_variable = { which = pe_df value > pe_deal_best } }
			set_variable = { which = pe_deal_best value = pe_df }
			set_variable = { which = pe_deal_y value = 1 }
		}
	}
	export_resource_stockpile_to_variable = { resource = consumer_goods variable = pe_ds_2 }
	export_resource_maximum_to_variable = { resource = consumer_goods variable = pe_dm }
	if = {
		limit = {
			check_variable = { which = pe_dm value > 0 }
			NOT = { check_variable = { which = pe_deal_x value = 2 } }
			NOT = { check_variable = { which = pe_def_consumer_goods value = 1 } }
		}
		set_variable = { which = pe_df value = pe_ds_2 }
		divide_variable = { which = pe_df value = pe_dm }
		if = {
			limit = { check_variable = { which = pe_df value > pe_deal_best } }
			set_variable = { which = pe_deal_best value = pe_df }
			set_variable = { which = pe_deal_y value = 2 }
		}
	}
	export_resource_stockpile_to_variable = { resource = minerals variable = pe_ds_3 }
	export_resource_maximum_to_variable = { resource = minerals variable = pe_dm }
	if = {
		limit = {
			check_variable = { which = pe_dm value > 0 }
			NOT = { check_variable = { which = pe_deal_x value = 3 } }
			NOT = { check_variable = { which = pe_def_minerals value = 1 } }
		}
		set_variable = { which = pe_df value = pe_ds_3 }
		divide_variable = { which = pe_df value = pe_dm }
		if = {
			limit = { check_variable = { which = pe_df value > pe_deal_best } }
			set_variable = { which = pe_deal_best value = pe_df }
			set_variable = { which = pe_deal_y value = 3 }
		}
	}
	export_resource_stockpile_to_variable = { resource = energy variable = pe_ds_4 }
	export_resource_maximum_to_variable = { resource = energy variable = pe_dm }
	if = {
		limit = {
			check_variable = { which = pe_dm value > 0 }
			NOT = { check_variable = { which = pe_deal_x value = 4 } }
			NOT = { check_variable = { which = pe_def_energy value = 1 } }
		}
		set_variable = { which = pe_df value = pe_ds_4 }
		divide_variable = { which = pe_df value = pe_dm }
		if = {
			limit = { check_variable = { which = pe_df value > pe_deal_best } }
			set_variable = { which = pe_deal_best value = pe_df }
			set_variable = { which = pe_deal_y value = 4 }
		}
	}
	export_resource_stockpile_to_variable = { resource = food variable = pe_ds_5 }
	export_resource_maximum_to_variable = { resource = food variable = pe_dm }
	if = {
		limit = {
			check_variable = { which = pe_dm value > 0 }
			NOT = { check_variable = { which = pe_deal_x value = 5 } }
			NOT = { check_variable = { which = pe_def_food value = 1 } }
		}
		set_variable = { which = pe_df value = pe_ds_5 }
		divide_variable = { which = pe_df value = pe_dm }
		if = {
			limit = { check_variable = { which = pe_df value > pe_deal_best } }
			set_variable = { which = pe_deal_best value = pe_df }
			set_variable = { which = pe_deal_y value = 5 }
		}
	}
	if = {
		limit = {
			check_variable = { which = pe_deal_x value > 0 }
			check_variable = { which = pe_deal_y value > 0 }
		}
		# Shortfall per month, market prices (per unit), the Y stock.
		if = { limit = { check_variable = { which = pe_deal_x value = 1 } } set_variable = { which = pe_deal_xamt value = pe_netm_alloys } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = alloys amount = 100 trade_type = market_buy } variable = pe_dpx } set_variable = { which = pe_dbase value = 4 } }
		else_if = { limit = { check_variable = { which = pe_deal_x value = 2 } } set_variable = { which = pe_deal_xamt value = pe_netm_consumer_goods } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = consumer_goods amount = 100 trade_type = market_buy } variable = pe_dpx } set_variable = { which = pe_dbase value = 2 } }
		else_if = { limit = { check_variable = { which = pe_deal_x value = 3 } } set_variable = { which = pe_deal_xamt value = pe_netm_minerals } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = minerals amount = 100 trade_type = market_buy } variable = pe_dpx } set_variable = { which = pe_dbase value = 1 } }
		else_if = { limit = { check_variable = { which = pe_deal_x value = 4 } } set_variable = { which = pe_deal_xamt value = pe_netm_energy } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = energy amount = 100 trade_type = market_buy } variable = pe_dpx } set_variable = { which = pe_dbase value = 1 } }
		else_if = { limit = { check_variable = { which = pe_deal_x value = 5 } } set_variable = { which = pe_deal_xamt value = pe_netm_food } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = food amount = 100 trade_type = market_buy } variable = pe_dpx } set_variable = { which = pe_dbase value = 1 } }
		if = { limit = { check_variable = { which = pe_deal_y value = 1 } } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = alloys amount = 100 trade_type = market_sell } variable = pe_dpy } set_variable = { which = pe_deal_ystock value = pe_ds_1 } set_variable = { which = pe_dbasey value = 4 } }
		else_if = { limit = { check_variable = { which = pe_deal_y value = 2 } } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = consumer_goods amount = 100 trade_type = market_sell } variable = pe_dpy } set_variable = { which = pe_deal_ystock value = pe_ds_2 } set_variable = { which = pe_dbasey value = 2 } }
		else_if = { limit = { check_variable = { which = pe_deal_y value = 3 } } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = minerals amount = 100 trade_type = market_sell } variable = pe_dpy } set_variable = { which = pe_deal_ystock value = pe_ds_3 } set_variable = { which = pe_dbasey value = 1 } }
		else_if = { limit = { check_variable = { which = pe_deal_y value = 4 } } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = energy amount = 100 trade_type = market_sell } variable = pe_dpy } set_variable = { which = pe_deal_ystock value = pe_ds_4 } set_variable = { which = pe_dbasey value = 1 } }
		else_if = { limit = { check_variable = { which = pe_deal_y value = 5 } } export_trigger_value_to_variable = { trigger = market_resource_price parameters = { resource = food amount = 100 trade_type = market_sell } variable = pe_dpy } set_variable = { which = pe_deal_ystock value = pe_ds_5 } set_variable = { which = pe_dbasey value = 1 } }
		divide_variable = { which = pe_dpx value = 100 }
		divide_variable = { which = pe_dpy value = 100 }
		if = { limit = { check_variable = { which = pe_dpx value <= 0 } } set_variable = { which = pe_dpx value = pe_dbase } multiply_variable = { which = pe_dpx value = 1.3 } }
		if = { limit = { check_variable = { which = pe_dpy value <= 0 } } set_variable = { which = pe_dpy value = pe_dbasey } multiply_variable = { which = pe_dpy value = 0.7 } }
		multiply_variable = { which = pe_deal_xamt value = -12 }
		if = { limit = { check_variable = { which = pe_deal_xamt value < 100 } } set_variable = { which = pe_deal_xamt value = 100 } }
		set_variable = { which = pe_deal_yamt value = pe_deal_xamt }
		multiply_variable = { which = pe_deal_yamt value = pe_dpx }
		multiply_variable = { which = pe_deal_yamt value = 0.8 }
		divide_variable = { which = pe_deal_yamt value = pe_dpy }
		set_variable = { which = pe_tmp value = pe_deal_ystock }
		multiply_variable = { which = pe_tmp value = 0.5 }
		if = {
			limit = { check_variable = { which = pe_deal_yamt value > pe_tmp } }
			set_variable = { which = pe_deal_yamt value = pe_tmp }
			set_variable = { which = pe_deal_xamt value = pe_deal_yamt }
			multiply_variable = { which = pe_deal_xamt value = pe_dpy }
			divide_variable = { which = pe_deal_xamt value = 0.8 }
			divide_variable = { which = pe_deal_xamt value = pe_dpx }
		}
		# Round to tens.
		divide_variable = { which = pe_deal_xamt value = 10 }
		round_variable = pe_deal_xamt
		multiply_variable = { which = pe_deal_xamt value = 10 }
		divide_variable = { which = pe_deal_yamt value = 10 }
		round_variable = pe_deal_yamt
		multiply_variable = { which = pe_deal_yamt value = 10 }
		set_variable = { which = pe_deal_profit value = pe_deal_xamt }
		multiply_variable = { which = pe_deal_profit value = pe_dpx }
		multiply_variable = { which = pe_deal_profit value = 0.25 }
		round_variable = pe_deal_profit
		if = {
			limit = { check_variable = { which = pe_deal_yamt value >= 10 } }
			country_event = { id = pe.260 }
		}
	}
}

# Country scope (event pe.260/261): the deal. Give Y, get X, the margin to the investors.
pe_deal_apply = {
	if = { limit = { check_variable = { which = pe_deal_x value = 1 } } add_resource = { alloys = 1 mult = pe_deal_xamt } }
	else_if = { limit = { check_variable = { which = pe_deal_x value = 2 } } add_resource = { consumer_goods = 1 mult = pe_deal_xamt } }
	else_if = { limit = { check_variable = { which = pe_deal_x value = 3 } } add_resource = { minerals = 1 mult = pe_deal_xamt } }
	else_if = { limit = { check_variable = { which = pe_deal_x value = 4 } } add_resource = { energy = 1 mult = pe_deal_xamt } }
	else_if = { limit = { check_variable = { which = pe_deal_x value = 5 } } add_resource = { food = 1 mult = pe_deal_xamt } }
	if = { limit = { check_variable = { which = pe_deal_y value = 1 } } add_resource = { alloys = -1 mult = pe_deal_yamt } }
	else_if = { limit = { check_variable = { which = pe_deal_y value = 2 } } add_resource = { consumer_goods = -1 mult = pe_deal_yamt } }
	else_if = { limit = { check_variable = { which = pe_deal_y value = 3 } } add_resource = { minerals = -1 mult = pe_deal_yamt } }
	else_if = { limit = { check_variable = { which = pe_deal_y value = 4 } } add_resource = { energy = -1 mult = pe_deal_yamt } }
	else_if = { limit = { check_variable = { which = pe_deal_y value = 5 } } add_resource = { food = -1 mult = pe_deal_yamt } }
	hidden_effect = { change_variable = { which = pe_capital value = pe_deal_profit } }
}
