# Story events of the private economy.
#   Patrons (pe_patron_events.txt): rich private people spend the investor pool. They are
#   recurring figures (pe_mag_id, reputation pe_mag_rep); a magnate retires after four
#   events and a new one appears.
#   Labour (pe_labour_events.txt): private worker jobs stay empty; firms and workers react.
#   Market (pe_market_events.txt): shares, bonds, takeovers, monopolies, crashes ...
#   pe_quarterly (1) -> pe_patron (2) -> pe_patron_fire (3) -> pe_pat_amounts (4)
#                    -> pe_labour (2) -> pe_labour_fire (3) -> pe_pat_amounts (4)

# --- Patrons ------------------------------------------------------------------------

# Country scope, quarterly: with a full investor pool (4500 or more) and no patron event
# in the last two years, a 25% chance per quarter of a patron event.
pe_patron = {
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 4500 }
			NOT = { has_country_flag = pe_patron_cd }
		}
		random = {
			chance = 25
			pe_patron_fire = yes
		}
	}
}

# Country scope: a patron event. The magnate puts 30% of the investor pool into it
# (pe_pat_spend); options that decline it pay it back. The results grow with it: amounts
# directly, modifiers, deposits and settlers by pe_pat_mult (see pe_pat_amounts).
pe_patron_fire = {
	set_timed_country_flag = { flag = pe_patron_cd days = 730 }
	pe_pat_amounts = yes
	capital_scope = { save_event_target_as = pe_pat_planet }
	capital_scope = { owner_main_species = { save_event_target_as = pe_main_species } }
	# Monopoly (pe.244): one main product has 40% of at least eight firms.
	set_variable = { which = pe_mono value = 0 }
	set_variable = { which = pe_mono_min value = pe_n_firms }
	multiply_variable = { which = pe_mono_min value = 0.4 }
	if = {
		limit = { check_variable = { which = pe_n_firms value >= 8 } }
		if = { limit = { check_variable = { which = pe_cnt_a value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_c value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_t value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_u value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_x value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_y value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_z value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_b value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_m value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_e value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_f value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
	}
	random_list = {
		10 = { country_event = { id = pe.200 } }
		10 = { country_event = { id = pe.202 } }
		8 = { country_event = { id = pe.203 } }
		8 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes OR = { free_housing < 0 free_amenities < 0 } } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes OR = { free_housing < 0 free_amenities < 0 } }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.204 }
		}
		8 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes free_jobs >= 200 count_deposits = { category = pe_deposit_cat_firms count > 0 } } } }
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_has_idle_pops = yes } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes free_jobs >= 200 count_deposits = { category = pe_deposit_cat_firms count > 0 } }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.205 }
		}
		# Prospecting find on a planet.
		8 = {
			random_owned_colony = { limit = { pe_planet_eligible = yes } save_event_target_as = pe_pat_planet }
			country_event = { id = pe.214 }
		}
		# Hydroponics while food is short.
		8 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_food value = 1 } } }
			country_event = { id = pe.215 }
		}
		8 = { country_event = { id = pe.220 } }
		8 = { country_event = { id = pe.221 } }
		6 = { country_event = { id = pe.222 } }
		# Private colonists: a colonisable planet in our borders that suits our main species.
		6 = {
			modifier = {
				factor = 0
				NOT = {
					any_planet_within_border = {
						is_colonizable = yes
						NOT = { exists = owner }
						habitability = { who = event_target:pe_main_species value >= 0.6 }
					}
				}
			}
			random_planet_within_border = {
				limit = {
					is_colonizable = yes
					NOT = { exists = owner }
					habitability = { who = event_target:pe_main_species value >= 0.6 }
				}
				save_event_target_as = pe_pat_colony
			}
			country_event = { id = pe.223 }
		}
		8 = { country_event = { id = pe.224 } }
		8 = {
			random_owned_colony = { limit = { pe_planet_eligible = yes } save_event_target_as = pe_pat_planet }
			country_event = { id = pe.225 }
		}
		# Disaster relief for a planet in crisis.
		8 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes planet_stability < 25 } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes planet_stability < 25 }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.226 }
		}
		# Market and corporate events (pe_market_events.txt).
		6 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_n_corp value > 0 } } }
			country_event = { id = pe.241 }
		}
		6 = { country_event = { id = pe.242 } }
		6 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.243 }
		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_mono value = 1 } } }
			country_event = { id = pe.244 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.245 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.246 }
		}
		6 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes OR = { has_district = district_mining has_district = district_farming has_district = district_generator } count_deposits = { category = pe_deposit_cat_firms count > 0 } } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes OR = { has_district = district_mining has_district = district_farming has_district = district_generator } count_deposits = { category = pe_deposit_cat_firms count > 0 } }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.247 }
		}
		# A crash: rare, only with a big pool; three times as likely after a bubble.
		2 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_capital value >= 7000 } } }
			modifier = { factor = 3 has_country_flag = pe_bubble }
			country_event = { id = pe.248 }
		}
		4 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_capital value >= 3500 } } }
			country_event = { id = pe.249 }
		}
		4 = { country_event = { id = pe.250 } }
		5 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes free_housing > 300 } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes free_housing > 300 }
				save_event_target_as = pe_pat_planet
			}
			country_event = { id = pe.251 }
		}
	}
}

# Country scope: the magnate (a new one after four events), the amount (30% of the
# investor pool, taken from it) and what the options give, all rounded.
pe_pat_amounts = {
	if = {
		limit = {
			OR = {
				NOT = { is_variable_set = pe_mag_id }
				check_variable = { which = pe_mag_events value >= 4 }
			}
		}
		# A magnate retires: their farewell (pe.240) a month later.
		if = {
			limit = { is_variable_set = pe_mag_id }
			set_variable = { which = pe_mag_old value = pe_mag_id }
			set_variable = { which = pe_mag_old_rep value = pe_mag_rep }
			country_event = { id = pe.240 days = 30 }
		}
		set_variable_to_random_value = { which = pe_mag_id min = 1 max = 16 rounded = yes }
		set_variable = { which = pe_mag_events value = 0 }
		set_variable = { which = pe_mag_rep value = 0 }
	}
	change_variable = { which = pe_mag_events value = 1 }
	set_variable = { which = pe_pat_spend value = pe_capital }
	multiply_variable = { which = pe_pat_spend value = 0.3 }
	round_variable = pe_pat_spend
	subtract_variable = { which = pe_capital value = pe_pat_spend }
	set_variable = { which = pe_pat_unity value = pe_pat_spend }
	multiply_variable = { which = pe_pat_unity value = 0.3 }
	round_variable = pe_pat_unity
	set_variable = { which = pe_pat_tax value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tax value = 0.25 }
	round_variable = pe_pat_tax
	set_variable = { which = pe_pat_res value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res value = 0.4 }
	round_variable = pe_pat_res
	set_variable = { which = pe_pat_res_free value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res_free value = 0.5 }
	round_variable = pe_pat_res_free
	set_variable = { which = pe_pat_proto_full value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_full value = 0.6 }
	round_variable = pe_pat_proto_full
	set_variable = { which = pe_pat_proto_part value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_part value = 0.2 }
	round_variable = pe_pat_proto_part
	set_variable = { which = pe_pat_art value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art value = 0.2 }
	round_variable = pe_pat_art
	set_variable = { which = pe_pat_art_state value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art_state value = 0.4 }
	round_variable = pe_pat_art_state
	set_variable = { which = pe_pat_tour value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tour value = 0.2 }
	round_variable = pe_pat_tour
	set_variable = { which = pe_pat_half value = pe_pat_spend }
	multiply_variable = { which = pe_pat_half value = 0.5 }
	round_variable = pe_pat_half
	set_variable = { which = pe_pat_influence value = pe_pat_spend }
	multiply_variable = { which = pe_pat_influence value = 0.05 }
	round_variable = pe_pat_influence
	if = { limit = { check_variable = { which = pe_pat_influence value > 150 } } set_variable = { which = pe_pat_influence value = 150 } }
	# Scale of the results that are no amounts (modifiers, deposits, settlers): the spend in
	# units of 1500 (30% of a pool of 5000), between 0.5 and 3, to one decimal.
	set_variable = { which = pe_pat_mult value = pe_pat_spend }
	divide_variable = { which = pe_pat_mult value = 150 }
	round_variable = pe_pat_mult
	divide_variable = { which = pe_pat_mult value = 10 }
	if = { limit = { check_variable = { which = pe_pat_mult value < 0.5 } } set_variable = { which = pe_pat_mult value = 0.5 } }
	if = { limit = { check_variable = { which = pe_pat_mult value > 3 } } set_variable = { which = pe_pat_mult value = 3 } }
	# Follow-ups and small amounts that used to be fixed.
	set_variable = { which = pe_pat_u50 value = pe_pat_mult }
	multiply_variable = { which = pe_pat_u50 value = 50 }
	round_variable = pe_pat_u50
	set_variable = { which = pe_pat_u100 value = pe_pat_mult }
	multiply_variable = { which = pe_pat_u100 value = 100 }
	round_variable = pe_pat_u100
	# Settlers of a private colony (pe.223): 100 x scale.
	set_variable = { which = pe_col_pops value = pe_pat_u100 }
	# Deposits of a prospecting find (pe.214) and size of the hydroponic farms (pe.215):
	# 1, 2 from scale 1.5, 3 from scale 2.5.
	set_variable = { which = pe_find_n value = 1 }
	if = { limit = { check_variable = { which = pe_pat_mult value >= 1.5 } } set_variable = { which = pe_find_n value = 2 } }
	if = { limit = { check_variable = { which = pe_pat_mult value >= 2.5 } } set_variable = { which = pe_find_n value = 3 } }
	set_variable = { which = pe_hydro_food value = pe_find_n }
	multiply_variable = { which = pe_hydro_food value = 15 }
}

# --- Labour -------------------------------------------------------------------------

# Colony scope: private worker jobs stand empty here (at least one job's worth).
# (Trigger in pe_triggers.txt: pe_rural_short.)

# Country scope, quarterly: when private worker jobs stand empty somewhere and the pool
# can pay wages (1000 or more), a 25% chance per quarter of a labour event; then two
# years' pause.
pe_labour = {
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 1500 }
			NOT = { has_country_flag = pe_labour_cd }
			any_owned_colony = { pe_planet_eligible = yes pe_rural_short = yes }
		}
		random = {
			chance = 25
			pe_labour_fire = yes
		}
	}
}

# Country scope: a labour event on a planet with empty private worker jobs. pe_lab_job:
# the job with the most empty places there (1 miner, 2 technician, 3 farmer).
pe_labour_fire = {
	set_timed_country_flag = { flag = pe_labour_cd days = 730 }
	pe_pat_amounts = yes
	random_owned_colony = {
		limit = { pe_planet_eligible = yes pe_rural_short = yes }
		save_event_target_as = pe_pat_planet
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_miner } variable = pe_lf_m }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_technician } variable = pe_lf_e }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_farmer } variable = pe_lf_f }
		root = {
			set_variable = { which = pe_lab_job value = 1 }
			set_variable = { which = pe_tmp value = prev.pe_lf_m }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_e } } set_variable = { which = pe_lab_job value = 2 } set_variable = { which = pe_tmp value = prev.pe_lf_e } }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_f } } set_variable = { which = pe_lab_job value = 3 } }
		}
	}
	random_list = {
		30 = { country_event = { id = pe.210 } }
		# Guest workers: an empire we know, whose founder species finds the planet
		# habitable and would not be purged here.
		25 = {
			modifier = {
				factor = 0
				NOT = {
					any_country = {
						is_country_type = default
						NOT = { is_same_value = root }
						has_communications = root
					}
				}
			}
			random_country = {
				limit = {
					is_country_type = default
					NOT = { is_same_value = root }
					has_communications = root
				}
				founder_species = { save_event_target_as = pe_gw_species }
			}
			country_event = { id = pe.211 }
		}
		25 = { country_event = { id = pe.212 } }
		20 = { country_event = { id = pe.213 } }
		# Company housing where housing is short.
		20 = {
			modifier = { factor = 0 NOT = { event_target:pe_pat_planet = { free_housing < 300 } } }
			country_event = { id = pe.230 }
		}
		# Robot start-up: the empire builds robots.
		20 = {
			modifier = { factor = 0 NOT = { has_technology = tech_robotic_workers } }
			modifier = { factor = 0 has_policy_flag = robots_outlawed }
			country_event = { id = pe.231 }
		}
		30 = { country_event = { id = pe.232 } }
		15 = { country_event = { id = pe.233 } }
	}
}
