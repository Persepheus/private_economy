# Story events of the private economy.
#   Patrons (pe_patron_events.txt): rich private people spend the investor pool. They are
#   recurring figures (pe_mag_id, reputation pe_mag_rep); a magnate retires after four
#   events and a new one appears.
#   Labour (pe_labour_events.txt): private worker jobs stay empty; firms and workers react.
#   Market (pe_market_events.txt): shares, bonds, takeovers, monopolies, crashes ...
#   Flair (pe_flair_events.txt): supply deals, modernisation, rationalisation, capital
#   levy, and the neighbours (investment abroad, foreign branches, licences, dumping).
#   pe_quarterly (1) -> pe_patron (2) -> pe_patron_fire (3) -> pe_pat_amounts (4)
#                    -> pe_labour (2) -> pe_labour_fire (3) -> pe_pat_amounts (4)
#                    -> pe_deal (2) -> pe_deal_fire (3)
#   pe_patron_fire (3) -> pe_pick_sector (4), pe_pick_free_spec (4)

# --- Patrons ------------------------------------------------------------------------

# Country scope, quarterly: with a full investor pool (4500 or more) and no patron event
# in the last two years, a 25% chance per quarter of a patron event.
pe_patron = {
	set_variable = { which = pe_fd_fire value = 0 }
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 4500 }
			NOT = { has_country_flag = pe_patron_cd }
		}
		random = {
			chance = 25
			set_variable = { which = pe_pat_src value = 0 }
			set_variable = { which = pe_fd_fire value = 1 }
			pe_patron_fire = yes
		}
	}
	# Corporate foundations: the richest corporation with 20000 cash funds such an event with a
	# tenth of its cash (25% a quarter, not in a quarter with a patron, then two years' pause;
	# the patrons keep their own pause). Its foundation stands in for the magnate.
	if = {
		limit = { check_variable = { which = pe_fd_fire value = 0 } NOT = { has_country_flag = pe_found_cd } }
		set_variable = { which = pe_pat_src value = 0 }
		set_variable = { which = pe_fd_best value = 19999 }
		if = { limit = { check_variable = { which = pe_c1_on value = 1 } check_variable = { which = pe_c1_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 1 } set_variable = { which = pe_fd_best value = pe_c1_cash } set_variable = { which = pe_pat_srcn value = pe_c1_name } }
		if = { limit = { check_variable = { which = pe_c2_on value = 1 } check_variable = { which = pe_c2_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 2 } set_variable = { which = pe_fd_best value = pe_c2_cash } set_variable = { which = pe_pat_srcn value = pe_c2_name } }
		if = { limit = { check_variable = { which = pe_c3_on value = 1 } check_variable = { which = pe_c3_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 3 } set_variable = { which = pe_fd_best value = pe_c3_cash } set_variable = { which = pe_pat_srcn value = pe_c3_name } }
		if = { limit = { check_variable = { which = pe_c4_on value = 1 } check_variable = { which = pe_c4_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 4 } set_variable = { which = pe_fd_best value = pe_c4_cash } set_variable = { which = pe_pat_srcn value = pe_c4_name } }
		if = { limit = { check_variable = { which = pe_c5_on value = 1 } check_variable = { which = pe_c5_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 5 } set_variable = { which = pe_fd_best value = pe_c5_cash } set_variable = { which = pe_pat_srcn value = pe_c5_name } }
		if = { limit = { check_variable = { which = pe_c6_on value = 1 } check_variable = { which = pe_c6_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 6 } set_variable = { which = pe_fd_best value = pe_c6_cash } set_variable = { which = pe_pat_srcn value = pe_c6_name } }
		if = { limit = { check_variable = { which = pe_c7_on value = 1 } check_variable = { which = pe_c7_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 7 } set_variable = { which = pe_fd_best value = pe_c7_cash } set_variable = { which = pe_pat_srcn value = pe_c7_name } }
		if = { limit = { check_variable = { which = pe_c8_on value = 1 } check_variable = { which = pe_c8_cash value > pe_fd_best } } set_variable = { which = pe_pat_src value = 8 } set_variable = { which = pe_fd_best value = pe_c8_cash } set_variable = { which = pe_pat_srcn value = pe_c8_name } }
		if = {
			limit = { check_variable = { which = pe_pat_src value > 0 } }
			random = {
				chance = 25
				set_timed_country_flag = { flag = pe_found_cd days = 730 }
				set_variable = { which = pe_cnpat value = pe_pat_srcn }
				pe_patron_fire = yes
			}
		}
		if = { limit = { NOT = { has_country_flag = pe_found_cd } } set_variable = { which = pe_pat_src value = 0 } }
	}
}

# Country scope: a patron event. The magnate puts 30% of the investor pool into it

		if = { limit = { check_variable = { which = pe_cnt_y value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }
		if = { limit = { check_variable = { which = pe_cnt_z value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }

		if = { limit = { check_variable = { which = pe_cnt_f value >= pe_mono_min } } set_variable = { which = pe_mono value = 1 } }

			}
			country_event = { id = pe.204 }
		}
		8 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes free_jobs >= 200 count_deposits = { category = pe_deposit_cat_firms count > 0 } } } }
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_has_idle_pops = yes } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes free_jobs >= 200 count_deposits = { category = pe_deposit_cat_firms count > 0 } }

			}

		8 = {

			country_event = { id = pe.214 }
		}
		# Hydroponics while food is short.
		8 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_def_food value = 1 } } }
			country_event = { id = pe.215 }
		}
		8 = { country_event = { id = pe.220 } }
		8 = { country_event = { id = pe.221 } }
		6 = { country_event = { id = pe.222 } }

			modifier = {
				factor = 0
				NOT = {
					any_planet_within_border = {
						is_colonizable = yes
						NOT = { exists = owner }
						habitability = { who = event_target:pe_main_species value >= 0.6 }
					}
				}
			}
			random_planet_within_border = {
				limit = {
					is_colonizable = yes
					NOT = { exists = owner }

			}

		}
		8 = { country_event = { id = pe.224 } }
		8 = {
			random_owned_colony = { limit = { pe_planet_eligible = yes } save_event_target_as = pe_pat_planet }
			country_event = { id = pe.225 }
		}

		8 = {

			if = {
				limit = { OR = { has_country_flag = pe_ship_lost_sci count_owned_ship = { limit = { is_ship_size = science } count <= 1 } } }
				set_variable = { which = pe_don_type value = 1 }
			}
			country_event = { id = pe.291 }
		}
		# Flair events (pe_flair_events.txt).
		8 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } NOT = { pe_has_idle_pops = yes } } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } NOT = { pe_has_idle_pops = yes } }

		}
		6 = {
			modifier = { factor = 0 NOT = { any_owned_colony = { pe_planet_eligible = yes pe_spec_short = yes } } }
			random_owned_colony = {
				limit = { pe_planet_eligible = yes pe_spec_short = yes }
				save_event_target_as = pe_pat_planet
				pe_pick_free_spec = yes
			}

		}
		10 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_capital value >= 10000 } } }
			country_event = { id = pe.264 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } } } }

				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.265 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } }
				save_event_target_as = pe_pat_country
			}
			random_owned_colony = { limit = { pe_planet_eligible = yes } save_event_target_as = pe_pat_planet }
			country_event = { id = pe.266 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root NOT = { is_at_war_with = root } }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.267 }
		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root } } }
			modifier = { factor = 0 NOT = { OR = { check_variable = { which = pe_cnt_a value >= 2 } check_variable = { which = pe_cnt_c value >= 2 } } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root }
				save_event_target_as = pe_pat_country
			}
			country_event = { id = pe.268 }
		}
		# Market and corporate events (pe_market_events.txt).
		6 = {
			modifier = { factor = 0 NOT = { check_variable = { which = pe_n_corp value > 0 } } }
			country_event = { id = pe.241 }
		}
		# State bonds: four times as often at war or in a deficit (energy, trade, alloys).
		6 = {
			modifier = { factor = 4 OR = { is_at_war = yes has_monthly_income = { resource = energy value < 0 } has_monthly_income = { resource = trade value < 0 } has_monthly_income = { resource = alloys value < 0 } } }

			random_owned_colony = {
				limit = { pe_planet_eligible = yes count_deposits = { category = pe_deposit_cat_firms count >= 2 } }

			}

		10 = {

		}
		5 = {
			modifier = { factor = 0 NOT = { any_country = { is_country_type = default NOT = { is_same_value = root } has_communications = root } } }
			random_country = {
				limit = { is_country_type = default NOT = { is_same_value = root } has_communications = root }

			modifier = { factor = 3 has_country_flag = pe_bubble }

# Country scope: the magnate (a new one after four events), the amount (30% of the
# investor pool, taken from it) and what the options give, all rounded.
pe_pat_amounts = {
	if = { limit = { NOT = { is_variable_set = pe_pat_src } } set_variable = { which = pe_pat_src value = 0 } }
	if = {
		limit = {
			check_variable = { which = pe_pat_src value = 0 }
			OR = {
				NOT = { is_variable_set = pe_mag_id }
				check_variable = { which = pe_mag_events value >= 4 }
			}
		}
		# A magnate retires: their farewell (pe.240) a month later.
		if = {
			limit = { is_variable_set = pe_mag_id }
			set_variable = { which = pe_mag_old value = pe_mag_id }
			set_variable = { which = pe_mag_old_rep value = pe_mag_rep }
			country_event = { id = pe.240 days = 30 }
		}
		set_variable_to_random_value = { which = pe_mag_id min = 1 max = 16 rounded = yes }
		set_variable = { which = pe_mag_events value = 0 }
		set_variable = { which = pe_mag_rep value = 0 }
	}
	# The money: a patron puts in 30% of the investor pool, a corporate foundation 10% of the
	# corporation's cash (pe_pat_src).
	if = {
		limit = { check_variable = { which = pe_pat_src value > 0 } }
		if = { limit = { check_variable = { which = pe_pat_src value = 1 } } set_variable = { which = pe_pat_spend value = pe_c1_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c1_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 2 } } set_variable = { which = pe_pat_spend value = pe_c2_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c2_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 3 } } set_variable = { which = pe_pat_spend value = pe_c3_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c3_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 4 } } set_variable = { which = pe_pat_spend value = pe_c4_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c4_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 5 } } set_variable = { which = pe_pat_spend value = pe_c5_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c5_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 6 } } set_variable = { which = pe_pat_spend value = pe_c6_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c6_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 7 } } set_variable = { which = pe_pat_spend value = pe_c7_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c7_cash value = pe_pat_spend } }
		else_if = { limit = { check_variable = { which = pe_pat_src value = 8 } } set_variable = { which = pe_pat_spend value = pe_c8_cash } multiply_variable = { which = pe_pat_spend value = 0.1 } round_variable = pe_pat_spend subtract_variable = { which = pe_c8_cash value = pe_pat_spend } }
	}
	else = {
		change_variable = { which = pe_mag_events value = 1 }
		set_variable = { which = pe_pat_spend value = pe_capital }
		multiply_variable = { which = pe_pat_spend value = 0.3 }
		round_variable = pe_pat_spend
		subtract_variable = { which = pe_capital value = pe_pat_spend }
	}
	set_variable = { which = pe_pat_unity value = pe_pat_spend }
	multiply_variable = { which = pe_pat_unity value = 0.3 }
	round_variable = pe_pat_unity
	set_variable = { which = pe_pat_tax value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tax value = 0.25 }
	round_variable = pe_pat_tax
	set_variable = { which = pe_pat_res value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res value = 0.4 }
	round_variable = pe_pat_res
	set_variable = { which = pe_pat_res_free value = pe_pat_spend }
	multiply_variable = { which = pe_pat_res_free value = 0.5 }
	round_variable = pe_pat_res_free
	set_variable = { which = pe_pat_proto_full value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_full value = 0.6 }
	round_variable = pe_pat_proto_full
	set_variable = { which = pe_pat_proto_part value = pe_pat_spend }
	multiply_variable = { which = pe_pat_proto_part value = 0.2 }
	round_variable = pe_pat_proto_part
	set_variable = { which = pe_pat_art value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art value = 0.2 }
	round_variable = pe_pat_art
	set_variable = { which = pe_pat_art_state value = pe_pat_spend }
	multiply_variable = { which = pe_pat_art_state value = 0.4 }
	round_variable = pe_pat_art_state
	set_variable = { which = pe_pat_tour value = pe_pat_spend }
	multiply_variable = { which = pe_pat_tour value = 0.2 }
	round_variable = pe_pat_tour
	set_variable = { which = pe_pat_half value = pe_pat_spend }

# --- Labour -------------------------------------------------------------------------

# Colony scope: private worker jobs stand empty here (at least one job's worth).
# (Trigger in pe_triggers.txt: pe_rural_short.)

# Country scope, quarterly: when private worker jobs stand empty somewhere and the pool
# can pay wages (1000 or more), a 25% chance per quarter of a labour event; then two
# years' pause.
pe_labour = {
	if = {
		limit = {
			check_variable = { which = pe_capital value >= 1500 }
			NOT = { has_country_flag = pe_labour_cd }
			any_owned_colony = { pe_planet_eligible = yes pe_rural_short = yes }
		}
		random = {
			chance = 25
			pe_labour_fire = yes
		}
	}
}

# Country scope: a labour event on a planet with empty private worker jobs. pe_lab_job:
# the job with the most empty places there (1 miner, 2 technician, 3 farmer).
pe_labour_fire = {
	set_variable = { which = pe_pat_src value = 0 }
	set_timed_country_flag = { flag = pe_labour_cd days = 730 }
	pe_pat_amounts = yes
	random_owned_colony = {
		limit = { pe_planet_eligible = yes pe_rural_short = yes }
		save_event_target_as = pe_pat_planet
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_miner } variable = pe_lf_m }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_technician } variable = pe_lf_e }
		export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { job = pe_private_farmer } variable = pe_lf_f }
		root = {
			set_variable = { which = pe_lab_job value = 1 }
			set_variable = { which = pe_tmp value = prev.pe_lf_m }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_e } } set_variable = { which = pe_lab_job value = 2 } set_variable = { which = pe_tmp value = prev.pe_lf_e } }
			if = { limit = { check_variable = { which = pe_tmp value < prev.pe_lf_f } } set_variable = { which = pe_lab_job value = 3 } }
		}
	}
	random_list = {
		30 = { country_event = { id = pe.210 } }
		# Guest workers: an empire we know, whose founder species finds the planet
		# habitable and would not be purged here.
		25 = {
			modifier = {
				factor = 0
				NOT = {
					any_country = {
						is_country_type = default
						NOT = { is_same_value = root }
						has_communications = root
					}
				}
			}
			random_country = {
				limit = {
					is_country_type = default
					NOT = { is_same_value = root }
					has_communications = root
				}
				founder_species = { save_event_target_as = pe_gw_species }
			}
			country_event = { id = pe.211 }
		}
		25 = { country_event = { id = pe.212 } }
		20 = { country_event = { id = pe.213 } }
		# Company housing where housing is short.
		20 = {
			modifier = { factor = 0 NOT = { event_target:pe_pat_planet = { free_housing < 300 } } }
			country_event = { id = pe.230 }
		}
		# Robot start-up: the empire builds robots.
		20 = {
			modifier = { factor = 0 NOT = { has_technology = tech_robotic_workers } }
			modifier = { factor = 0 has_policy_flag = robots_outlawed }
			country_event = { id = pe.231 }
		}
		30 = { country_event = { id = pe.232 } }
		# Family support: only where pops grow (organic ones; not robots, not machines).
		15 = {
			modifier = { factor = 0 event_target:pe_pat_planet = { NOT = { any_owned_pop_group = { species = { is_robotic = no } } } } }
			country_event = { id = pe.233 }
		}
	}
}

# --- Supply deals ---------------------------------------------------------------------

# Country scope, quarterly: when a market good is short and its store less than half full, a
# private firm that makes it offers a supply deal (pe.260): for two years it delivers every
# month what we lack, in exchange for a good we have plenty of, cheaper than the market would
# be for us. Its margin goes to the investors. At most every three years and never while a
# deal runs, a 30% chance per quarter.
pe_deal = {
	if = {
		limit = {
			NOT = { has_country_flag = pe_deal_cd }
			NOT = { check_variable = { which = pe_deal_mon value > 0 } }
			OR = {
				AND = { check_variable = { which = pe_def_alloys value = 1 } check_variable = { which = pe_cnt_a value > 0 } }
				AND = { check_variable = { which = pe_def_consumer_goods value = 1 } check_variable = { which = pe_cnt_c value > 0 } }
				AND = { check_variable = { which = pe_def_minerals value = 1 } check_variable = { which = pe_cnt_m value > 0 } }
				AND = { check_variable = { which = pe_def_energy value = 1 } check_variable = { which = pe_cnt_e value > 0 } }
				AND = { check_variable = { which = pe_def_food value = 1 } check_variable = { which = pe_cnt_f value > 0 } }
			}
		}
		random = {
			chance = 30
			pe_deal_fire = yes
		}
	}
}

		else_if = { limit = { check_variable = { which = pe_deal_x value = 4 } } set_variable = { which = pe_deal_xamt value = pe_netm_energy } export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = energy } variable = pe_dg } }
		else_if = { limit = { check_variable = { which = pe_deal_x value = 5 } } set_variable = { which = pe_deal_xamt value = pe_netm_food } export_trigger_value_to_variable = { trigger = resource_revenue_compare parameters = { resource = food } variable = pe_dg } }
		multiply_variable = { which = pe_deal_xamt value = -1 }
		multiply_variable = { which = pe_dg value = 0.05 }
		if = { limit = { check_variable = { which = pe_deal_xamt value < pe_dg } } set_variable = { which = pe_deal_xamt value = pe_dg } }
		if = { limit = { check_variable = { which = pe_deal_xamt value < 5 } } set_variable = { which = pe_deal_xamt value = 5 } }
		if = { limit = { check_variable = { which = pe_deal_xamt value > 500 } } set_variable = { which = pe_deal_xamt value = 500 } }
		round_variable = pe_deal_xamt
		# The payment per month, at most 500 and what half the Y stock pays for 24 months.
		set_variable = { which = pe_deal_yamt value = pe_deal_xamt }
		multiply_variable = { which = pe_deal_yamt value = pe_dpx }
		multiply_variable = { which = pe_deal_yamt value = 0.8 }
		divide_variable = { which = pe_deal_yamt value = pe_dpy }
		set_variable = { which = pe_tmp value = pe_deal_ystock }
		divide_variable = { which = pe_tmp value = 48 }
		if = { limit = { check_variable = { which = pe_tmp value > 500 } } set_variable = { which = pe_tmp value = 500 } }
		if = {
			limit = { check_variable = { which = pe_deal_yamt value > pe_tmp } }
			set_variable = { which = pe_deal_yamt value = pe_tmp }
			set_variable = { which = pe_deal_xamt value = pe_deal_yamt }
			multiply_variable = { which = pe_deal_xamt value = pe_dpy }
			divide_variable = { which = pe_deal_xamt value = 0.8 }
			divide_variable = { which = pe_deal_xamt value = pe_dpx }
			if = { limit = { check_variable = { which = pe_deal_xamt value > 500 } } set_variable = { which = pe_deal_xamt value = 500 } }
			round_variable = pe_deal_xamt
		}
		round_variable = pe_deal_yamt
		set_variable = { which = pe_deal_ytot value = pe_deal_yamt }
		multiply_variable = { which = pe_deal_ytot value = 24 }
		set_variable = { which = pe_deal_xtot value = pe_deal_xamt }
		multiply_variable = { which = pe_deal_xtot value = 24 }
		set_variable = { which = pe_deal_profit value = pe_deal_xamt }
		multiply_variable = { which = pe_deal_profit value = pe_dpx }
		multiply_variable = { which = pe_deal_profit value = 0.25 }
		round_variable = pe_deal_profit
		if = {
			limit = { check_variable = { which = pe_deal_yamt value >= 1 } check_variable = { which = pe_deal_xamt value >= 1 } }
			country_event = { id = pe.260 }
		}
	}
}

# Country scope (event pe.260/261): the deal starts. For 24 months the supply deal deposit on
# the capital produces X and costs Y (budget line "Supply Deal"); pe_deal_month counts down,
# pays the firm's margin to the investors and ends it.
pe_deal_apply = {
	set_variable = { which = pe_deal_mon value = 24 }
	set_variable = { which = pe_deal_pm value = pe_deal_profit }
	set_variable = { which = pe_deal_my value = pe_deal_yamt }
	set_variable = { which = pe_deal_myr value = pe_deal_y }
	every_owned_planet = { limit = { has_deposit = d_pe_supply_deal } remove_deposit = d_pe_supply_deal }
	capital_scope = {
		set_variable = { which = pe_dlx value = root.pe_deal_x }
		set_variable = { which = pe_dly value = root.pe_deal_y }
		root = { set_variable = { which = pe_lv value = pe_deal_xamt } }
		pe_ledger_bits = { V = dlxa }
		root = { set_variable = { which = pe_lv value = pe_deal_yamt } }
		pe_ledger_bits = { V = dlya }
		add_deposit = d_pe_supply_deal
	}
}

# Country scope, monthly (pe.100): a running supply deal. The margin goes to the investors;
# after 24 months, or when we cannot pay (the Y stock below a month's payment), it ends.
pe_deal_month = {
	if = {
		limit = { check_variable = { which = pe_deal_mon value > 0 } }
		change_variable = { which = pe_deal_mon value = -1 }
		change_variable = { which = pe_capital value = pe_deal_pm }
		set_variable = { which = pe_deal_end value = 0 }
		if = { limit = { check_variable = { which = pe_deal_mon value <= 0 } } set_variable = { which = pe_deal_end value = 1 } }
		if = { limit = { check_variable = { which = pe_deal_myr value = 1 } } export_resource_stockpile_to_variable = { resource = alloys variable = pe_dys } }
		else_if = { limit = { check_variable = { which = pe_deal_myr value = 2 } } export_resource_stockpile_to_variable = { resource = consumer_goods variable = pe_dys } }
		else_if = { limit = { check_variable = { which = pe_deal_myr value = 3 } } export_resource_stockpile_to_variable = { resource = minerals variable = pe_dys } }
		else_if = { limit = { check_variable = { which = pe_deal_myr value = 4 } } export_resource_stockpile_to_variable = { resource = energy variable = pe_dys } }
		else_if = { limit = { check_variable = { which = pe_deal_myr value = 5 } } export_resource_stockpile_to_variable = { resource = food variable = pe_dys } }
		if = { limit = { check_variable = { which = pe_deal_end value = 0 } check_variable = { which = pe_dys value < pe_deal_my } } set_variable = { which = pe_deal_end value = 2 } }
		if = {
			limit = { check_variable = { which = pe_deal_end value > 0 } }
			set_variable = { which = pe_deal_mon value = 0 }
			every_owned_planet = {
				limit = { has_deposit = d_pe_supply_deal }
				remove_deposit = d_pe_supply_deal
				save_event_target_as = pe_msg_planet
			}
			if = {
				limit = { is_ai = no exists = event_target:pe_msg_planet }
				if = { limit = { check_variable = { which = pe_deal_end value = 1 } } if = { limit = { NOT = { has_country_flag = pe_mute_patrons } } create_message = { type = pe_private_economy_patrons localization = pe_msg_deal_end days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } } } }
				else = { if = { limit = { NOT = { has_country_flag = pe_mute_patrons } } create_message = { type = pe_private_economy_patrons localization = pe_msg_deal_broken days = 30 target = event_target:pe_msg_planet variable = { type = name localization = PLANET scope = event_target:pe_msg_planet } } } }
			}
		}
	}
}
