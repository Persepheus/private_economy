# Private Economy: the corporate habitat (see pe_corp_effects_generated.txt, pe_chab_*).
# Our Private Holdings get the site (stage 0, spawned by pe_chab_exec) and build it up to the
# complex at once: vanilla build time of a habitat, at our megastructure build speed
# (pe_hold_mega_speed). When it is done, the vanilla habitat appears like after a vanilla
# build, and our empire colonizes it for the corporation (pe.301). Neither can be built by
# anyone else (potential), and neither costs anything here (the corporation and we have paid).

@pe_chab_build_time = 1800

pe_corp_habitat_site = {
	entity = "habitat_phase_03_entity"
	construction_entity = "habitat_phase_03_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	show_galactic_map_icon = no
	show_in_outliner = no
	entity_offset = { x = 7 y = -7 }
	build_time = 1
	resources = {
		category = megastructures_habitat
	}
	construction_blocks_and_blocked_by = none
	potential = { always = no }
	possible = { always = yes }
}

pe_corp_habitat_complex = {
	entity = "habitat_phase_03_entity"
	construction_entity = "habitat_phase_03_entity"
	portrait = "GFX_megastructure_habitat_background"
	place_entity_on_planet_plane = no
	show_galactic_map_icon = no
	show_in_outliner = no
	entity_offset = { x = 7 y = -7 }
	build_time = @pe_chab_build_time
	resources = {
		category = megastructures_habitat
	}
	construction_blocks_and_blocked_by = none
	upgrade_from = {
		pe_corp_habitat_site
	}
	potential = { is_country_type = pe_holdings }
	possible = { always = yes }

	# from = the holdings, fromfrom = this megastructure. The habitat only if the system still
	# belongs to the holdings' empire and has no habitat yet (vanilla rule).
	on_build_complete = {
		fromfrom.planet = { save_event_target_as = target_planet }
		fromfrom.solar_system = { save_event_target_as = pe_chab_system }
		from = { save_event_target_as = pe_chab_hold }
		event_target:pe_chab_system = {
			if = { limit = { exists = owner } owner = { save_event_target_as = pe_chab_owner } }
		}
		if = {
			limit = {
				exists = event_target:pe_chab_owner
				event_target:pe_chab_hold = { has_relation_flag = { who = event_target:pe_chab_owner flag = pe_holdings_of } }
				event_target:pe_chab_system = {
					NOT = { any_system_planet = { OR = { has_planet_flag = habitat has_planet_flag = hold_the_line_habitat } } }
				}
			}
			if = {
				limit = { event_target:target_planet = { is_star = yes } }
				spawn_habitat_effect = {
					HABITAT_OWNER = event_target:pe_chab_owner
					TARGET_PLANET = event_target:target_planet
					DISTANCE = 19.798
				}
			}
			else = {
				spawn_habitat_effect = {
					HABITAT_OWNER = event_target:pe_chab_owner
					TARGET_PLANET = event_target:target_planet
					DISTANCE = 9.899
				}
			}
			event_target:pe_chab_system = { set_star_flag = has_habitat }
			event_target:target_planet = {
				event_target:target_habitat = {
					set_planet_flag = pe_chab_new
					set_variable = { which = pe_chab_corp value = prev.pe_chab_k }
					set_variable = { which = pe_chab_nm value = prev.pe_chab_nm }
				}
			}
			event_target:pe_chab_owner = { country_event = { id = pe.301 } }
		}
		# A habitat came first (vanilla allows one per system): our money back (pe.304).
		else_if = {
			limit = {
				exists = event_target:pe_chab_owner
				event_target:pe_chab_hold = { has_relation_flag = { who = event_target:pe_chab_owner flag = pe_holdings_of } }
			}
			event_target:target_planet = {
				if = { limit = { NOT = { is_variable_set = pe_chab_inf } } set_variable = { which = pe_chab_inf value = 200 } }
				set_planet_flag = pe_chab_refund
				event_target:pe_chab_owner = {
					set_variable = { which = pe_chab_rk value = prev.pe_chab_k }
					set_variable = { which = pe_chab_rnm value = prev.pe_chab_nm }
					set_variable = { which = pe_chab_rinf value = prev.pe_chab_inf }
				}
			}
			event_target:pe_chab_owner = { country_event = { id = pe.304 } }
		}
		event_target:target_planet = {
			remove_planet_flag = has_megastructure
			remove_planet_flag = pe_chab_site
		}
		remove_megastructure = fromfrom
	}
}

pe_corp_swarm_build = {
	entity = "dyson_swarm_phase_1_entity"
	construction_entity = "dyson_swarm_phase_1_entity"
	portrait = "GFX_megastructure_dyson_swarm_background"
	entity_offset = { x = 0.1 y = 0.1 }
	rotate_to_center = no
	scale_offset = yes
	place_entity_on_planet_plane = yes
	show_in_outliner = no
	build_time = 1080
	resources = {
		category = megastructures
	}
	construction_blocks_and_blocked_by = none
	upgrade_from = {
		pe_corp_swarm_site
	}
	potential = { is_country_type = pe_holdings }
	possible = { always = yes }

	# from = the owner (the holdings, or our empire if the corporation is gone), fromfrom = this.
	on_build_complete = {
		fromfrom.planet = { save_event_target_as = pe_sw_target }
		fromfrom.solar_system = { save_event_target_as = pe_sw_tsys }
		from = { save_event_target_as = pe_sw_owner }
		remove_megastructure = fromfrom
		event_target:pe_sw_tsys = {
			set_star_flag = conduit_excluded
			spawn_megastructure = {
				type = dyson_swarm_1
				owner = event_target:pe_sw_owner
				planet = event_target:pe_sw_target
			}
		}
		event_target:pe_sw_target = {
			if = { limit = { NOT = { has_modifier = dyson_swarm_1_mod } } add_modifier = { modifier = dyson_swarm_1_mod days = -1 } }
			set_planet_flag = has_megastructure
			remove_planet_flag = pe_swb
		}
		event_target:pe_sw_tsys = {
			if = {
				limit = { exists = owner owner = { is_ai = no } }
				owner = { create_message = { type = pe_private_economy localization = pe_msg_swarm_done days = 30 target = event_target:pe_sw_target variable = { type = name localization = PLANET scope = event_target:pe_sw_target } } }
			}
		}
	}
}

				planet = star
				graphical_culture = event_target:pe_cit_owner
				orbit_distance = trigger:inner_radius
				orbit_angle = 200
			}
		}
		event_target:pe_cit_owner = {
			random_owned_fleet = {
				limit = { is_ship_size = starbase_deep_space_citadel_1 NOT = { has_fleet_flag = pe_ccit } NOT = { has_fleet_flag = pe_proto_citadel } }

				set_variable = { which = pe_ccit_p value = 0 }
				set_variable = { which = pe_ccit_new value = 1 }
			}
		}
		fromfrom = {
			if = {
				limit = { exists = event_target:pe_cit_newfleet }
				event_target:pe_cit_newfleet = {
					set_variable = { which = pe_ccit_k value = prev.pe_cit_k }
