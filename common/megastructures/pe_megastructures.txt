# Private Economy: the corporate habitat (see pe_corp_effects_generated.txt, pe_chab_*).
# Our Private Holdings get the site (stage 0, spawned by pe_chab_exec) and build it up to the
# complex at once: vanilla build time of a habitat, at our megastructure build speed
# (pe_hold_mega_speed). When it is done, the vanilla habitat appears like after a vanilla
# build, and our empire colonizes it for the corporation (pe.301). Neither can be built by
# anyone else (potential), and neither costs anything here (the corporation and we have paid).

@pe_chab_build_time = 1800

pe_corp_habitat_site = {
	entity = "habitat_phase_03_entity"
	construction_entity = "habitat_phase_03_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	show_galactic_map_icon = no
	show_in_outliner = no
	entity_offset = { x = 7 y = -7 }
	build_time = 1
	resources = {
		category = megastructures_habitat
	}
	construction_blocks_and_blocked_by = none
	potential = { always = no }
	possible = { always = yes }
}

pe_corp_habitat_complex = {
	entity = "habitat_phase_03_entity"
	construction_entity = "habitat_phase_03_entity"
	portrait = "GFX_megastructure_habitat_background"
	place_entity_on_planet_plane = no
	show_galactic_map_icon = no
	show_in_outliner = no
	entity_offset = { x = 7 y = -7 }
	build_time = @pe_chab_build_time
	resources = {
		category = megastructures_habitat
	}
	construction_blocks_and_blocked_by = none
	upgrade_from = {
		pe_corp_habitat_site
	}
	potential = { is_country_type = pe_holdings }
	possible = { always = yes }

	# from = the holdings, fromfrom = this megastructure. The habitat only if the system still
	# belongs to the holdings' empire and has no habitat yet (vanilla rule).
	on_build_complete = {
		fromfrom.planet = { save_event_target_as = target_planet }
		fromfrom.solar_system = { save_event_target_as = pe_chab_system }
		from = { save_event_target_as = pe_chab_hold }
		event_target:pe_chab_system = {
			if = { limit = { exists = owner } owner = { save_event_target_as = pe_chab_owner } }
		}
		if = {
			limit = {
				exists = event_target:pe_chab_owner
				event_target:pe_chab_hold = { has_relation_flag = { who = event_target:pe_chab_owner flag = pe_holdings_of } }
				event_target:pe_chab_system = {
					NOT = { any_system_planet = { OR = { has_planet_flag = habitat has_planet_flag = hold_the_line_habitat } } }
				}
			}
			if = {
				limit = { event_target:target_planet = { is_star = yes } }
				spawn_habitat_effect = {
					HABITAT_OWNER = event_target:pe_chab_owner
					TARGET_PLANET = event_target:target_planet
					DISTANCE = 19.798
				}
			}
			else = {
				spawn_habitat_effect = {
					HABITAT_OWNER = event_target:pe_chab_owner
					TARGET_PLANET = event_target:target_planet
					DISTANCE = 9.899
				}
			}
			event_target:pe_chab_system = { set_star_flag = has_habitat }
			event_target:target_planet = {
				event_target:target_habitat = {
					set_planet_flag = pe_chab_new
					set_variable = { which = pe_chab_corp value = prev.pe_chab_k }
					set_variable = { which = pe_chab_nm value = prev.pe_chab_nm }
				}
			}
			event_target:pe_chab_owner = { country_event = { id = pe.301 } }
		}
		event_target:target_planet = {
			remove_planet_flag = has_megastructure
			remove_planet_flag = pe_chab_site
		}
		remove_megastructure = fromfrom
	}
}
