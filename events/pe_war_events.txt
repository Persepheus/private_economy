# Private economy at war (pe_war_effects.txt).
#   pe.138 a colony with firms is occupied: contributions   pe.139 it is liberated: collaborators
#   pe.140 / 141 ground combat begins: corporate guards ten days later (75%), if it still
#                 goes on; once per ground combat                                  pe.297 / 298 the offer, haggling
#   pe.299 what to do with the collaborators
namespace = pe

# on_planet_occupied: root = the planet, from = its owner, fromfrom = the occupier.
carrier_event = {
	id = pe.138
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = { is_country_type = default is_gestalt = no }
		count_deposits = { category = pe_deposit_cat_firms count > 0 }
	}
	immediate = {
		save_event_target_as = pe_occ_planet
		pe_occ_levy = yes
		if = {
			limit = { exists = fromfrom check_variable = { which = pe_occ_sum value > 0 } }
			fromfrom = { add_resource = { trade = 1 mult = prev.pe_occ_sum } }
		}
		if = {
			limit = { exists = from from = { is_ai = no } }
			from = { create_message = { type = pe_private_economy localization = pe_msg_occ_levy days = 30 target = event_target:pe_occ_planet variable = { type = name localization = PLANET scope = event_target:pe_occ_planet } } }
		}
	}
}

# on_planet_returned: root = the planet, from = its owner, fromfrom = the former occupier.
carrier_event = {
	id = pe.139
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = { is_country_type = default is_gestalt = no }
		count_deposits = { category = pe_deposit_cat_firms count > 0 }
		is_variable_set = pe_occ_m
		check_variable = { which = pe_occ_m value >= 3 }
	}
	immediate = {
		save_event_target_as = pe_col_planet
		set_variable = { which = pe_occ_m value = 0 }
		pe_collab_sum = yes
		owner = {
			set_variable = { which = pe_col_fine value = prev.pe_col_sum }
			if = { limit = { is_ai = no } country_event = { id = pe.299 } }
			else = { pe_collab_punish = yes }
		}
	}
}

# on_ground_combat_started: root = the colony, from = the attacker.
carrier_event = {
	id = pe.140
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = { is_country_type = default is_gestalt = no }
		exists = from
		NOT = { from = { is_same_value = root.owner } }
		# one check per ground combat: none is waiting here yet
		NOT = { has_planet_flag = pe_gc_wait }
	}
	immediate = {
		if = {
			limit = { owner = { has_country_flag = pe_gc_force } }
			set_planet_flag = pe_gc_wait
			owner = { country_event = { id = pe.141 days = 10 } }
		}
		else = {
			random = {
				chance = 75
				set_planet_flag = pe_gc_wait
				owner = { country_event = { id = pe.141 days = 10 } }
			}
		}
	}
}

# Ten days after a ground combat began: if it still goes on, a corporation helps.
country_event = {
	id = pe.141
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		random_owned_colony = {
			limit = { has_planet_flag = pe_gc_wait has_ground_combat = yes }
			save_event_target_as = pe_gc_planet
			remove_planet_flag = pe_gc_wait
		}
		every_owned_colony = {
			limit = { has_planet_flag = pe_gc_wait has_ground_combat = no }
			remove_planet_flag = pe_gc_wait
		}
		if = {
			limit = { exists = event_target:pe_gc_planet }
			pe_corp_ground = yes
		}
	}
}

# A corporation offers its guards against a ground attack, for a price.
country_event = {
	id = pe.297
	title = pe.297.name
	desc = pe.297.desc
	picture = GFX_evt_ground_combat
	is_triggered_only = yes
	immediate = { hidden_effect = { if = { limit = { check_variable = { which = pe_gc_k value = 1 } } set_variable = { which = pe_cn297 value = pe_c1_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 2 } } set_variable = { which = pe_cn297 value = pe_c2_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 3 } } set_variable = { which = pe_cn297 value = pe_c3_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 4 } } set_variable = { which = pe_cn297 value = pe_c4_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 5 } } set_variable = { which = pe_cn297 value = pe_c5_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 6 } } set_variable = { which = pe_cn297 value = pe_c6_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 7 } } set_variable = { which = pe_cn297 value = pe_c7_name } } else_if = { limit = { check_variable = { which = pe_gc_k value = 8 } } set_variable = { which = pe_cn297 value = pe_c8_name } } } }
	location = event_target:pe_gc_planet

	option = {
		name = pe.297.a
		trigger = { check_variable = { which = pe_gc_ind value > 0 } }
		custom_tooltip = pe.297.a.tt
		hidden_effect = { pe_corp_ground_firms = yes pe_corp_ground_start = yes }
	}
	option = {
		name = pe.297.b
		allow = { check_variable = { which = pe_gc_stock value >= pe_gc_alloys } }
		custom_tooltip = pe.297.b.tt
		hidden_effect = { add_resource = { alloys = -1 mult = pe_gc_alloys } pe_corp_ground_start = yes }
	}
	option = {
		name = pe.297.c
		allow = { has_resource = { type = influence amount >= 100 } }
		custom_tooltip = pe.297.c.tt
		hidden_effect = { add_resource = { influence = -100 } pe_corp_ground_start = yes }
	}
	option = {
		name = pe.297.d
		custom_tooltip = pe.297.d.tt
		hidden_effect = {
			random_list = {
				50 = { set_variable = { which = pe_gc_ok value = 1 } }
				50 = { set_variable = { which = pe_gc_ok value = 0 } }
			}
			country_event = { id = pe.298 }
		}
	}
	option = { name = pe.297.e }
}
