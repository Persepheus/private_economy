# A science or construction ship was lost (on_ship_destroyed_victim: this = owner, fromfrom =
# the ship): remember it for two years (patron donation pe.291).
country_event = {
	id = pe.290
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_country_type = default
		fromfrom = { OR = { is_ship_size = science is_ship_size = constructor } }
	}
	immediate = {
		set_timed_country_flag = { flag = pe_ship_lost days = 720 }
		if = {
			limit = { fromfrom = { is_ship_size = science } }
			set_timed_country_flag = { flag = pe_ship_lost_sci days = 720 }
		}
	}
}
