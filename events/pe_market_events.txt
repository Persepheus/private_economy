# A magnate's farewell
country_event = {
	id = pe.240
	title = pe.240.name
	desc = pe.240.desc
	picture = GFX_evt_city_planet
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_variable = { which = pe_fw_amt value = pe_capital }
			multiply_variable = { which = pe_fw_amt value = 0.15 }
			round_variable = pe_fw_amt
			subtract_variable = { which = pe_capital value = pe_fw_amt }
			set_variable = { which = pe_fw_res value = pe_fw_amt }
			multiply_variable = { which = pe_fw_res value = 0.15 }
			round_variable = pe_fw_res
			set_variable = { which = pe_fw_tax value = pe_fw_amt }
			multiply_variable = { which = pe_fw_tax value = 0.4 }
			round_variable = pe_fw_tax
		}
	}
	option = {
		name = pe.240.a
		add_resource = { physics_research = 1 mult = pe_fw_res }
		add_resource = { society_research = 1 mult = pe_fw_res }
		add_resource = { engineering_research = 1 mult = pe_fw_res }
		capital_scope = { add_modifier = { modifier = pe_farewell days = 360 } }
		ai_chance = { factor = 40 }
	}
	option = {
		name = pe.240.b
		add_resource = { trade = 1 mult = pe_fw_tax }
		ai_chance = { factor = 40 }
	}
	option = {
		name = pe.240.c
		custom_tooltip = pe.240.c.tt
		hidden_effect = {
			change_variable = { which = pe_capital value = pe_fw_amt }
			set_variable = { which = pe_mag_rep value = pe_mag_old_rep }
			divide_variable = { which = pe_mag_rep value = 2 }
			round_variable = pe_mag_rep
		}
		ai_chance = { factor = 20 }
	}
}

# Hostile takeover
country_event = {
	id = pe.243
	title = pe.243.name
	desc = pe.243.desc
	picture = GFX_evt_galactic_senate
	is_triggered_only = yes
	location = event_target:pe_pat_planet

	option = {
		name = pe.243.a
		custom_tooltip = pe.243.a.tt
		hidden_effect = {
			change_variable = { which = pe_capital value = pe_pat_spend }
			# The smallest firm here is bought by one at least as big.
			set_variable = { which = pe_to_min value = 3 }
			set_variable = { which = pe_to_fcap value = 0 }
			event_target:pe_pat_planet = {
				every_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms is_variable_set = pe_stage }
					if = {
						limit = { check_variable = { which = pe_stage value < root.pe_to_min } }
						root = { set_variable = { which = pe_to_min value = prev.pe_stage } }
					}
				}
				random_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms is_variable_set = pe_stage check_variable = { which = pe_stage value = root.pe_to_min } }
					if = { limit = { is_variable_set = pe_fcap } root = { set_variable = { which = pe_to_fcap value = prev.pe_fcap } } }
					remove_deposit = yes
				}
				random_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms }
					change_variable = { which = pe_fcap value = root.pe_to_fcap }
					change_variable = { which = pe_fcap value = 100 }
				}
				check_planet_employment = yes
			}
		}
		ai_chance = { factor = 50 }
	}
	option = {
		name = pe.243.b
		hidden_effect = { change_variable = { which = pe_capital value = pe_pat_spend } }
		event_target:pe_pat_planet = { add_modifier = { modifier = pe_business_upset days = 720 } }
		ai_chance = { factor = 50 }
	}
}

# Stock market crash
country_event = {
	id = pe.248
	title = pe.248.name
	desc = pe.248.desc
	picture = GFX_evt_large_explosion
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			change_variable = { which = pe_capital value = pe_pat_spend }
			multiply_variable = { which = pe_capital value = 0.5 }
			every_owned_colony = {
				every_deposit = {
					limit = { has_deposit_category = pe_deposit_cat_firms }
					multiply_variable = { which = pe_fcap value = 0.5 }
				}
			}
			if = {
				limit = { check_variable = { which = pe_shares value > 0 } }
				divide_variable = { which = pe_shares value = 2 }
				floor_variable = pe_shares
			}
			remove_country_flag = pe_bubble
		}
	}
	option = {
		name = pe.248.a
		allow = { has_resource = { type = trade amount >= 300 } }
		add_resource = { trade = -300 }
		ai_chance = { factor = 50 }
	}
	option = {
		name = pe.248.b
		add_modifier = { modifier = pe_recession days = 360 }
		ai_chance = { factor = 50 }
	}
}
