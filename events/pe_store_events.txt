# Storage contracts and arms orders (pe_store_effects.txt).
#   pe.324  a firm offers to store a full store's surplus
#   pe.325  it offers to extend the contract
#   pe.326  arms order from the alloy store
#   pe.327  the arms order is delivered (hidden)
namespace = pe

country_event = {
	id = pe.324
	title = pe.324.name
	desc = { trigger = { check_variable = { which = pe_sto_r value = 1 } } text = pe.324.desc.mi }
	desc = { trigger = { check_variable = { which = pe_sto_r value = 2 } } text = pe.324.desc.en }
	desc = { trigger = { check_variable = { which = pe_sto_r value = 3 } } text = pe.324.desc.fo }
	desc = { trigger = { check_variable = { which = pe_sto_r value = 4 } } text = pe.324.desc.al }
	desc = { trigger = { check_variable = { which = pe_sto_r value = 5 } } text = pe.324.desc.cg }
	desc = { trigger = { check_variable = { which = pe_sto_r value = 6 } } text = pe.324.desc.tr }
	picture = GFX_evt_tradedeal
	is_triggered_only = yes
	location = event_target:pe_sto_planet

	option = {
		name = pe.324.a
		custom_tooltip = pe.324.a.tt
		hidden_effect = { pe_store_sign = yes }
		ai_chance = { factor = 80 }
	}
	option = {
		name = pe.324.b
		hidden_effect = {
			if = { limit = { check_variable = { which = pe_sto_r value = 1 } } set_timed_country_flag = { flag = pe_sto_cd_mi days = 1800 } }
			else_if = { limit = { check_variable = { which = pe_sto_r value = 2 } } set_timed_country_flag = { flag = pe_sto_cd_en days = 1800 } }
			else_if = { limit = { check_variable = { which = pe_sto_r value = 3 } } set_timed_country_flag = { flag = pe_sto_cd_fo days = 1800 } }
			else_if = { limit = { check_variable = { which = pe_sto_r value = 4 } } set_timed_country_flag = { flag = pe_sto_cd_al days = 1800 } }
			else_if = { limit = { check_variable = { which = pe_sto_r value = 5 } } set_timed_country_flag = { flag = pe_sto_cd_cg days = 1800 } }
			else_if = { limit = { check_variable = { which = pe_sto_r value = 6 } } set_timed_country_flag = { flag = pe_sto_cd_tr days = 1800 } }
		}
		ai_chance = { factor = 20 }
	}
}
