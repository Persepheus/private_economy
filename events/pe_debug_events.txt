# Prototype test events, fired from the console (country scope = your empire):
#   event pe.1   private steelworks with trade division on your capital
#   event pe.2   private physics lab with biology division on your capital
#   event pe.3   private exchange building on your capital
#   event pe.4   remove all firms and the private exchange from your capital
#   event pe.5   steelworks with trade division on every non-gestalt AI capital
#   event pe.6   diagnosis popup: capital, resource scores, the capital's counts and best firm
#   event pe.7   run the quarterly logic now and clear all planet cooldowns first
#   event pe.8   finish all private construction sites now
#   event pe.9   try a firm relocation now (stuck planet, ignores the yearly limit)
#   event pe.10  fire a patron event now
#   event pe.11  fire a labour event now (a planet with empty private worker jobs)
#   event pe.12  offer a supply deal now, also without a shortage (a private firm must make the good)
#   event pe.13  the Command Economy now: every firm becomes a state district
#   event pe.14  found a corporation now (needs an independent stage-3 firm; run pe.7 first)
#   event pe.273 the register of corporations
#   event pe.274 a corporation takes a station now (holdings created if needed)
#   event pe.277 corporate offers now: war lease (also in peace), clearing, exploration charter,
#                convoy protection, salvage, Dyson swarm (each only where its condition holds)
#   event pe.278 end all running corporate fleet contracts (fleets return, no loot)
#   event pe.279 the supply chain step now (run pe.7 first)
#   event pe.134 cede our planet with the most firms (not the capital) to another empire
#   event pe.135 take the ceded planet back (both fire on_colony_transfer like a peace treaty)
namespace = pe

country_event = {
	id = pe.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = {
			add_deposit = d_pe_firm_aaat
			last_added_deposit = { pe_firm_identify = yes set_variable = { which = pe_fcap value = 10 } }
		}
		log = "private_economy: pe.1 added d_pe_firm_aaat to the capital"
	}
}

# event pe.278: end all running corporate fleet contracts (for testing new offers).
country_event = {
	id = pe.278
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_controlled_fleet = {
			limit = { has_fleet_flag = pe_leased }
			if = { limit = { exists = leader } kill_leader = { show_notification = no fire = yes } }
			delete_fleet = this
		}
		every_system = {
			limit = { OR = { has_star_flag = pe_clear@root has_star_flag = pe_convoy@root } }
			remove_star_flag = pe_clear@root
			remove_star_flag = pe_convoy@root
		}
		set_variable = { which = pe_lease_war_q value = 0 }
		set_variable = { which = pe_lease_clear_q value = 0 }
		set_variable = { which = pe_charter_q value = 0 }
		set_variable = { which = pe_convoy_q value = 0 }
		log = "private_economy: pe.278 all corporate fleet contracts ended"
	}
}

# Debug: cede our planet with the most firms (not the capital) to another normal empire.
country_event = {
	id = pe.134
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_variable = { which = pe_dbg_best value = 0 }
		every_owned_colony = {
			limit = { NOT = { is_capital = yes } count_deposits = { category = pe_deposit_cat_firms count > 0 } }
			set_variable = { which = pe_dbg_n value = 0 }
			every_deposit = { limit = { has_deposit_category = pe_deposit_cat_firms } prev = { change_variable = { which = pe_dbg_n value = 1 } } }
			if = {
				limit = { check_variable = { which = pe_dbg_n value > root.pe_dbg_best } }
				root = { set_variable = { which = pe_dbg_best value = prev.pe_dbg_n } }
				save_event_target_as = pe_dbg_planet
			}
		}
		random_country = {
			limit = { is_country_type = default is_gestalt = no NOT = { is_same_value = root } }
			save_event_target_as = pe_dbg_to
		}
		if = {
			limit = { exists = event_target:pe_dbg_planet exists = event_target:pe_dbg_to }
			event_target:pe_dbg_planet = { set_planet_flag = pe_dbg_ceded set_owner = event_target:pe_dbg_to }
			log = "private_economy: pe.134 ceded [event_target:pe_dbg_planet.GetName] ([Root.pe_dbg_best] firms) to [event_target:pe_dbg_to.GetName]"
		}
		else = { log = "private_economy: pe.134 no planet with firms (besides the capital) or no other empire" }
	}
}

# Debug: take the planet ceded by pe.134 back.

	id = pe.135

		}

			limit = { exists = event_target:pe_dbg_planet }
			event_target:pe_dbg_planet = { remove_planet_flag = pe_dbg_ceded set_owner = root }
			log = "private_economy: pe.135 took [event_target:pe_dbg_planet.GetName] back"
		}
	}
}
