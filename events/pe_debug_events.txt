# Prototype test events, fired from the console (country scope = your empire):
#   event pe.1   private steelworks on your capital
#   event pe.2   conglomerate on your capital
#   event pe.3   private exchange building on your capital
#   event pe.4   remove all firms and the private exchange from your capital
#   event pe.5   conglomerate on every non-gestalt AI capital (to watch the AI)
#   event pe.6   diagnosis popup: capital, resource scores, the capital's counts and best firm
#   event pe.7   run the quarterly logic now and clear all planet cooldowns first
namespace = pe

country_event = {
	id = pe.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = { add_deposit = d_pe_firm_aaa }
		log = "private_economy: pe.1 added d_pe_firm_aaa to the capital"
	}
}

country_event = {
	id = pe.2
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = { add_deposit = d_pe_firm_act }
		log = "private_economy: pe.2 added d_pe_firm_act to the capital"
	}
}

country_event = {
	id = pe.3
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = { add_building = pe_building_private_exchange }
		log = "private_economy: pe.3 added pe_building_private_exchange to the capital"
	}
}

country_event = {
	id = pe.4
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = {
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_firms }
				remove_deposit = yes
			}
			if = {
				limit = { has_building = pe_building_private_exchange }
				remove_building = pe_building_private_exchange
			}
		}
		log = "private_economy: pe.4 removed firms and the private exchange from the capital"
	}
}

country_event = {
	id = pe.5
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_playable_country = {
			limit = { is_ai = yes is_country_type = default is_gestalt = no exists = capital_scope }
			capital_scope = { add_deposit = d_pe_firm_act }
		}
		log = "private_economy: pe.5 added d_pe_firm_act to every AI capital"
	}
}

country_event = {
	id = pe.6
	title = pe.6.name
	desc = pe.6.desc
	picture = GFX_evt_tradedeal
	is_triggered_only = yes
	immediate = {
		pe_init = yes
		pe_capital_income = yes
		pe_scores = yes
		pe_compute_values = yes
		capital_scope = {
			pe_planet_counts = yes
			if = {
				limit = { free_district_slots > 0 }
				pe_choose_variant = { MODE = found }
			}
			else = {
				pe_choose_variant = { MODE = buy }
			}
			root = {
				set_variable = { which = pe_diag_districts value = prev.pe_districts }
				set_variable = { which = pe_diag_free value = prev.pe_free }
				set_variable = { which = pe_diag_firms value = prev.pe_firms }
				set_variable = { which = pe_diag_target value = prev.pe_target }
			}
		}
		log = "private_economy: pe.6 share [Root.pe_private_share] districts [Root.pe_diag_districts] free [Root.pe_diag_free] firms [Root.pe_diag_firms] target [Root.pe_diag_target] best [Root.pe_best_variant] value [Root.pe_best] capital [Root.pe_capital] income [Root.pe_capital_income]"
	}
	option = { name = pe.6.a }
}

country_event = {
	id = pe.7
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_owned_planet = { remove_planet_flag = pe_cooldown }
		set_variable = { which = pe_months value = 0 }
		pe_quarterly = yes
		log = "private_economy: pe.7 ran the quarterly logic"
	}
}
