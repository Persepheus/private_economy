# Prototype test events, fired from the console (country scope = your empire):
#   event pe.1   private steelworks on your capital
#   event pe.2   private physics lab on your capital
#   event pe.3   private exchange building on your capital
#   event pe.4   remove all firms and the private exchange from your capital
#   event pe.5   steelworks with trade division on every non-gestalt AI capital
#   event pe.6   diagnosis popup: capital, resource scores, the capital's counts and best firm
#   event pe.7   run the quarterly logic now and clear all planet cooldowns first
#   event pe.8   finish all private construction sites now
namespace = pe

country_event = {
	id = pe.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = {
			add_deposit = d_pe_firm_aa
			last_added_deposit = { set_variable = { which = pe_code value = 1 } set_variable = { which = pe_stage value = 1 } set_variable = { which = pe_fcap value = 10 } }
		}
		log = "private_economy: pe.1 added d_pe_firm_aa to the capital"
	}
}

country_event = {
	id = pe.3
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = { add_building = pe_building_private_exchange }
		log = "private_economy: pe.3 added pe_building_private_exchange to the capital"
	}
}

country_event = {
	id = pe.4
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		capital_scope = {
			every_deposit = {
				limit = { OR = { has_deposit_category = pe_deposit_cat_firms has_deposit_category = pe_deposit_cat_sites } }
				remove_deposit = yes
			}
			remove_planet_flag = pe_site
			check_planet_employment = yes
			if = {
				limit = { has_building = pe_building_private_exchange }
				remove_building = pe_building_private_exchange
			}
		}
		log = "private_economy: pe.4 removed firms and the private exchange from the capital"
	}
}

country_event = {
	id = pe.5
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_playable_country = {
			limit = { is_ai = yes is_country_type = default is_gestalt = no exists = capital_scope }
			capital_scope = {
			add_deposit = d_pe_firm_at
			last_added_deposit = { set_variable = { which = pe_code value = 3 } set_variable = { which = pe_stage value = 1 } set_variable = { which = pe_fcap value = 10 } }
		}
		}
		log = "private_economy: pe.5 added d_pe_firm_at to every AI capital"
	}
}

country_event = {
	id = pe.6
	title = pe.6.name
	desc = pe.6.desc
	picture = GFX_evt_tradedeal
	is_triggered_only = yes
	immediate = {
		pe_init = yes
		pe_census = yes
		pe_market = yes
		pe_job_values = yes
		pe_compute_values = yes
		capital_scope = {
			pe_planet_counts = yes
			pe_planet_prepare = yes
			pe_planet_fill = yes
			root = { set_variable = { which = pe_best_min value = 0 } }
			if = {
				limit = { free_district_slots > 0 }
				pe_choose_variant = { MODE = found }
			}
			else = {
				pe_choose_variant = { MODE = buy }
			}
			root = {
				set_variable = { which = pe_diag_districts value = prev.pe_districts }
				set_variable = { which = pe_diag_free value = prev.pe_free }
				set_variable = { which = pe_diag_firms value = prev.pe_firms }
				set_variable = { which = pe_diag_target value = prev.pe_target }
			}
		}
		log = "private_economy: pe.6 share [Root.pe_private_share] districts [Root.pe_diag_districts] free [Root.pe_diag_free] firms [Root.pe_diag_firms] target [Root.pe_diag_target] best [Root.pe_best_variant] value [Root.pe_best] fill [Root.pe_fill_cur] pool [Root.pe_capital] pool income [Root.pe_pool_income] prices alloys [Root.pe_s_alloys] goods [Root.pe_s_consumer_goods] minerals [Root.pe_s_minerals] energy [Root.pe_s_energy] food [Root.pe_s_food] trade [Root.pe_s_trade] research [Root.pe_s_physics_research] emergency [Root.pe_emergency] amenity value [Root.pe_ent_val] va metallurgist [Root.pe_vaj_pe_private_metallurgist] artisan [Root.pe_vaj_pe_private_artisan] trader [Root.pe_vaj_pe_private_trader] miner [Root.pe_vaj_pe_private_miner] technician [Root.pe_vaj_pe_private_technician] farmer [Root.pe_vaj_pe_private_farmer]"
	}
	option = { name = pe.6.a }
}

country_event = {
	id = pe.7

	is_triggered_only = yes

		pe_monthly_sites = yes

		pe_quarterly = yes
		log = "private_economy: pe.7 ran the quarterly logic"
	}
}

country_event = {
	id = pe.8
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_owned_colony = {
			limit = { has_planet_flag = pe_site }
			every_deposit = {
				limit = { has_deposit_category = pe_deposit_cat_sites }
				set_variable = { which = pe_left value = 1 }
			}
		}
		pe_monthly_sites = yes
		log = "private_economy: pe.8 finished the construction sites"
	}
}
