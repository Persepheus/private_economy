# Patron events: rich private people spend the investor pool (see pe_patron_effects.txt).
#   pe.200 gala            pe.201 gala scandal (follow-up)
#   pe.202 foundation      pe.203 prototype   pe.206 prototype result
#   pe.204 charity         pe.205 recruitment campaign
#   pe.214 prospecting find   pe.215 hydroponics   pe.220 arts   pe.221 league   pe.227 betting scandal
#   pe.222 party donations   pe.223 private colonists   pe.224 university
#   pe.225 space tourism   pe.228 tourist accident   pe.226 disaster relief
# Scope: country. root.pe_pat_spend = what the magnate puts in (already taken from the
# investor pool; declining options pay it back). event_target:pe_pat_planet = the planet.
namespace = pe

# The gala of the century
country_event = {
	id = pe.200
	title = pe.200.name
	desc = { trigger = { check_variable = { which = pe_seen_200 value <= 1 } } text = pe.200.desc }
	desc = { trigger = { check_variable = { which = pe_seen_200 value > 1 } } text = pe.200.desc.again }
	picture = GFX_evt_city_planet
	is_triggered_only = yes
	immediate = { hidden_effect = { change_variable = { which = pe_seen_200 value = 1 } } }
	location = event_target:pe_pat_planet

	option = {
		name = pe.200.a
		event_target:pe_pat_planet = { add_modifier = { modifier = pe_gala days = 1800 multiplier = root.pe_pat_mult } }
		hidden_effect = { change_variable = { which = pe_mag_rep value = 1 } }
		hidden_effect = {
			random = {
				chance = 25
				country_event = { id = pe.201 days = 180 }
			}
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 is_egalitarian = no }
		}
	}
	option = {
		name = pe.200.b
		add_resource = { unity = 1 mult = pe_pat_unity }
		hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 3 is_egalitarian = yes }
			modifier = { factor = 2 is_spiritualist = yes }
		}
	}
	option = {
		name = pe.200.c
		add_resource = { trade = 1 mult = pe_pat_tax }
		hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 is_authoritarian = yes }
		}
	}
}

# A prospecting find
country_event = {
	id = pe.214
	title = pe.214.name
	desc = pe.214.desc
	picture = GFX_evt_engineering_cache
	is_triggered_only = yes
	location = event_target:pe_pat_planet

	immediate = {
		hidden_effect = {
			random_list = {
				1 = { set_variable = { which = pe_find value = 1 } }
				1 = { set_variable = { which = pe_find value = 2 } }
				1 = { set_variable = { which = pe_find value = 3 } }
			}
		}
	}
	# The prospectors mine it themselves: the deposit and, if there is room, a private
	# firm on it at once.
	option = {
		name = pe.214.a
		custom_tooltip = pe.214.a.tt
		hidden_effect = {
		event_target:pe_pat_planet = {
			if = {
				limit = { root = { check_variable = { which = pe_find value = 1 } } }
				add_deposit = d_mineral_fields
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_mineral_fields }
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_mineral_fields }
				if = { limit = { free_district_slots > 0 check_variable = { which = pe_idle value >= 50 } } add_deposit = d_pe_firm_mmmm last_added_deposit = { pe_firm_identify = yes set_variable = { which = pe_fcap value = 10 } set_variable = { which = pe_age value = 0 } } }
			}
			else_if = {
				limit = { root = { check_variable = { which = pe_find value = 2 } } }
				add_deposit = d_geothermal_vent
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_geothermal_vent }
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_geothermal_vent }
				if = { limit = { free_district_slots > 0 check_variable = { which = pe_idle value >= 50 } } add_deposit = d_pe_firm_eeee last_added_deposit = { pe_firm_identify = yes set_variable = { which = pe_fcap value = 10 } set_variable = { which = pe_age value = 0 } } }
			}
			else = {
				add_deposit = d_fertile_lands
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_fertile_lands }
				if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_fertile_lands }
				if = { limit = { free_district_slots > 0 check_variable = { which = pe_idle value >= 50 } } add_deposit = d_pe_firm_ffff last_added_deposit = { pe_firm_identify = yes set_variable = { which = pe_fcap value = 10 } set_variable = { which = pe_age value = 0 } } }
			}
			check_planet_employment = yes
		}
		change_variable = { which = pe_mag_rep value = 1 }
		}
		ai_chance = { factor = 50 }
	}
	# The state takes the find; half of the money comes back, the firms are upset.
	option = {
		name = pe.214.b
		custom_tooltip = pe.214.b.tt
		event_target:pe_pat_planet = { add_modifier = { modifier = pe_business_upset days = 720 } }
		hidden_effect = {
			event_target:pe_pat_planet = {
				if = {
					limit = { root = { check_variable = { which = pe_find value = 1 } } }
					add_deposit = d_mineral_fields
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_mineral_fields }
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_mineral_fields }
				}
				else_if = {
					limit = { root = { check_variable = { which = pe_find value = 2 } } }
					add_deposit = d_geothermal_vent
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_geothermal_vent }
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_geothermal_vent }
				}
				else = {
					add_deposit = d_fertile_lands
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 2 } } } add_deposit = d_fertile_lands }
					if = { limit = { root = { check_variable = { which = pe_find_n value >= 3 } } } add_deposit = d_fertile_lands }
				}
			}
		}
		hidden_effect = {
			change_variable = { which = pe_capital value = pe_pat_half }
			change_variable = { which = pe_mag_rep value = -1 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 2 is_authoritarian = yes }
		}
	}
	option = {
		name = pe.214.c
		add_resource = { unity = 1 mult = pe_pat_art }
		hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		ai_chance = {
			factor = 20
			modifier = { factor = 3 is_spiritualist = yes }
		}
	}
}
