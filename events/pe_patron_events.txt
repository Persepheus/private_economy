# Patron events: rich private people spend the investor pool (see pe_patron_effects.txt).
#   pe.200 gala            pe.201 gala scandal (follow-up)
#   pe.202 foundation      pe.203 prototype   pe.206 prototype result
#   pe.204 charity         pe.205 recruitment campaign
# Scope: country. root.pe_pat_spend = what the magnate puts in (already taken from the
# investor pool; declining options pay it back). event_target:pe_pat_planet = the planet.
namespace = pe

# The gala of the century
country_event = {
	id = pe.200
	title = pe.200.name
	desc = pe.200.desc
	picture = GFX_evt_city_planet
	is_triggered_only = yes
	location = event_target:pe_pat_planet

	option = {
		name = pe.200.a
		event_target:pe_pat_planet = { add_modifier = { modifier = pe_gala days = 1800 } }
		hidden_effect = { change_variable = { which = pe_mag_rep value = 1 } }
		hidden_effect = {
			random = {
				chance = 25
				country_event = { id = pe.201 days = 180 }
			}
		}
		ai_chance = {
			factor = 50
			modifier = { factor = 2 is_egalitarian = no }
		}
	}
	option = {
		name = pe.200.b
		add_resource = { unity = 1 mult = pe_pat_unity }
		hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 3 is_egalitarian = yes }
			modifier = { factor = 2 is_spiritualist = yes }
		}
	}
	option = {
		name = pe.200.c
		add_resource = { trade = 1 mult = pe_pat_tax }
		hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 is_authoritarian = yes }
		}
	}
}

# The prototype: result
country_event = {
	id = pe.206
	title = pe.206.name
	desc = { trigger = { check_variable = { which = pe_proto_result value = 1 } } text = pe.206.desc.success }
	desc = { trigger = { check_variable = { which = pe_proto_result value = 2 } } text = pe.206.desc.partial }
	desc = { trigger = { check_variable = { which = pe_proto_result value = 3 } } text = pe.206.desc.failure }
	picture = GFX_evt_engineering_bay
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { check_variable = { which = pe_proto_result value = 1 } }
			add_resource = { engineering_research = 1 mult = pe_pat_proto_full }
			hidden_effect = { change_variable = { which = pe_mag_rep value = 1 } }
		}
		else_if = {
			limit = { check_variable = { which = pe_proto_result value = 2 } }
			add_resource = { engineering_research = 1 mult = pe_pat_proto_part }
		}
		else = {
			capital_scope = { add_modifier = { modifier = pe_prototype_accident days = 720 } }
			hidden_effect = { change_variable = { which = pe_mag_rep value = -1 } }
		}
	}
	option = { name = pe.206.a }
}
