# Private economy runtime events.
#   pe.100  monthly pulse (on_monthly_pulse_country): construction sites advance monthly,
#           the private economy decides every third month
#   pe.110  Free Market: a firm offers to buy a state district (human empires only)
namespace = pe

country_event = {
	id = pe.100
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_country_type = default
		is_gestalt = no
		exists = capital_scope
	}
	immediate = {
		pe_init = yes
		pe_monthly_sites = yes
		pe_deal_month = yes
		pe_chab_monthly = yes
		pe_sprawl_step = yes
		pe_corp_defence = yes
		pe_bombard_step = yes
		pe_occupation_step = yes
		pe_guard_cleanup = yes
		change_variable = { which = pe_months value = 1 }
		if = {
			limit = { check_variable = { which = pe_months value >= 3 } }
			set_variable = { which = pe_months value = 0 }
			pe_quarterly = yes
		}
	}
}

# Scope: country. event_target:pe_offer_planet = the planet, pe_offer_variant = firm variant code,
# pe_offer_class = 1 city, 2 mining, 3 generator, 4 farming.
country_event = {
	id = pe.110
	title = pe.110.name
	desc = { trigger = { NOT = { check_variable = { which = pe_offer_hab value = 1 } } OR = { check_variable = { which = pe_offer_class value = 1 } check_variable = { which = pe_offer_class value = 5 } } } text = pe.110.desc.urban }
	desc = { trigger = { NOT = { check_variable = { which = pe_offer_hab value = 1 } } check_variable = { which = pe_offer_class value > 1 } check_variable = { which = pe_offer_class value < 5 } } text = pe.110.desc.rural }
	desc = { trigger = { check_variable = { which = pe_offer_hab value = 1 } OR = { check_variable = { which = pe_offer_class value = 1 } check_variable = { which = pe_offer_class value = 5 } } } text = pe.110.desc.urban_hab }
	desc = { trigger = { check_variable = { which = pe_offer_hab value = 1 } check_variable = { which = pe_offer_class value > 1 } check_variable = { which = pe_offer_class value < 5 } } text = pe.110.desc.rural_hab }
	picture = GFX_evt_tradedeal
	is_triggered_only = yes
	location = event_target:pe_offer_planet

	option = {
		name = pe.110.a
		event_target:pe_offer_planet = { pe_buyout_by_offer = yes }
		remove_country_flag = pe_offer_pending
	}
	option = {
		name = pe.110.b
		event_target:pe_offer_planet = {
			add_modifier = { modifier = pe_buyout_refused days = 720 }
		}
		remove_country_flag = pe_offer_pending
	}
}

# A new ascension perk, tradition, policy or technology may change the empire size modifiers:
# the empire size of the private economy is recounted at once (pe_sprawl_step).
country_event = {
	id = pe.36
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_country_type = default
		is_gestalt = no
		OR = { has_modifier = pe_sprawl_firms has_modifier = pe_sprawl_corp_hab has_modifier = pe_sprawl_corp_hab_d }
	}
	immediate = { pe_sprawl_step = yes }
}
